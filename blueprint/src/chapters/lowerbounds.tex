\chapter{Circuit lower bounds and proof complexity}
\label{chap:lowerbounds}

This chapter collects unconditional lower bounds for concrete computational
models, together with the proof-complexity side of the same program. The
library proves Shannon's counting bounds, which determine the maximum circuit
complexity of an $N$-bit function up to a constant factor, the essential-input
bound and Schnorr's $2N - 1$ bound for parity, Valiant's depth-reduction lemma,
Spira's formula balancing, the monotone Karchmer--Wigderson correspondence, and
the soundness of resolution. For $\cc{AC}^0$ it proves a finite, width-sensitive
switching lemma, iterates it through arbitrary depth-bounded formulas, and
derives a finite counting obstruction for every formula computing parity. All
of these results are finite or nonuniform. The next target is the family-level
theorem that parity is not in $\cc{AC}^0$ as defined in this library. Circuits
are already normalized to formulas
(Theorem~\ref{thm:lowerbounds-ac0-normalization}), so what remains is
arithmetic on top of the finite obstruction (roadmap track L4). Beyond it, the
plan covers monotone circuit lower bounds, pseudorandom restrictions, and
resolution size and width lower bounds leading to the pigeonhole principle and
to Cook--Reckhow proof systems (roadmap track L5).
Section~\ref{sec:lowerbounds-algebraic} collects the results of the imported
algebraic-circuits library: parity bounds in the De Morgan basis and in its own
$\cc{AC}^0$ model (including parity outside that model's $\cc{AC}^0$ and an
explicit Håstad-type size bound), monotone CLIQUE at explicit parameters,
Nechiporuk's formula bound, Karchmer--Wigderson, a circuit-size hierarchy,
conditional complexity, algebraic lower bounds, and mass production; the
imported sunflower lemma appears in Section~\ref{sec:lowerbounds-monotone}. Its
circuit results use CSLib's circuits, each over the imported library's own
signature and cost model; Nechiporuk's bound and Karchmer--Wigderson are about
formulas, and the Waring bound is an identity of polynomials.

\section{Counting bounds}
\label{sec:lowerbounds-counting}

\begin{theorem}[Shannon lower bound]
  \label{thm:lowerbounds-shannon-lower}
  \lean{Complexity.shannon_lower_bound_circuit, Complexity.shannon_sizeComplexity}
  \leanok
  \uses{def:circuit, def:circuits-bases, def:circuits-size-complexity}
  For every $N \ge 6$ there is a function $f : \{0,1\}^N \to \{0,1\}$ that no
  fan-in-two AND/OR circuit of size at most $\lfloor 2^N / (5N) \rfloor$
  computes, where the size $G + 1$ of a single-output circuit counts its output
  gate. Consequently $\mathrm{size}(f) > \lfloor 2^N / (5N) \rfloor$.
\end{theorem}
\begin{proof}
  \leanok
  Encode circuits of a given size by short descriptors and compare their
  number with the $2^{2^N}$ functions.
\end{proof}

\begin{theorem}[Shannon upper bound]
  \label{thm:lowerbounds-shannon-upper}
  \lean{Complexity.shannon_upper_bound}
  \leanok
  \uses{def:circuits-size-complexity}
  For every $N \ge 16$, every $f : \{0,1\}^N \to \{0,1\}$ satisfies
  $\mathrm{size}(f) \le \lfloor 18 \cdot 2^N / N \rfloor$ over the fan-in-two
  AND/OR basis.
\end{theorem}
\begin{proof}
  \leanok
  Select among the columns of the truth table with a multiplexer, reading each
  column from a library that contains every function of the remaining
  variables.
\end{proof}

\begin{theorem}[Lupanov]
  \label{thm:lowerbounds-lupanov}
  \lean{Complexity.lupanov_sizeComplexity}
  \leanok
  \uses{def:circuits-size-complexity, thm:circuits-cslib}
  For every $\varepsilon > 0$ and all sufficiently large $N$, every
  $f : \{0,1\}^N \to \{0,1\}$ satisfies
  $\mathrm{size}(f) \le (1 + \varepsilon)\, 2^N / N$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:circuits-cslib}
  CSLib proves the bound for its De Morgan circuits by refining the column
  library: the rows of the truth table are grouped into strips, and each
  strip's column patterns are computed once and shared. A CSLib circuit with
  $g$ gates is a fan-in-two AND/OR circuit of size $g + 1$, and the extra gate
  is absorbed by the slack in $\varepsilon$.
\end{proof}

\begin{theorem}[Shannon lower bound via CSLib]
  \label{thm:lowerbounds-shannon-cslib}
  \lean{Complexity.exists_sizeComplexity_gt_cslib}
  \leanok
  \uses{def:circuits-size-complexity, thm:circuits-cslib}
  For all sufficiently large $N$, some $f : \{0,1\}^N \to \{0,1\}$ satisfies
  $2^N / N < N + 2\,\mathrm{size}(f)$, that is,
  $\mathrm{size}(f) > (2^N / N - N) / 2$, with $2^N / N$ a real quotient.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:circuits-cslib}
  CSLib's counting argument gives a function that no De Morgan circuit with at
  most $2^N / N$ gates computes, and a fan-in-two AND/OR circuit of size $s$ is
  a CSLib circuit with at most $N + 2s$ gates.
\end{proof}

\begin{theorem}[Circuit size of languages]
  \label{thm:lowerbounds-size-classes}
  \lean{Complexity.mem_SIZE_iff_sliceSizeComplexity_le,
    Complexity.exists_cslib_of_mem_SIZE, Complexity.mem_SIZE_of_cslib,
    Complexity.lupanov_sliceSizeComplexity, Complexity.exists_mem_SIZE_lupanov,
    Complexity.exists_mem_SIZE_bigO_two_pow_div,
    Complexity.exists_language_not_mem_SIZE,
    Complexity.exists_language_not_mem_SIZE_littleO}
  \leanok
  \uses{def:SIZE, thm:lowerbounds-lupanov, thm:circuits-cslib}
  Let the slice complexity of a language $L$ at length $n \ge 1$ be the size
  complexity of its length-$n$ slice, and $0$ at $n = 0$, where a circuit
  family answers with a stored bit. Then $L \in \cc{SIZE}(s)$ exactly when the
  slice complexity is at most $s(n)$ at every $n$. If $L \in \cc{SIZE}(s)$, then
  every length-$n$ slice, $n = 0$ included, is decided by a CSLib De Morgan
  circuit with at most $n + 2s(n) + 1$ gates; conversely, CSLib circuits with at
  most $s(n)$ gates at every length $n \ge 1$ put $L$ in $\cc{SIZE}(s + 1)$.

  For every $\varepsilon > 0$ there is one threshold $N_0$ such that every
  language has slice complexity at most $(1 + \varepsilon) 2^n / n$ at every
  length $n \ge N_0$. Hence every language is in $\cc{SIZE}(s)$ for some $s$
  with $s(n) \le (1 + \varepsilon) 2^n / n$ for all large $n$, and for some
  $s = O(\lfloor 2^n / n \rfloor)$. Conversely, a single language lies outside
  $\cc{SIZE}(s)$ for every $s$ with $n + 2s(n) \le 2^n / n$ for all large $n$,
  in particular for every $s = o(\lfloor 2^n / n \rfloor)$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:lowerbounds-lupanov, thm:circuits-cslib}
  Apply Lupanov's bound to every slice. A big-O bound only constrains large
  lengths, so Lupanov's bound with $\varepsilon = 1$ gives
  $O(\lfloor 2^n / n \rfloor)$, the floor costing at most a factor of two. For
  the lower bound,
  CSLib's family-level Shannon bound gives a language with
  $2^n / n < n + 2\,\mathrm{size}$ at every large length after translating
  circuits as in Theorem~\ref{thm:circuits-cslib}; a size bound $s$ with
  $n + 2s(n) \le 2^n / n$ for large $n$ contradicts this at some length.
\end{proof}

\section{Results from the algebraic-circuits library}
\label{sec:lowerbounds-algebraic}

These results come from the algebraic-circuits library, imported wholesale
under \texttt{Complexitylib/Algebraic}, and keep the \texttt{Algebraic}
namespace. The circuit results are stated for CSLib's straight-line circuits
(\texttt{Cslib.Circuits.Circuit}), but each over the imported library's own
signature and cost model, not over CSLib's De Morgan signature or the bases of
this library. For example, its De Morgan signature adds an identity gate to
constants, negation, conjunction, and disjunction; the De Morgan parity bound
charges only conjunctions and disjunctions, while the hierarchy counts every
gate; its $\cc{AC}^0$ circuits have unbounded fan-in, with negations free and
adding no depth; mass production charges negations, conjunctions, and
disjunctions but not constants or identities. Each node names its model and
what it counts. Nechiporuk's bound and Karchmer--Wigderson are stated for the
imported library's formula types, not for circuits, and the Waring bound is an
identity of polynomials with no circuit in its statement. Conditional complexity is defined in this library,
although it lives in CSLib's namespace. The consolidation plan in
\texttt{ROADMAP.md} relates these results to the statements elsewhere in this
chapter.

\begin{theorem}[Parity in the De Morgan basis]
  \label{thm:lowerbounds-algebraic-xor}
  \lean{Algebraic.DeMorgan.xor_lowerBound}
  \leanok
  For every $n$, every circuit over the imported library's De Morgan signature
  computing $n$-input parity has at least $3(n - 1)$ AND and OR gates; constant,
  identity, and negation gates are free.
\end{theorem}
\begin{proof}
  \leanok
  Gate elimination: fixing a well-chosen input removes at least three AND or
  OR gates while leaving a parity function on one fewer input.
\end{proof}

\begin{theorem}[Parity is not in $\cc{AC}^0$]
  \label{thm:lowerbounds-algebraic-parity-ac0}
  \lean{Algebraic.AC0.parity_not_computable, Algebraic.AC0.parity_not_raw_computable}
  \leanok
  No nonuniform family of unbounded fan-in AND/OR circuits with negations at
  the inputs, with polynomially many AND and OR gates (negations free) and
  constant AND/OR depth (negations add no depth), computes parity at every
  input length. The same holds when negation gates may occur anywhere in the
  circuits. This is stated for the imported library's own $\cc{AC}^0$ model;
  deriving Theorem~\ref{thm:lowerbounds-parity-not-ac0} for $\cc{AC}^0$ as
  defined here from it is part of the consolidation plan.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:lowerbounds-algebraic-hastad}
  Håstad's switching lemma: random restrictions collapse each bottom layer
  of such a circuit to a shallow decision tree, reducing the depth, while
  parity restricts to parity on the remaining inputs. At input length
  $(20(t + 1))^{d - 1}$ this forces $2^{t + 1} \le 20\,t\,S$ for a depth-$d$
  circuit with $S$ AND and OR gates, which a polynomial bound on $S$ violates
  for large $t$.
\end{proof}

\begin{theorem}[Håstad's size bound, imported model]
  \label{thm:lowerbounds-algebraic-hastad}
  \lean{Algebraic.AC0.Circuit.parity_size_tradeoff_at_scale_raw,
    Algebraic.AC0.Circuit.parity_andOrCost_lower_bound_at_root_raw}
  \leanok
  Consider circuits of the imported library's $\cc{AC}^0$ model: unbounded
  fan-in AND and OR gates and negation gates anywhere, where $S$ counts the AND
  and OR gates and the depth counts AND and OR levels only. Let $d \ge 2$ and
  $t \ge 1$. Every such circuit with $n \ge (20(t + 1))^{d - 1}$ inputs and
  depth at most $d$ that computes $n$-input parity satisfies
  $2^{t + 1} \le 20\,t\,S$. In particular, if $n \ge 40^{d - 1}$ and
  $q = \lfloor \lfloor n^{1/(d-1)} \rfloor / 20 \rfloor$, with
  $\lfloor n^{1/(d-1)} \rfloor$ the integer root, then
  $S \ge \lfloor 2^q / (20(q - 1)) \rfloor$. This bound is of order
  $2^{\Omega(n^{1/(d-1)})}$, but no asymptotic form is stated. It is not
  transferred to the circuits of this library; see
  Theorem~\ref{thm:lowerbounds-hastad-size}.
\end{theorem}
\begin{proof}
  \leanok
  Apply $d - 1$ rounds of random restrictions. Each round switches one layer,
  with a union bound over the $S$ AND and OR gates, while an explicit integer
  schedule keeps enough inputs free; the unreduced top gate is then a
  bounded-width normal form, which cannot compute parity on the surviving
  inputs. The switching condition is equivalent to $20\,t\,S < 2^{t + 1}$, and
  $n \ge (20(t + 1))^{d - 1}$ implies the survivor condition.
\end{proof}

\begin{theorem}[Monotone CLIQUE]
  \label{thm:lowerbounds-algebraic-clique}
  \lean{Algebraic.Monotone.Clique.Exponential.twoPow_lt_circuitSize}
  \leanok
  Let $t \ge 4$ and $N = 2^{20t}$. Every circuit over the imported library's
  binary AND/OR signature (fan-in-two AND and OR gates, no constants and no
  negations, every gate counted) whose inputs are the $\binom{N}{2}$ edge
  variables of an $N$-vertex graph and which computes $2^{4t}$-CLIQUE has more
  than $2^{t \cdot 2^t}$ gates.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:lowerbounds-sunflower}
  Razborov's method of approximations, with the sunflower lemma bounding the
  error introduced at each gate.
\end{proof}

\begin{theorem}[Nechiporuk]
  \label{thm:lowerbounds-algebraic-nechiporuk}
  \lean{Algebraic.Nechiporuk.eventually_sq_le_leaves}
  \leanok
  Suppose that, for all large $n$, the length-$n$ member of a family of
  Boolean functions is $K(n)$-rectangle-free (under every split of its inputs,
  every rectangle of accepted inputs has a side with fewer than $K(n)$
  elements), with $K(n) \le n^c$ for a natural number $c$, and has at least
  $2^{n - 2}$ accepting inputs. Then for all large $n$ every formula over the
  full binary basis (all sixteen binary gates, constant leaves allowed)
  computing the length-$n$ member satisfies
  $n^2 \le 64(c + 3)\log_2 n \cdot L$, where $L$ counts variable leaves and
  constants are free; that is, it has at least $n^2 / (64(c + 3)\log_2 n)$
  variable leaves. This is a conditional statement about any such family: no
  explicit rectangle-free family is formalized.
\end{theorem}
\begin{proof}
  \leanok
  Count subfunctions on each block of a partition of the inputs;
  rectangle-freeness forces the maximal number on every block at once.
\end{proof}

\begin{theorem}[Karchmer--Wigderson]
  \label{thm:lowerbounds-algebraic-kw}
  \lean{Algebraic.KW.formulaDepth_eq_protocolDepth, Algebraic.KW.formulaSize_eq_protocolSize}
  \leanok
  Consider the imported library's De Morgan formulas, trees of literal leaves
  $x_i$ and $\neg x_i$, constant leaves, and binary AND and OR gates, with depth
  the length of a longest root-to-leaf path and size the number of leaves,
  constants included. In the Karchmer--Wigderson game of $f$, Alice holds $x$
  with $f(x) = 1$, Bob holds $y$ with $f(y) = 0$, and a deterministic protocol
  must output a coordinate where $x$ and $y$ differ. For every $f$ on $n \ge 1$
  inputs, the minimum depth of a formula computing $f$ equals the minimum
  depth of a protocol for this game, and the minimum size of a formula equals
  the minimum number of leaves of a protocol. The hypothesis $n \ge 1$ is used
  because a protocol leaf names an input position; the fact that no protocol
  exists at $n = 0$ is a remark, not a formalized statement.
\end{theorem}
\begin{proof}
  \leanok
  Formulas and protocols translate into each other gate for move: an AND gate
  is a move of the player holding a rejected input, an OR gate a move of the
  player holding an accepted input.
\end{proof}

\begin{theorem}[Polynomial circuit size hierarchy]
  \label{thm:lowerbounds-algebraic-hierarchy}
  \lean{Algebraic.DeMorgan.polynomialSize_ssubset}
  \leanok
  For real exponents $1 \le a < b$, the class of Boolean function families,
  one function for each input length, whose De Morgan circuit complexity in
  the imported library's signature (every gate counted, constants and
  identities included) is at most $C \lfloor n^a \rfloor$ for some constant $C$
  and all large $n$ is strictly contained in the corresponding class for
  $\lfloor n^b \rfloor$. The families are nonuniform and no explicit separating
  family is given.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:lowerbounds-algebraic-lipschitz}
  At each length walk from a constant function towards a function of
  complexity above $2^n / n$, which Shannon's counting bound supplies, one
  truth-table entry at a time. Each step changes the complexity by at most
  $2n$, so some function has complexity in
  $(\lfloor n^b \rfloor - 2n, \lfloor n^b \rfloor]$. The resulting family is in
  the larger class, and since $\lfloor n^b \rfloor - 2n$ eventually exceeds
  every constant multiple of $\lfloor n^a \rfloor$, not in the smaller one.
\end{proof}

\begin{lemma}[Hamming-Lipschitz circuit complexity]
  \label{lem:lowerbounds-algebraic-lipschitz}
  \lean{Algebraic.DeMorgan.complexity_dist_le, Algebraic.DeMorgan.exists_complexity_between}
  \leanok
  Let $C(f)$ be the minimum number of gates of a single-output circuit over the
  imported library's De Morgan signature computing $f : \{0,1\}^n \to \{0,1\}$,
  every gate counted, constants and identities included. Then
  $\lvert C(f) - C(g) \rvert \le 2n \cdot d_H(f, g)$, where $d_H$ is the
  number of truth-table entries on which $f$ and $g$ differ. Consequently, if
  $t \ge 1$ and $t < C(h)$ for some $h$, then some $f$ has
  $t < C(f) \le t + 2n$.
\end{lemma}
\begin{proof}
  \leanok
  Patching one truth-table entry costs at most $2n$ gates, by combining the old
  circuit with an indicator of the changed point. Walk from a constant function
  to a hard one entry by entry.
\end{proof}

\paragraph{Sparse corrections in CSLib's circuit model.}
The following results use the native CSLib De Morgan signature, counting
AND, OR, NOT, and constant gates, with free fan-out and output designation.
They refine the entry-by-entry estimate above by sharing the support and
the labels of a whole correction. The synthesis methods are classical:
see A.~V.~Chashkin, \emph{On computing partial Boolean functions} (in Russian),
\emph{Mathematical Problems of Cybernetics} 22 (2024), pp.~152--222,
\href{https://doi.org/10.20948/mvk-2024-152}{Sections 2--3}, which credit
L.~A.~Sholomov for partial-function synthesis and O.~B.~Lupanov for
bounded-weight synthesis. The formalization proves the needed specialization
from finite pattern banks and affine hashing.

\begin{definition}[Rows and active outputs of a correction]
  \label{def:lowerbounds-correction}
  \lean{Cslib.Circuits.Boolean.Correction.errorSupport,
    Cslib.Circuits.Boolean.Correction.rowDistance,
    Cslib.Circuits.Boolean.Correction.activeOutputs,
    Cslib.Circuits.Boolean.Correction.errorVector}
  \leanok
  For $F,G:\{0,1\}^n\to\{0,1\}^m$, let
  $S=\{x:F(x)\ne G(x)\}$, $d=|S|$, and
  $R=\{j:F_j\not\equiv G_j\}$, with $r=|R|$.
  The error vector $E_R=(F\oplus G)|_R$ contains only the selected output
  coordinates. Write $C^S(E_R)$ for the minimum size of a circuit agreeing
  with $E_R$ on $S$; its values outside $S$ are unrestricted.
\end{definition}

\begin{theorem}[Finite relative correction bound]
  \label{thm:lowerbounds-relative-correction}
  \lean{Cslib.Circuits.Boolean.Correction.complexityGiven_le_of_cover,
    Cslib.Circuits.Boolean.Correction.complexity_dist_le_of_cover,
    Cslib.Circuits.Boolean.Correction.complexity_dist_le_sum_of_cover}
  \leanok
  \uses{def:lowerbounds-correction}
  Let $S$ contain every row on which $F$ and $G$ differ, and let $R$ contain
  every output coordinate that changes. With the original input and $G(x)$
  supplied for free,
  \[
    C(F\mid G)\le C(1_S)+C^S(E_R)+5r,
    \qquad
    |C(F)-C(G)|\le C(1_S)+C^S(E_R)+5r.
  \]
  Moreover $C^S(E_R)\le\sum_{j\in R}C^S((E_R)_j)$.
\end{theorem}
\begin{proof}
  \leanok
  Build the support indicator and the partial label circuit once. Mask each
  selected label with the support bit, then XOR it with the corresponding
  output of $G$. This costs one AND and four further gates per selected
  output; the other outputs reuse their old wires. Exchange $F,G$ for the
  reverse inequality. Scalar label circuits can be appended with shared
  inputs.
\end{proof}

\begin{theorem}[Uniform sparse and partial synthesis at square-root weight]
  \label{thm:lowerbounds-square-root-synthesis}
  \lean{Complexity.CircuitSparseSynthesis.complexity_indicator_le,
    Complexity.CircuitSparseSynthesis.complexityOn_le}
  \leanok
  \uses{def:randomness-hash}
  For every $\varepsilon>0$, for all sufficiently large integers $p$,
  uniformly over all $S\subseteq\{0,1\}^{2p}$ with $|S|\le 2^p$ and all
  Boolean functions $u$,
  \[
    C(1_S)\le(1+\varepsilon)2^p,
    \qquad
    C^S(u)\le(1+\varepsilon)\frac{2^p}{p}.
  \]
\end{theorem}
\begin{proof}
  \leanok
  Share column and row minterms and a bank of short patterns. Sparse rows
  are partitioned into tuples of true columns. For partial rows, a greedy
  covering argument supplies a bank extending every assignment on at most
  $K$ positions; interval masks protect the specified values of other chunks.
  The leading assembly cost is one OR per chunk. Affine hashing reduces the
  input width at polynomial cost. Repeated membership filters synthesize
  $1_S$, while one hash and explicit collision repairs suffice for partial
  labels. Choose parameters with an exponential gap for every secondary
  table and repair term, and absorb the polynomial factors in that gap.
\end{proof}

\begin{theorem}[Uniform square-root continuity for vector circuits]
  \label{thm:lowerbounds-square-root-correction}
  \lean{Complexity.CircuitSparseSynthesis.complexity_dist_le_three_sqrt,
    Complexity.CircuitSparseSynthesis.complexity_dist_le_three_sqrt_of_cover}
  \leanok
  \uses{thm:lowerbounds-relative-correction,thm:lowerbounds-square-root-synthesis}
  For every $\varepsilon>0$, there exists $p_0$ such that for all $p\ge p_0$,
  every output width $m$, and all
  $F,G:\{0,1\}^{2p}\to\{0,1\}^m$ with at most $2^p$ differing input rows
  and at most $2p$ active output coordinates,
  \[
    |C(F)-C(G)|\le(3+\varepsilon)2^p.
  \]
  The threshold is independent of $m,F,G$. The same bound holds when only
  covering sets for the erroneous rows and active outputs are supplied.
\end{theorem}
\begin{proof}
  \leanok
  Apply the finite correction bound. The support costs $(1+o(1))2^p$;
  at most $2p$ scalar partial labels cost a total of $(2+o(1))2^p$.
  The $5r\le 10p$ correction gates are lower order. The proof keeps the
  estimates uniform before converting them to the displayed epsilon bound.
\end{proof}

These results do not assert the unrestricted vector entropy formula
$H/\log_2 H+O(n+r)$ for arbitrary $n,r,d$. In particular, counting possible
error maps alone is not an upper-bound synthesis theorem. The theorem above
establishes the proposed coefficient-three regime without that assumption.

\begin{theorem}[Scalar square-root continuity with leading constant one]
  \label{thm:lowerbounds-scalar-square-root-correction}
  \lean{Cslib.Circuits.Boolean.Correction.complexity_dist_le_indicator,
    Complexity.CircuitSparseSynthesis.complexity_dist_le_sqrt}
  \leanok
  \uses{def:lowerbounds-correction,thm:lowerbounds-square-root-synthesis}
  For scalar functions and $S=\{x:f(x)\ne g(x)\}$,
  \[
    |C(f)-C(g)|\le C(1_S)+4.
  \]
  Consequently, for every $\varepsilon>0$ and all sufficiently large $p$,
  uniformly over scalar functions on $2p$ bits with at most $2^p$ errors,
  \[
    |C(f)-C(g)|\le(1+\varepsilon)2^p.
  \]
\end{theorem}
\begin{proof}
  \leanok
  Since $f\oplus g=1_S$, append a circuit for $1_S$ and four XOR gates.
  Exchange $f,g$, apply sparse support synthesis, and absorb the four gates
  into the epsilon slack.
\end{proof}

\begin{corollary}[Exact hardness forces row errors]
  \label{cor:lowerbounds-sparse-hardness-transfer}
  \lean{Complexity.CircuitSparseSynthesis.rowDistance_gt_of_complexity_gt,
    Complexity.CircuitSparseSynthesis.scalar_rowDistance_gt_of_complexity_gt}
  \leanok
  \uses{thm:lowerbounds-square-root-correction,
    thm:lowerbounds-scalar-square-root-correction}
  For every $\varepsilon>0$ and all sufficiently large $p$, let
  $F:\{0,1\}^{2p}\to\{0,1\}^m$ with $m\le2p$. If
  \[
    C(F)>s+(3+\varepsilon)2^p,
  \]
  then every circuit with at most $s$ gates disagrees with $F$ on more than
  $2^p$ input rows. For a scalar function, replace $3$ by $1$.
  The thresholds are uniform in the functions and in $s$.
\end{corollary}
\begin{proof}
  \leanok
  A size-$s$ circuit defines a function $G$ with $C(G)\le s$. If there
  were at most $2^p$ erroneous rows, the continuity bound would contradict
  the assumed lower bound. The resulting error fraction is greater than
  $2^{-p}$; this statement does not give a constant error fraction.
\end{proof}

\begin{corollary}[Vanishing relative cost of sparse edits]
  \label{cor:lowerbounds-sparse-relative-stability}
  \lean{Complexity.CircuitSparseSynthesis.complexity_dist_isLittleO}
  \leanok
  \uses{thm:lowerbounds-square-root-correction}
  Let $F_p,G_p$ be vector functions on $2p$ bits, with arbitrary common
  output width depending on $p$. Suppose that eventually they differ on at
  most $2^p$ rows and at most $2p$ output coordinates. If
  $2^p=o(C(F_p))$, then
  \[
    |C(F_p)-C(G_p)|=o(C(F_p)).
  \]
\end{corollary}
\begin{proof}
  \leanok
  The uniform continuity theorem with $\varepsilon=1$ gives
  $|C(F_p)-C(G_p)|=O(2^p)$. Compose this estimate with the assumed little-o
  relation. No structure or computability assumption on the edits is needed.
\end{proof}

\begin{theorem}[Optimal order of the sparse correction cost]
  \label{thm:lowerbounds-sparse-correction-sharpness}
  \lean{Complexity.CircuitSparseSynthesis.exists_sparse_complexity_gt,
    Complexity.CircuitSparseSynthesis.exists_rowDistance_eq_complexity_dist_ge,
    Complexity.CircuitSparseSynthesis.not_isLittleO_of_uniform_correction_bound}
  \leanok
  \uses{def:lowerbounds-correction}
  For every $p\ge4$ there exists $S\subseteq\{0,1\}^{2p}$ with $|S|=2^p$
  and $C(1_S)>2^p/16$. There is also a scalar $f$ at exactly $2^p$ errors
  from zero satisfying
  \[
    |C(f)-C(0)|\ge 2^p/16.
  \]
  Consequently, any real-valued function $b(p)$ eventually bounding the
  complexity difference of every scalar pair at distance at most $2^p$
  fails to satisfy $b(p)=o(2^p)$.
\end{theorem}
\begin{proof}
  \leanok
  Apply Shannon's circuit-counting argument to graph indicators. The
  $2^{p2^p}$ maps $\{0,1\}^p\to\{0,1\}^p$ have distinct graphs, each of
  size $2^p$. CSLib's De Morgan count, even after discarding its factorial
  saving, bounds the number of functions with at most $s$ gates by
  \[
    (s+1)\bigl(5(2p+s+1)^2\bigr)^s(2p+s).
  \]
  At $s=2^{p-4}$ and $p\ge4$ this is strictly less than $2^{p2^p}$.
  A missing graph gives the first assertion. Since $C(0)\le1$ and circuit
  sizes are integers, the same witness gives the complexity-distance bound.
  A hypothetical little-o modulus is eventually at most $2^p/32$, a
  contradiction. The witnesses are nonconstructive, and the constant
  $1/16$ is not asserted to be optimal.
\end{proof}

\begin{definition}[Circuit supports as a set family]
  \label{def:lowerbounds-circuit-supports}
  \lean{Complexity.CircuitSparseSynthesis.circuitSupports}
  \leanok
  \uses{def:lowerbounds-correction}
  Let $\mathcal C_{n,s}$ be the finite family of sets
  $T\subseteq\{0,1\}^n$ with $C(1_T)\le s$.
  A family shatters $S$ if every $A\subseteq S$ is $S\cap T$ for some
  member $T$ of the family. Its VC dimension is the largest cardinality
  of a shattered set, using Mathlib's finite set-family definitions.
\end{definition}

\begin{theorem}[Uniform sample interpolation and shattering]
  \label{thm:lowerbounds-circuit-shattering}
  \lean{Complexity.CircuitSparseSynthesis.exists_interpolating_circuit,
    Complexity.CircuitSparseSynthesis.circuitSupports_shatters,
    Complexity.CircuitSparseSynthesis.le_vcDim_circuitSupports}
  \leanok
  \uses{thm:lowerbounds-square-root-synthesis,def:lowerbounds-circuit-supports}
  For every $\varepsilon>0$ and all sufficiently large $p$, every assignment
  of Boolean labels to any sample $S\subseteq\{0,1\}^{2p}$ of size at most
  $2^p$ is computed on $S$ by a circuit of size at most
  \[
    s_p=\left\lfloor(1+\varepsilon)\frac{2^p}{p}\right\rfloor.
  \]
  Thus $\mathcal C_{2p,s_p}$ shatters every such $S$, and its VC dimension
  is at least $2^p$. With $D=2^p$, this budget is
  $(1+o(1))D/\log_2 D$.
\end{theorem}
\begin{proof}
  \leanok
  Extend the sample labels arbitrarily to the full cube and apply partial
  synthesis. Its minimum is attained by a concrete circuit, and its integer
  size is at most the displayed floor. For a subset $A\subseteq S$, use
  its indicator as the labels and take the support of the resulting circuit;
  its intersection with $S$ is exactly $A$. A sample of cardinality $2^p$
  gives the VC-dimension bound. Circuit outputs outside the sample are
  unrestricted; this is an interpolation result, not a learning algorithm.
\end{proof}

\begin{definition}[Conditional circuit complexity]
  \label{def:lowerbounds-algebraic-conditional}
  \lean{Cslib.Circuits.Circuit.conditionalGateComplexity}
  \leanok
  Fix a signature and an interpretation of it. The conditional complexity of a
  target $f$, possibly with several outputs, given supplied functions
  $g_1, \ldots, g_k$ of the same input is the minimum number of gates, every
  gate counted, of a circuit $h$ with $h(x, g(x)) = f(x)$ for every $x$: the
  original inputs and the supplied values are free extra inputs, and nothing is
  required of $h(x, y)$ when $y \ne g(x)$. It is $\infty$ when no circuit
  computes $f$ this way.
\end{definition}

\begin{theorem}[Laws of conditional complexity]
  \label{thm:lowerbounds-algebraic-conditional}
  \lean{Cslib.Circuits.Circuit.conditionalGateComplexity_triangle,
    Algebraic.ConditionalComplexity.Linear.conditionalGateComplexity_eq_min_weight}
  \leanok
  \uses{def:lowerbounds-algebraic-conditional}
  Over any signature and interpretation, conditional complexity obeys the
  triangle inequality: computing $h$ given $f$ costs at most computing $h$
  given $g$ plus computing $g$ given $f$.
  Over $\mathbb{F}_2$, take any signature whose operations have arity at most
  two, interpreted over $\mathbb{F}_2$, in which some one-gate circuit computes
  $x_0 + x_1$. Then the conditional
  complexity of a nonzero linear form $t \cdot x$ given linear helpers
  $s_1 \cdot x, \ldots, s_k \cdot x$ is exactly
  \[
    \min_{c \in \mathbb{F}_2^k}
      \Bigl(\lvert c \rvert + \bigl\lvert t + \textstyle\sum_j c_j s_j \bigr\rvert
        - 1\Bigr),
  \]
  where $\lvert\cdot\rvert$ is Hamming weight: the fewest helpers and inputs
  whose sum is the target, minus one. Every gate is counted, constant gates
  included. The signature may contain other operations as well, nonlinear ones
  included; the single-gate addition is a hypothesis of the theorem.
\end{theorem}
\begin{proof}
  \leanok
  Compose the two circuits for the triangle inequality. For linear forms, a
  circuit computing the form must combine at least the minimum number of
  input and helper terms, and such a combination is realized directly.
\end{proof}

\begin{theorem}[Hessian rank bounds multiplications]
  \label{thm:lowerbounds-algebraic-hessian}
  \lean{Algebraic.Applications.hessianRank_lowerBound}
  \leanok
  Every arithmetic circuit over a field, with addition, multiplication, and
  constant gates, whose output equals $p$ as a formal polynomial has at least
  $\lceil \operatorname{rank} H_p(a) / 2 \rceil$ multiplication gates, where
  $H_p(a)$ is the Hessian of $p$ at any point $a$. Additions and constants are
  free, but every multiplication gate is counted, including multiplication by
  a constant. This is weaker than the classical form of the bound, which
  counts only nonscalar multiplications.
\end{theorem}
\begin{proof}
  \leanok
  Each multiplication gate raises the rank of the second-order part of the
  computed polynomials by at most two, while additions and constants do not.
\end{proof}

\begin{theorem}[Waring bound for the squarefree monomial]
  \label{thm:lowerbounds-algebraic-waring}
  \lean{Algebraic.Applications.waringSum_lowerBound}
  \leanok
  Over a field of characteristic zero, any expression of $x_1 x_2 \cdots x_{2n}$
  as a polynomial identity $\sum_{i \in I} a_i \ell_i^{2n}$, with scalars $a_i$
  and linear forms $\ell_i$ without constant term, has
  $\lvert I \rvert \ge \binom{2n}{n}$. This bounds the number of terms of this
  restricted representation, not arithmetic circuit size.
\end{theorem}
\begin{proof}
  \leanok
  Compare the dimension of the space of partial derivatives of order $n$ of
  each side.
\end{proof}

\begin{theorem}[Sharp mass production]
  \label{thm:lowerbounds-algebraic-mass-production}
  \lean{Algebraic.MassProduction.BlockInduction.exponentialMassProduction,
    Algebraic.MassProduction.Nonuniform.realSharpMassProduction}
  \leanok
  Circuits are over the imported library's De Morgan signature with its
  standard cost: negation, AND, and OR gates cost one, constants and identities
  are free. For $f$ on $n$ inputs and $t \ge 1$, the mass complexity of $f$ at
  $t$ copies is the minimum cost of one circuit that evaluates $f$
  independently on $t$ disjoint blocks of $n$ inputs.

  For every real $0 \le \gamma < 1$ and $\varepsilon > 0$ there is a threshold
  such that for every $n$ above it, every $f$ on $n$ inputs, and every
  $1 \le t \le 2^{\gamma n}$, the mass complexity of $f$ at $t$ copies is at
  most $(1/(1 - \gamma) + \varepsilon)\, 2^n / n$. In a coarser discrete form,
  for all natural numbers $a < b$ there is a constant $C$ such that at every
  length $n \ge 1$, every $f$ on $n$ inputs, and every
  $1 \le t \le 2^{\lfloor a n / b \rfloor}$, the mass complexity is at most
  $C \lfloor 2^n / n \rfloor$.
\end{theorem}
\begin{proof}
  \leanok
  A nonuniform construction that routes all copies through shared table
  lookups: sorting and routing networks gather the requests, each table entry
  is computed once and broadcast to every copy that needs it, and finite-field
  local-recovery codes with disjoint-line scheduling serve the requests. The
  discrete form comes from an induction over equal input blocks, which reaches
  every rational rate below one.
\end{proof}

\begin{theorem}[Uhlig's mass production]
  \label{thm:lowerbounds-algebraic-uhlig}
  \lean{Algebraic.MassProduction.LupanovSynthesis.uhlig_massProduction}
  \leanok
  \uses{thm:lowerbounds-algebraic-mass-production}
  With mass complexity as in
  Theorem~\ref{thm:lowerbounds-algebraic-mass-production}, let
  $D : \mathbb{N} \to \mathbb{N}$ satisfy $D(n) = o(n / \log n)$ in the
  discrete form: for every natural number $M \ge 1$, eventually
  $M \cdot D(n) \cdot \lfloor \log_2 n \rfloor \le n$. Then for every natural
  number $P \ge 1$, for all large $n$, every $f$ on $n$ inputs, and every
  $1 \le t \le 2^{D(n)}$, the mass complexity of $f$ at $t$ copies is at most
  $(1 + 1/P)\, 2^n / n$, stated without division as
  $P\,n \cdot \mathrm{mass}(f, t) \le (P + 1)\, 2^n$.
\end{theorem}
\begin{proof}
  \leanok
  Combine a coefficient-one Lupanov circuit for a single copy with Uhlig's
  recursive two-copy construction, in which XORs of subfunctions at adjacent
  prefix values serve as shared resources from which each request is recovered
  through its own disjoint set of resources.
\end{proof}

\begin{definition}[Prescribed endpoint bags]
  \label{def:lowerbounds-algebraic-path-endpoint}
  \lean{Algebraic.Cutwidth.PathDecomposition.EndsAt,
    Algebraic.Cutwidth.PathDecomposition.StartsAt,
    Algebraic.Cutwidth.PathDecomposition.reverse,
    Algebraic.Cutwidth.PathDecomposition.trivial_endsAt}
  \leanok
  A path decomposition ends at $X$ if it has a last bag and that bag is
  exactly $X$; it starts at $X$ if its first bag is exactly $X$.
  Reversal exchanges the endpoint roles. The one-bag decomposition ends
  at the full vertex set.
\end{definition}

\begin{theorem}[Restoring a boundary vertex]
  \label{thm:lowerbounds-algebraic-endpoint-restore}
  \lean{Algebraic.Cutwidth.PathDecomposition.EndsAt.exists_subset,
    Algebraic.Cutwidth.PathDecomposition.exists_endsAt_of_delete}
  \leanok
  \uses{def:lowerbounds-algebraic-path-endpoint}
  Let $v\in Y$, $X\subseteq Y$, and suppose all neighbors of $v$ lie in
  $Y$. A path decomposition of $H-v$ ending at $Y\setminus\{v\}$ with
  bags of size at most $b$ extends to one of $H$ ending at $X$, provided
  $|Y|\le b$. This supplies the deletion cases in Fomin and H{\o}ie's
  endpoint induction.
\end{theorem}
\begin{proof}
  \leanok
  Append $Y$, then $X$. All incident edges are covered, and every old
  vertex in an appended bag was already present in the previous last bag.
\end{proof}

\begin{theorem}[Structural reduction for a large boundary]
  \label{thm:lowerbounds-algebraic-boundary-reduction}
  \lean{Algebraic.Cutwidth.boundary_reduction,
    Algebraic.Cutwidth.complement_card_edges_lt,
    Algebraic.Cutwidth.exists_tree_component_of_card_edgeFinset_lt}
  \leanok
  In a finite graph of maximum degree three, if $|V|<3|X|$, then some
  vertex of $X$ has at most one neighbor outside $X$, or $H-X$ has a
  connected component that is a tree.
\end{theorem}
\begin{proof}
  \leanok
  Otherwise the cut has at least $2|X|$ edges. Degree counting gives
  $2|E(H-X)|+2|X|\le 3|V\setminus X|$, hence an edge deficit in $H-X$.
  If every component contained a cycle, each would have at least as many
  edges as vertices, contradicting that deficit.
\end{proof}

\begin{theorem}[Small tree components from a linear edge deficit]
  \label{thm:lowerbounds-algebraic-small-tree-component}
  \lean{Algebraic.Cutwidth.exists_small_tree_component_of_edge_deficit}
  \leanok
  In a finite simple graph, if $(M+1)|E|<M|V|$, some connected
  component is a tree with at most $M$ vertices. No degree bound is needed.
\end{theorem}
\begin{proof}
  \leanok
  Suppose every tree component has more than $M$ vertices. For a tree
  component with $v$ vertices and $e=v-1$ edges, $Mv\le(M+1)e$.
  For a component containing a cycle, $v\le e$ gives the same inequality.
  Summing over the components contradicts the assumed deficit.
\end{proof}

\begin{theorem}[Transition between cut boundaries]
  \label{thm:lowerbounds-algebraic-boundary-transition}
  \lean{Algebraic.Cutwidth.PathDecomposition.exists_between_of_crossing}
  \leanok
  Let $H$ be a finite simple graph and $S$ a set of vertices such that every
  vertex has a neighbor across the cut $(S,S^c)$. If the cut contains $b$
  edges, $H$ has a path decomposition with first bag $S$, last bag $S^c$,
  and every bag of size at most $b+1$. This is the boundary transition used
  in Fomin and H{\o}ie's proof of the pathwidth bound for cubic graphs;
  this transition itself does not require a degree bound.
\end{theorem}
\begin{proof}
  \leanok
  Scan the vertices of $S$, introducing their neighbors in $S^c$ before
  forgetting them. Charge each introduced right vertex to an edge from an
  already scanned left vertex, and each remaining left vertex to an edge
  from itself. The charges are disjoint except for the current left vertex,
  which accounts for the extra one. Edges within each side are covered by
  its endpoint bag.
\end{proof}

\begin{theorem}[Tree centroid]
  \label{thm:lowerbounds-algebraic-tree-centroid}
  \lean{Algebraic.Cutwidth.exists_tree_centroid}
  \leanok
  Every finite tree on $n$ vertices has a vertex whose deletion leaves
  connected components of at most $n/2$ vertices.
\end{theorem}
\begin{proof}
  \leanok
  Choose a vertex minimizing the sum of distances to all vertices. Moving
  into a component larger than half the tree decreases the distances to
  its vertices by one and increases every other distance by at most one,
  contradicting minimality. Uniqueness of paths proves the distance change.
\end{proof}

\begin{theorem}[Logarithmic tree and forest decompositions]
  \label{thm:lowerbounds-algebraic-tree-decomposition}
  \lean{Algebraic.Cutwidth.PathDecomposition.exists_tree_of_card_le_pow,
    Algebraic.Cutwidth.PathDecomposition.exists_tree,
    Algebraic.Cutwidth.PathDecomposition.exists_forest,
    Algebraic.Cutwidth.PathDecomposition.exists_of_components,
    Algebraic.Cutwidth.PathDecomposition.exists_addVertex}
  \leanok
  \uses{thm:lowerbounds-algebraic-tree-centroid}
  A tree with at most $2^k$ vertices has a path decomposition with bags of
  size at most $k+1$. In particular, every tree or forest on $n$ vertices
  has bags of size at most $\lceil\log_2 n\rceil+1$, with the logarithmic
  term defined to be zero when $n=0$. This elementary bound suffices for
  the tree component in Fomin and H{\o}ie's endpoint induction.
\end{theorem}
\begin{proof}
  \leanok
  Recurse on the components left by a centroid, concatenate their
  decompositions, and insert the centroid into every bag. Append its
  singleton to cover also the one-vertex case. Each recursion halves the
  component size and adds at most one vertex per bag. For forests,
  concatenate the tree decompositions without increasing their bag sizes.
\end{proof}

\begin{theorem}[Concatenation along a boundary]
  \label{thm:lowerbounds-algebraic-endpoint-glue}
  \lean{Algebraic.Cutwidth.PathDecomposition.exists_glue,
    Algebraic.Cutwidth.PathDecomposition.exists_glue_embeddings,
    Algebraic.Cutwidth.PathDecomposition.exists_pad,
    Algebraic.Cutwidth.PathDecomposition.exists_attach,
    Algebraic.Cutwidth.PathDecomposition.EndsAt.reverse,
    Algebraic.Cutwidth.PathDecomposition.StartsAt.reverse}
  \leanok
  \uses{def:lowerbounds-algebraic-path-endpoint}
  Suppose two induced graphs together cover every vertex and edge, and
  their adjoining endpoint bags contain every shared vertex. Their
  decompositions concatenate without enlarging any bag. In particular,
  an induced subgraph with bag bound $c$ can be attached along a boundary
  $X$ to a decomposition of the rest ending at $X$: the new decomposition
  ends at $X$ and has bag bound $\max\{b,|X|+c\}$, where $b$ was the bound
  for the rest. Every edge leaving the attached subgraph must meet $X$.
\end{theorem}
\begin{proof}
  \leanok
  Shared vertices persist through the join because they lie in both
  adjoining bags. For attachment, add $X$ to each subgraph bag and put
  $X$ at both ends before concatenating.
\end{proof}

\begin{theorem}[Prescribed endpoints in subcubic graphs]
  \label{thm:lowerbounds-algebraic-subcubic-endpoint}
  \lean{Algebraic.Cutwidth.PathDecomposition.exists_subcubic_endsAt}
  \leanok
  \uses{thm:lowerbounds-algebraic-endpoint-restore,
    thm:lowerbounds-algebraic-boundary-reduction,
    thm:lowerbounds-algebraic-tree-decomposition,
    thm:lowerbounds-algebraic-endpoint-glue}
  Every graph of maximum degree three on $n$ vertices has a path
  decomposition ending at any prescribed set $X$, with bags of size at most
  \[
    \max\{|X|,\lfloor n/3\rfloor+1\}+\lceil\log_2 n\rceil+1.
  \]
  The logarithmic term is zero for $n=0$. This proves the endpoint lemma
  in Fomin and H{\o}ie's argument with an elementary logarithmic remainder.
  It does not require the bisection theorem.
\end{theorem}
\begin{proof}
  \leanok
  For a nonempty graph, enlarge $X$ to a boundary $Y$ of size
  $\max\{|X|,\lfloor n/3\rfloor+1\}$. If a boundary vertex has at most
  one outside neighbor, delete it, recurse with that neighbor added to
  the boundary, and restore the vertex. Otherwise the complement of $Y$
  has a tree component. Delete the component, recurse, and attach its
  logarithmic-width decomposition after adding $Y$ to its bags. Both
  recursions reduce the number of vertices, and concatenation preserves
  the bound on the remaining graph. Finally append $X$.
\end{proof}

\begin{definition}[Cubic bisection hypothesis]
  \label{def:lowerbounds-algebraic-bisection}
  \lean{Algebraic.Cutwidth.BisectionBound}
  \leanok
  The predicate $\mathrm{BisectionBound}(\xi,N_0)$ says that every simple
  $3$-regular graph on $n>N_0$ vertices has a cut $(S,S^c)$ with
  $||S|-|S^c||\le 1$ and at most $(1/6+\xi)n$ crossing edges. The
  Monien--Preis theorem asserting this for every $\xi>0$ and some $N_0$
  remains unformalized.
\end{definition}

\begin{definition}[Helpfulness of a vertex move]
  \label{def:lowerbounds-algebraic-helpfulness}
  \lean{Algebraic.Cutwidth.helpfulness}
  \leanok
  Write $\partial S$ for the edges crossing the vertex cut $(S,S^c)$.
  The helpfulness of flipping a vertex set $X$ is the signed integer
  \[
    h_S(X)=|\partial X\cap\partial S|-|\partial X\setminus\partial S|.
  \]
  For $X\subseteq S$, this is the helpfulness in Monien and Preis's
  bisection argument: crossing edges removed minus internal edges made
  crossing by moving $X$ to the other side.
\end{definition}

\begin{theorem}[Exact accounting and minimum bisections]
  \label{thm:lowerbounds-algebraic-cut-improvement}
  \lean{Algebraic.Cutwidth.cutFinset_symmDiff,
    Algebraic.Cutwidth.helpfulness_eq_sub,
    Algebraic.Cutwidth.helpfulness_add,
    Algebraic.Cutwidth.Bisection.exists_min_cut_of_card,
    Algebraic.Cutwidth.Bisection.exists_min_bisection,
    Algebraic.Cutwidth.Bisection.two_moves_le_zero,
    Algebraic.Cutwidth.abs_helpfulness_le_mul_card}
  \leanok
  \uses{def:lowerbounds-algebraic-helpfulness}
  The helpfulness is exactly
  $h_S(X)=|\partial S|-|\partial(S\mathbin{\triangle}X)|$, and
  successive moves have additive helpfulness. A minimum cut exists at
  each prescribed side cardinality, and a minimum balanced cut exists
  also for graphs of odd order. At a minimum cut $S$, if $X\subseteq S$,
  $Y\subseteq(S\setminus X)^c$, and $|X|=|Y|$, then
  \[
    h_S(X)+h_{(S\setminus X)^c}(Y)\le 0.
  \]
  If all vertices of $X$ have degree at most $d$, then
  $|h_S(X)|\le d|X|$.
\end{theorem}
\begin{proof}
  \leanok
  Flipping $X$ toggles exactly the edges of $\partial X$.
  Subtracting the cardinalities gives the signed identity, which
  telescopes for successive moves. Minimize over the finite family of
  cuts with the chosen cardinality; complementing a cut handles either
  orientation of an odd balanced partition. The equally sized second
  move restores the cardinality, so positive total helpfulness would
  contradict minimality. At most the edges incident to $X$ can change.
\end{proof}

\begin{theorem}[Degree accounting and local helpful sets]
  \label{thm:lowerbounds-algebraic-local-helpful}
  \lean{Algebraic.Cutwidth.degree_sum_cut,
    Algebraic.Cutwidth.helpfulness_eq_degree_sum,
    Algebraic.Cutwidth.one_le_helpfulness_singleton,
    Algebraic.Cutwidth.card_sub_two_le_helpfulness,
    Algebraic.Cutwidth.one_le_helpfulness_of_connected_boundary}
  \leanok
  \uses{def:lowerbounds-algebraic-helpfulness}
  For $X\subseteq S$, let $d_{S^c}(v)$ count the neighbors of $v$ outside
  $S$. Then
  \[
    h_S(X)=2\sum_{v\in X}d_{S^c}(v)+2|E(H[X])|-\sum_{v\in X}\deg(v).
  \]
  In a graph of maximum degree three, a vertex with two crossing edges
  is at least $1$-helpful. A connected set $X$ of boundary vertices has
  $h_S(X)\ge |X|-2$; thus three or more connected boundary vertices
  form a helpful set. These include the first two local cases of
  Monien--Preis's bounded helpful-set lemma.
\end{theorem}
\begin{proof}
  \leanok
  The degree sum on $X$ counts its internal edges twice and its crossing
  edges once. The helpfulness formula follows by separating edges going
  outside $S$. For connected boundary vertices, each summand
  $d_{S^c}(v)$ is at least one, there are at least $|X|-1$ internal
  edges, and the degree sum is at most $3|X|$.
\end{proof}

\begin{definition}[Boundary lift and internal red edges]
  \label{def:lowerbounds-algebraic-boundary-lift}
  \lean{Algebraic.Cutwidth.Bisection.boundaryLift,
    Algebraic.Cutwidth.Bisection.sharedBoundary}
  \leanok
  Let $C$ be the vertex boundary of a side $S$ and let $X\subseteq S\setminus C$.
  The lift of $X$ is $X$ together with all vertices of $C$ adjacent to $X$.
  The shared boundary consists of those vertices of $C$ with at least two
  neighbors in $X$. When $C$ is independent and each of its vertices has
  one outside neighbor in a cubic graph, each vertex of $C$ represents a
  red edge between its two interior neighbors. The shared boundary counts
  internal red edges, preserving parallel edges as distinct boundary vertices.
\end{definition}

\begin{theorem}[Lifting a positive red/black witness]
  \label{thm:lowerbounds-algebraic-boundary-lift}
  \lean{Algebraic.Cutwidth.Bisection.boundaryLift_subset,
    Algebraic.Cutwidth.Bisection.card_boundaryLift_le,
    Algebraic.Cutwidth.Bisection.helpfulness_boundaryLift,
    Algebraic.Cutwidth.Bisection.exists_helpful_set_of_red_surplus}
  \leanok
  \uses{def:lowerbounds-algebraic-boundary-lift,
    thm:lowerbounds-algebraic-local-helpful}
  Under the normalized boundary hypotheses above, the lift $Y$ of $X$
  lies in $S$, has $|Y|\le4|X|$, and satisfies
  \[
    h_S(Y)=|\mathrm{sharedBoundary}(X)|-
      |\partial X\setminus\partial C|.
  \]
  Thus more internal red edges than external black edges give a
  $1$-helpful set. This proves the lifting step of Monien--Preis's
  bounded helpful-set argument; it does not assume or prove that such
  a positive witness exists.
\end{theorem}
\begin{proof}
  \leanok
  Every selected interior vertex contributes minus its remaining black
  neighbors. An attached boundary vertex contributes zero if it has one
  selected neighbor and one if it has two. Sum these contributions.
  Each selected vertex has at most three adjacent boundary vertices,
  giving the size bound.
\end{proof}

\begin{theorem}[Red density and a small black tree component]
  \label{thm:lowerbounds-algebraic-red-density}
  \lean{Algebraic.Cutwidth.Bisection.card_cutBoundary_eq_cut,
    Algebraic.Cutwidth.Bisection.card_boundary_interior_neighbors,
    Algebraic.Cutwidth.Bisection.card_interior_neighbors,
    Algebraic.Cutwidth.Bisection.boundary_red_density,
    Algebraic.Cutwidth.Bisection.card_interior_edges,
    Algebraic.Cutwidth.Bisection.exists_small_interior_tree_component}
  \leanok
  \uses{def:lowerbounds-algebraic-boundary-lift,
    thm:lowerbounds-algebraic-small-tree-component}
  In a normalized cubic side, put $D=S\setminus C$ and let $B$ be
  the edges induced by $D$. Then $|C|=|\partial S|$ and
  $2|B|+2|C|=3|D|$. If $|\partial S|>(1/3+\xi)|S|$ with $\xi>0$,
  then $|C|>(1/2+3\xi/2)|D|$. Consequently the black graph has a
  tree component of at most $M$ vertices whenever $(M+1)3\xi/2\ge1$.
  This component need not itself be a positive red/black witness.
\end{theorem}
\begin{proof}
  \leanok
  Every boundary vertex has two interior neighbors, and every interior
  vertex has total red-plus-black degree three. Double counting gives
  the identities. The density inequality uses $|S|=|C|+|D|$.
  Hence $|B|<(1-3\xi/2)|D|$, which gives
  $(M+1)|B|<M|D|$ under the stated budget.
\end{proof}

\begin{theorem}[Growing helpful sets through boundary neighbors]
  \label{thm:lowerbounds-algebraic-helpful-growth}
  \lean{Algebraic.Cutwidth.helpfulness_insert,
    Algebraic.Cutwidth.Bisection.helpfulness_union_boundary,
    Algebraic.Cutwidth.Bisection.helpfulness_union_boundaryLift_of_closed,
    Algebraic.Cutwidth.Bisection.exists_helpful_extension_of_neighbor_types}
  \leanok
  \uses{thm:lowerbounds-algebraic-local-helpful,
    def:lowerbounds-algebraic-boundary-lift}
  In a subcubic graph, adjoining boundary vertices adjacent to a moved
  set increases its helpfulness by at least one per new vertex.
  More generally, let $X\subseteq S$ and $Z\subseteq S\setminus C$.
  Suppose every neighbor of $Z$ lies in $X\cup Z\cup C$, and an edge
  joins $X$ to $Z\setminus X$. Then adjoining the boundary lift of $Z$
  increases helpfulness by at least one.

  Consequently, suppose $h_S(X)\ge0$ and an interior vertex $v\notin X$
  is adjacent to $X$. If every neighbor of $v$ outside $X$ either belongs
  to $C$ or is an interior vertex with at least two boundary neighbors,
  there is a helpful set of at most $|X|+7$ vertices.
\end{theorem}
\begin{proof}
  \leanok
  Adding a vertex contributes its singleton helpfulness plus twice its
  number of neighbors already selected. For the closure statement, write
  helpfulness as a sum over selected vertices of twice their number of
  neighbors outside $S$ minus their number of neighbors outside the
  selected set. New interior vertices have no remaining external neighbors;
  new boundary vertices contribute nonnegatively. Old contributions cannot
  decrease, and the attaching edge increases one of them strictly.
  For the last claim, include $v$, its interior neighbors outside $X$,
  and their boundary neighbors. At most two neighbors of $v$ lie outside
  $X$, and each adds at most three vertices including itself. Together
  with $v$, at most seven vertices are added.
\end{proof}

\begin{theorem}[The remaining small helpful configurations]
  \label{thm:lowerbounds-algebraic-normalization-configurations}
  \lean{Algebraic.Cutwidth.Bisection.nonneg_helpfulness_boundary_pair,
    Algebraic.Cutwidth.Bisection.nonneg_helpfulness_boundary_star,
    Algebraic.Cutwidth.Bisection.exists_helpful_of_boundary_pair_three_neighbors,
    Algebraic.Cutwidth.Bisection.exists_helpful_of_shared_boundary_neighbor,
    Algebraic.Cutwidth.Bisection.exists_helpful_of_boundary_neighbor_configuration}
  \leanok
  \uses{thm:lowerbounds-algebraic-helpful-growth}
  Write $D_3$ for the interior vertices with three boundary neighbors.
  The remaining local configurations in Monien--Preis's normalization
  argument give the following helpful sets in a subcubic graph:
  \begin{itemize}
    \item A boundary edge with one endpoint adjacent to $D_3$ gives
      a set of at most five vertices.
    \item Two distinct $D_3$ vertices sharing a boundary neighbor give
      a set of at most seven vertices.
    \item Let an interior vertex $v$ be adjacent to $c\in C$, where
      $c$ also has a boundary neighbor or a $D_3$ neighbor distinct from $v$.
      If every other neighbor of $v$ belongs to $C$ or is an interior
      vertex with at least two boundary neighbors, a helpful set of
      at most eleven vertices exists.
  \end{itemize}
  Boundary neighborhoods may overlap.
\end{theorem}
\begin{proof}
  \leanok
  A boundary edge is nonnegative in helpfulness, as is a $D_3$ vertex
  together with its three neighbors. Adding a $D_3$ vertex and its
  neighbors to a set containing one of those neighbors adds at most
  three vertices and improves helpfulness by at least one. This proves
  the five- and seven-vertex cases. In the last case, start from the
  boundary edge or the four-vertex star and apply the seven-vertex
  extension bound. Repeated neighbors reduce the size and improve the
  helpfulness accounting.
\end{proof}

\begin{theorem}[The neighbor needed for a normalization switch]
  \label{thm:lowerbounds-algebraic-switch-neighbor}
  \lean{Algebraic.Cutwidth.Bisection.exists_switch_neighbor,
    Algebraic.Cutwidth.Bisection.outside_eq_one_of_no_small_helpful,
    Algebraic.Cutwidth.Bisection.boundary_degree_le_one_of_no_small_helpful}
  \leanok
  \uses{thm:lowerbounds-algebraic-normalization-configurations,
    thm:lowerbounds-algebraic-local-helpful}
  If no helpful subset of $S$ has at most eleven vertices, every boundary
  vertex has exactly one neighbor outside $S$ and at most one neighbor
  within $C$. Suppose an interior vertex $v$ is adjacent to $c\in C$,
  and $c$ has either a boundary neighbor or a $D_3$ neighbor distinct
  from $v$. Then $v$ has another neighbor $p\in S\setminus C$ with at
  most one boundary neighbor.
  These are the local prerequisites for Monien--Preis's edge switches.
  Both phases and their combined reverse bounds are formalized below.
\end{theorem}
\begin{proof}
  \leanok
  Two outside neighbors would give a helpful singleton. Two boundary
  neighbors would give three connected boundary vertices. If every other
  neighbor of $v$ had one of the types in the eleven-vertex configuration,
  that configuration would supply a forbidden helpful set. An interior
  vertex has no neighbors outside $S$, so the remaining neighbor has
  the required type.
\end{proof}

\begin{definition}[A valid edge switch]
  \label{def:lowerbounds-algebraic-edge-switch}
  \lean{Algebraic.Cutwidth.Bisection.Switchable,
    Algebraic.Cutwidth.Bisection.switchEdges}
  \leanok
  Four distinct vertices $a,b,c,d$ form a valid switch when $ab$ and
  $cd$ are edges and $ac$ and $bd$ are absent. The switched graph
  deletes $ab,cd$ and adds $ac,bd$.
\end{definition}

\begin{theorem}[Invariants and helpfulness of one switch]
  \label{thm:lowerbounds-algebraic-edge-switch-invariants}
  \lean{Algebraic.Cutwidth.Bisection.switchEdges_degree,
    Algebraic.Cutwidth.Bisection.switchEdges_regular,
    Algebraic.Cutwidth.Bisection.switchEdges_cut,
    Algebraic.Cutwidth.Bisection.switchEdges_neighbors_outside,
    Algebraic.Cutwidth.Bisection.switchEdges_boundary,
    Algebraic.Cutwidth.Bisection.helpfulness_switchEdges}
  \leanok
  \uses{def:lowerbounds-algebraic-edge-switch,
    thm:lowerbounds-algebraic-helpful-growth}
  A valid switch preserves every degree. If all endpoints lie in $S$,
  it preserves the cut of $S$, every vertex's neighbors outside $S$,
  and the boundary vertex set. For $X\subseteq S$, let $\chi$ be its
  membership indicator. The helpfulness in the switched graph satisfies
  \[
    h'_S(X)=h_S(X)+2(\chi(a)-\chi(d))(\chi(c)-\chi(b)).
  \]
\end{theorem}
\begin{proof}
  \leanok
  Each endpoint exchanges one old neighbor for one new neighbor, and
  all changed edges lie within $S$. In the neighbor-sum expression for
  helpfulness, only the four endpoints change. Sum their contributions
  to obtain the indicator formula.
\end{proof}

\begin{theorem}[Existence of the two normalization switches]
  \label{thm:lowerbounds-algebraic-edge-switch-existence}
  \lean{Algebraic.Cutwidth.Bisection.exists_boundary_switch,
    Algebraic.Cutwidth.Bisection.exists_three_neighbor_switch}
  \leanok
  \uses{def:lowerbounds-algebraic-edge-switch,
    thm:lowerbounds-algebraic-switch-neighbor,
    thm:lowerbounds-algebraic-normalization-configurations}
  In a cubic graph with no helpful subset of $S$ of size at most eleven,
  a boundary edge $ab$ admits a valid switch $a,b,c,d$. So does an edge
  $ab$ with $a$ an interior vertex having three boundary neighbors and
  $b\in C$. In both cases $bc$ is an edge, $c,d\in S\setminus C$,
  $c$ has at most two boundary neighbors, and $d$ has at most one.
\end{theorem}
\begin{proof}
  \leanok
  Vertex $b$ has one outside neighbor and two inside neighbors. Let $c$
  be its inside neighbor other than $a$. The five- and seven-vertex
  configurations rule out the forbidden types of $c$. The switching-neighbor
  lemma supplies $d$. Degree counting excludes $bd$. In the first case,
  an existing $ac$ would make the boundary edge together with $c$ helpful;
  in the second case, all neighbors of $a$ are boundary vertices, so $ac$
  is absent. The vertex types and adjacencies establish distinctness.
\end{proof}

\begin{theorem}[Local reverse transfer through a switch]
  \label{thm:lowerbounds-algebraic-edge-switch-restoration}
  \lean{Algebraic.Cutwidth.Bisection.helpfulness_reverse_switch_le,
    Algebraic.Cutwidth.Bisection.exists_restore_boundary_switch,
    Algebraic.Cutwidth.Bisection.exists_restore_three_neighbor_switch}
  \leanok
  \uses{thm:lowerbounds-algebraic-edge-switch-invariants,
    thm:lowerbounds-algebraic-helpful-growth}
  Reversing a valid normalization switch transfers any $X\subseteq S$
  to a superset $Y\subseteq S$ with $h_S(Y)\ge h'_S(X)$. At most two
  vertices are added for a boundary-pair switch and at most four for a
  three-boundary-neighbor switch. If $Y\ne X$, both $a,b$ belong to $Y$
  and exactly one belonged to $X$.
  This is a bound for one switch. Both phases receive combined bounds below.
\end{theorem}
\begin{proof}
  \leanok
  The indicator formula loses two units on reversal only when $X$
  selects exactly $a,c$ or exactly $b,d$ among the endpoints. In the first
  case add $b$: its outside edge and two selected neighbors gain at
  least three. In the second case add $a,c$ for a boundary-pair switch.
  For the other switch also add the boundary neighbors of $a$; $b$ is
  already selected, so at most four new vertices suffice. Each step gains
  at least the two units lost. In all other cases reversal does not
  decrease helpfulness; the exact formula also allows an increase.
\end{proof}

\begin{theorem}[Boundary-edge normalization with a uniform reverse bound]
  \label{thm:lowerbounds-algebraic-boundary-normalization}
  \lean{Algebraic.Cutwidth.Bisection.exists_independent_boundary_or_small_helpful}
  \leanok
  \uses{thm:lowerbounds-algebraic-edge-switch-existence,
    thm:lowerbounds-algebraic-edge-switch-invariants,
    thm:lowerbounds-algebraic-edge-switch-restoration,
    thm:lowerbounds-algebraic-switch-neighbor}
  For a side $S$ of any cubic graph $H$, either a helpful subset of
  $S$ has at most 33 vertices, or there is a cubic graph $G$ on the same
  vertices with the same cut and boundary $C$. In $G$, the boundary is
  independent and every boundary vertex has one outside neighbor.
  Moreover, every $X\subseteq S$ admits $X\subseteq Y\subseteq S$ with
  $|Y|\le3|X|$ and $h^H_S(Y)\ge h^G_S(X)$.
  This completes the first normalization phase. The next phase eliminates
  interior vertices with three boundary neighbors.
\end{theorem}
\begin{proof}
  \leanok
  If a helpful set of size at most eleven already exists, stop.
  Otherwise the original boundary graph is a matching. Repeatedly
  apply the first switch; each removes exactly one boundary edge, so
  induction on their number terminates. Stop early if a helpful set
  of at most eleven vertices appears.

  For the reverse transfer, complete pairs in the original matching.
  Fix the closure $U$ of the initially selected boundary vertices under
  their original partners. This set has size at most $2|X\cap C|$.
  The local restoration adds only the two endpoints of the restored
  boundary edge and possibly one interior vertex, so selected boundary
  vertices remain in $U$. Every nontrivial extension adds at least one
  boundary vertex and at most two vertices in total. Thus
  \[
    |Y|+2|X\cap C|\le |X|+2|Y\cap C|
  \]
  telescopes over the entire reverse sequence. Since
  $|Y\cap C|\le |U|\le2|X\cap C|$, this gives $|Y|\le3|X|$.
  An early eleven-vertex witness therefore lifts to at most 33 vertices.
\end{proof}

\begin{theorem}[Eliminating three-boundary-neighbor vertices]
  \label{thm:lowerbounds-algebraic-neighbor-normalization}
  \lean{Algebraic.Cutwidth.Bisection.exists_no_three_neighbors_or_small_helpful}
  \leanok
  \uses{thm:lowerbounds-algebraic-edge-switch-existence,
    thm:lowerbounds-algebraic-edge-switch-invariants,
    thm:lowerbounds-algebraic-edge-switch-restoration,
    thm:lowerbounds-algebraic-normalization-configurations}
  Suppose the boundary $C$ of a side $S$ in a cubic graph $H$ is independent.
  Either a helpful subset of $S$ has at most 55 vertices, or there is a
  cubic graph $G$ on the same vertices with the same cut and boundary,
  independent boundary vertices, one outside neighbor per boundary vertex,
  and at most two boundary neighbors per interior vertex.
  Every $X\subseteq S$ admits $X\subseteq Y\subseteq S$ with
  $|Y|\le5|X|$ and $h^H_S(Y)\ge h^G_S(X)$.
\end{theorem}
\begin{proof}
  \leanok
  Stop if a helpful set of at most eleven vertices already exists.
  Otherwise the original three-neighbor stars are disjoint: a shared
  boundary neighbor would give a helpful set of at most seven vertices.
  Let $A$ be the original centers, and $P\subseteq A$ those already
  processed. Each processed center retains two of its original boundary
  neighbors; each unprocessed center retains three. All other interior
  vertices have at most two. The second switch processes a fresh center
  $a$, and its endpoint $d$, having at most one boundary neighbor, cannot
  be in $A$. The boundary remains independent. Induction on $|A\setminus P|$
  therefore terminates, stopping early if a small helpful set appears.

  Write $B(P)=\bigcup_{a\in P}(N_H(a)\cap C)$ and define
  \[
    \Phi_P(X)=|X|+4|X\cap B(P)|+|X\cap P|\le5|X|.
  \]
  The last inequality uses $B(P)\subseteq C$ and $P\cap C=\varnothing$.
  When a switch is reversed, its extension $Y$ satisfies
  $\Phi_P(Y)\le\Phi_{P\cup\{a\}}(X)$. If $a$ was selected, removing
  its one credit pays for the added boundary vertex $b$. In the other
  exceptional case $b$ was selected, and removing its four credits pays
  for all added vertices. The added interior vertex $c$ cannot be an
  original center: its edge to $b$ would put $b$ in two original stars.
  Thus the extension selects no earlier center or boundary vertex of
  another original star. In all other cases no extension is needed.
  The potential inequalities telescope to $|Y|\le5|X|$, while local
  helpfulness inequalities compose. An early eleven-vertex witness lifts
  to at most 55 vertices.
\end{proof}

\begin{theorem}[Complete normalization with a uniform reverse bound]
  \label{thm:lowerbounds-algebraic-complete-normalization}
  \lean{Algebraic.Cutwidth.Bisection.exists_normalization_or_small_helpful}
  \leanok
  \uses{thm:lowerbounds-algebraic-boundary-normalization,
    thm:lowerbounds-algebraic-neighbor-normalization}
  For any side $S$ of a cubic graph $H$, either a helpful subset of $S$
  has at most 165 vertices, or there is a cubic graph $G$ with the same
  cut and boundary $C$, independent boundary vertices, one outside neighbor
  per boundary vertex, and at most two boundary neighbors per interior vertex.
  Every $X\subseteq S$ admits $X\subseteq Y\subseteq S$ with
  $|Y|\le15|X|$ and $h^H_S(Y)\ge h^G_S(X)$.
\end{theorem}
\begin{proof}
  \leanok
  Apply boundary-edge normalization, then eliminate three-neighbor vertices.
  The reverse size factors multiply to $3\cdot5=15$. The first phase's
  early witness has size at most 33; the second phase's early witness lifts
  to at most $3\cdot55=165$ vertices. Both phases preserve the cut and boundary.
\end{proof}

\begin{theorem}[Constructing and transferring through the suppressed graph]

\begin{proof}

\begin{theorem}[Positive sets from small components]
  \label{thm:lowerbounds-algebraic-red-black-components}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.exists_positive_of_component_edge,
    Algebraic.Cutwidth.Bisection.RedBlack.exists_positive_of_attachments,
    Algebraic.Cutwidth.Bisection.RedBlack.card_cut_le_two_of_connected}
  \leanok
  \uses{def:lowerbounds-algebraic-red-black}
  A red edge joining two black components of at most $M$ vertices gives
  a positive set of at most $2M$ vertices; the components may coincide.
  More generally, attach sets with empty black cuts to a set $P$ along
  distinct red edges. If these red edges outnumber the black cut of $P$,
  their union with $P$ is positive. If each of the $k$ attached sets has
  at most $M$ vertices, the size is at most $|P|+Mk$.
  In particular, three attachments suffice when $P$ is connected and all
  its vertices have black degree at most two.
\end{theorem}
\begin{proof}
  \leanok
  Adding sets with empty black cuts cannot increase the black cut, while
  each distinct attaching edge becomes internal and red. A connected set
  of black degree at most two has at least $|P|-1$ internal black edges;
  degree counting bounds its external black edges by two.
\end{proof}

\begin{theorem}[A bounded positive set from a thin black walk]
  \label{thm:lowerbounds-algebraic-red-black-thin-walk}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.exists_adjacent_pair_le_three_average,
    Algebraic.Cutwidth.Bisection.RedBlack.exists_three_close_positions,
    Algebraic.Cutwidth.Bisection.RedBlack.exists_positive_of_thin_walk}
  \leanok
  \uses{thm:lowerbounds-algebraic-red-black-components}
  Let a black walk have length $\ell$, black degree at most two at all
  its vertices, and $m\ge3$ distinct red attachments indexed in order
  along the walk. Suppose $\ell\le Mm$, and each attachment reaches a
  set with empty black cut and at most $M$ vertices. Then a positive
  set of at most $8M+1$ vertices exists. This formalizes the thin-path
  witness construction in Monien--Preis's core lemma.
\end{theorem}
\begin{proof}
  \leanok
  Among at least two nonnegative integer weights, some adjacent pair
  sums to at most three times their average: otherwise summing all
  adjacent pairs contradicts twice the total weight (the two-weight case
  is immediate). Apply this to the gaps between attachment positions.
  Three consecutive attachments have span at most $3\ell/(m-1)\le5M$.
  Their connecting subwalk has at most $5M+1$ vertices and at most two
  external black edges. Add the three attached sets to obtain the bound
  $8M+1$ and at least three internal red edges.
\end{proof}

\begin{theorem}[A forest weight inequality]
  \label{thm:lowerbounds-algebraic-forest-weight}
  \lean{Algebraic.Cutwidth.Bisection.WeightedTree.sum_nonneg_of_leaf_nonneg,
    Algebraic.Cutwidth.Bisection.WeightedTree.exists_adjacent_pair_lt_of_sum_lt}
  \leanok
  In a finite forest, assign real weights to vertices. If every leaf and
  isolated vertex has nonnegative weight, and the endpoint weights of
  every edge sum to a nonnegative value, then the total weight is
  nonnegative. Consequently, if leaves and isolated vertices weigh at
  least $A$ but the total weight is less than $A$ times the number of
  vertices, some adjacent pair has weight less than $2A$.
\end{theorem}
\begin{proof}
  \leanok
  Choose a vertex $l$ of degree at most one. If it has a neighbor $p$ of
  negative weight, remove both: their combined weight is nonnegative,
  and every newly exposed leaf is adjacent to $p$, hence has nonnegative
  weight. Otherwise remove $l$ alone; any newly exposed leaf was its
  nonnegative neighbor. Induction proves the claim. Subtract $A$ from
  every vertex weight to obtain the shifted version.
\end{proof}

\begin{theorem}[A light adjacent pair in a weighted tree]
  \label{thm:lowerbounds-algebraic-weighted-tree-pair}
  \lean{Algebraic.Cutwidth.Bisection.WeightedTree.exists_adjacent_pair_le}
  \leanok
  \uses{thm:lowerbounds-algebraic-forest-weight}
  Let a tree on $n\ge2$ vertices have nonnegative integer weights of
  total $W$. Suppose each leaf $l$ satisfies $W\le(n-1)w(l)$.
  Then an edge $uv$ satisfies $(n-1)(w(u)+w(v))\le2W$.
  This is Monien--Preis's weighted-tree lemma with denominators cleared.
\end{theorem}
\begin{proof}
  \leanok
  If $W=0$, every weight is zero and any edge works. Otherwise set
  $A=W/(n-1)$. Leaves weigh at least $A$ and $W<An$, so the forest
  inequality gives an adjacent pair of weight less than $2A$.
  Clear the positive denominator and weaken to a non-strict inequality.
\end{proof}

\begin{theorem}[Local restoration of deleted black edges]
  \label{thm:lowerbounds-algebraic-red-black-restoration}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.positive_restore_cut,
    Algebraic.Cutwidth.Bisection.RedBlack.positive_restore_two_boundary}
  \leanok
  \uses{def:lowerbounds-algebraic-red-black}
  Write $R(Y)$ for the red edges internal to $Y$, and $\partial Y$
  for its black cut in $B$. Suppose $X$ is positive after deleting
  $\partial A$ from $B$, the overlap $X\cap A$ has empty black cut,
  and
  \[
    |\partial A\setminus\partial X|\le|R(A)\setminus R(X)|.
  \]
  Then $X\cup A$ is positive in $B$. In particular, it suffices that
  $|\partial A|\le2$, at least one of these boundary edges belongs
  to $\partial X$, and $A$ contains a red edge not already internal
  to $X$. This proves a local compensation step in Monien--Preis's
  argument. The next result bounds simultaneous restoration of a family.
\end{theorem}
\begin{proof}
  \leanok
  The empty-cut overlap implies
  $\partial(X\cup A)=\partial X\mathbin{\triangle}\partial A$.
  Thus the restored cut has the modified cut size of $X$ plus
  $|\partial A\setminus\partial X|$ edges. The union retains all
  internal red edges of $X$ and adds those in $R(A)\setminus R(X)$.
  The budget inequality preserves strict positivity.
\end{proof}

\begin{definition}[A family of restorable regions]
  \label{def:lowerbounds-algebraic-restoration-family}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.deletedEdges,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.deletedGraph,
    Algebraic.Cutwidth.Bisection.RedBlack.boundaryMarks,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.endpointMarks}
  \leanok
  \uses{def:lowerbounds-algebraic-red-black}
  A restoration family consists of pairwise disjoint regions $P_i$ with
  $|\partial_B P_i|\le2$, attached sets $A_i$ with empty black cut,
  and a designated red edge between $P_i$ and $A_i$. Every $P_i$ is
  disjoint from every $A_j$; the attached sets may overlap one another.
  Its deleted graph is $B$ with all cuts $\partial_B P_i$ removed.
  Each boundary edge gives one endpoint mark to its endpoint outside
  $P_i$. The family's endpoint marks sum these contributions, retaining
  multiplicities at a shared outside vertex.
  These are the data used to restore a supplied family of isolated paths.
\end{definition}

\begin{theorem}[Endpoint marks count cut incidences and lost degrees]
  \label{thm:lowerbounds-algebraic-endpoint-marks}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.degree_boundaryMarks,
    Algebraic.Cutwidth.Bisection.RedBlack.boundaryMarks_apply,
    Algebraic.Cutwidth.Bisection.RedBlack.sum_boundaryMarks,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.degree_endpointMarks,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.sum_endpointMarks,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.endpointMarks_apply,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.degree_deletedGraph_add_endpointMarks}
  \leanok
  \uses{def:lowerbounds-algebraic-restoration-family}
  For one region, total endpoint marks equal its cut size. For a restoration
  family, write $m(v)$ for its endpoint marks and $H$ for its deleted graph.
  At every vertex $v$ outside all regions,
  \[
    \deg_H(v)+m(v)=\deg_B(v).
  \]
  If $X$ avoids all regions, then
  \[
    \sum_{v\in X}m(v)=\sum_i|\partial_B P_i\cap\partial_B X|.
  \]
\end{theorem}
\begin{proof}
  \leanok
  Each cut edge has exactly one outside endpoint. A mark lies on $X$
  exactly when that edge also crosses $X$. At a vertex outside all regions,
  distinct regions contribute distinct incident edges, by disjointness.
  Their union is the incident deleted-edge set. Partition the original
  incidence set into its deleted and retained edges to get the degree identity.
\end{proof}

\begin{theorem}[Simultaneous restoration with a uniform size bound]
  \label{thm:lowerbounds-algebraic-family-restoration}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.exists_positive_restore,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.exists_positive_restore_by_marks,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.exists_positive_restore_of_degree}
  \leanok
  \uses{def:lowerbounds-algebraic-restoration-family,
    thm:lowerbounds-algebraic-endpoint-marks}
  Suppose $X$ is positive in the deleted graph of a restoration family,
  avoids every region $P_i$, and $|P_i\cup A_i|\le L$ for every $i$.
  There is a positive set $Y\supseteq X$ in the original graph with
  \[
    |Y|\le |X|+L|\partial_B X|.
  \]
  If every vertex of $X$ has black degree at most $d$, then
  $|Y|\le(1+dL)|X|$. In particular, $d=3$ and $L=4M$ give
  $(1+12M)|X|$, independently of the total number of isolated regions.
  Attached sets may overlap $X$ arbitrarily.
  The mark-based version gives a witness satisfying
  \[
    |Y|\le |X|+L\sum_{v\in X}m(v),
  \]
  charging only endpoint marks on the original witness.
\end{theorem}
\begin{proof}
  \leanok
  Select exactly the indices $i$ for which
  $\partial_B P_i\cap\partial_B X$ is nonempty, and add their regions
  and attachments. Each selected region leaves at most one of its two
  boundary edges outside $X$. Each also supplies a distinct new internal
  red edge: its endpoint in $P_i$ was outside $X$, and disjointness prevents
  two designated red edges from coinciding.

  A black edge crossing the enlarged set either already crossed $X$
  and was not deleted, or belongs to the remaining boundary of a selected
  region. Closed attachments add no crossing edges. Thus at most one
  new black crossing per selected region is paid for by its new red edge,
  preserving positivity. Since $X$ avoids all pairwise disjoint regions,
  the sets $\partial_B P_i\cap\partial_B X$ are pairwise disjoint.
  The number of selected indices is at most $|\partial_B X|$, giving
  the size bound. Summing degrees over $X$ gives the final inequality.
  Alternatively, each selected region contributes at least one cut
  incidence on $X$, hence at least one endpoint mark there. This bounds
  the number of added regions by the mark sum.
\end{proof}

\begin{theorem}[A positive witness from a closed core set]
  \label{thm:lowerbounds-algebraic-closed-core-positive}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.exists_positive_of_closed_core,
    Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.exists_positive_of_closed_core_of_degree}
  \leanok
  \uses{def:lowerbounds-algebraic-restoration-family,
    thm:lowerbounds-algebraic-family-restoration}
  Suppose $X$ avoids all regions of a restoration family, has empty black
  cut after their boundaries are deleted, and $|P_i\cup A_i|\le L$ for
  every $i$. If some region has nonempty boundary entirely contained in
  $\partial_B X$, there is a positive set $Y\supseteq X$ in the original
  graph with
  \[
    |Y|\le |X|+L|\partial_B X|.
  \]
  If the black degree on $X$ is at most $d$, the bound is
  $|Y|\le(1+dL)|X|$. In particular, a core set of size at most $2M$
  with $d=3$ and $L=3M$ gives a witness of size at most $2M(1+9M)$.
\end{theorem}
\begin{proof}
  \leanok
  Add every region whose boundary meets $\partial_B X$, together with
  its chosen closed attachment. Every added region contributes a distinct
  internal red edge and leaves at most one black boundary edge. The
  specified region leaves none, since its whole boundary meets $X$.
  As $X$ has no surviving black cut, the final black cut is strictly
  smaller than the number of added internal red edges. The same counting
  as in simultaneous restoration gives the size bound.
\end{proof}

\begin{theorem}[Deleting designated edges to bridges]
  \label{thm:lowerbounds-algebraic-cycle-edge-selection}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.exists_delete_to_bridges}
  \leanok
  Let $T\subseteq E(B)$ be designated edges of a finite simple graph.
  There is $F\subseteq T$ such that $B-F$ has exactly the same
  reachability relation as $B$, every edge of $T\setminus F$ is a bridge
  in $B-F$, and
  \[
    |F|+|V|\le |E(B)|+c(B),
  \]
  where $c(B)$ is the number of connected components. Thus the number
  deleted is at most the original cycle rank.
\end{theorem}
\begin{proof}
  \leanok
  Choose a largest subset of $T$ whose deletion preserves reachability.
  Any retained designated non-bridge could also be deleted, contradicting
  maximality. Each component of the remaining graph has at least one fewer
  edge than vertices. Sum over components, use the unchanged component
  count, and account for the $|F|$ deleted edges.
\end{proof}

\begin{theorem}[Reserving rank for untouched cyclic components]
  \label{thm:lowerbounds-algebraic-cycle-rank-reserve}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.card_vertices_add_cyclic_le,
    Algebraic.Cutwidth.Bisection.RedBlack.card_removed_add_untouched_cyclic_le}
  \leanok
  Let $F\subseteq E(B)$ have reachability-preserving deletion. Let
  $\mathcal C$ be any collection of original connected components,
  each containing a cycle and having all its internal edges retained.
  Then
  \[
    |F|+|\mathcal C|+|V|\le |E(B)|+c(B).
  \]
  In particular, the deleted edges and the untouched cyclic components
  share one cycle-rank budget.
\end{theorem}
\begin{proof}
  \leanok
  A connected graph has at least one fewer edge than vertices, with
  equality only for a tree. Each cyclic component therefore supplies
  one additional edge. Sum these inequalities over components using
  the degree-sum identity. Reachability-preserving deletion identifies
  the old and new components with equal supports, and untouched internal
  edges preserve each designated cycle. Account for the deleted edges.
\end{proof}

\begin{theorem}[Isolating regions to eliminate cycles through them]
  \label{thm:lowerbounds-algebraic-cycle-region-selection}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.cycle_contains_boundary_of_meets_region,
    Algebraic.Cutwidth.Bisection.RedBlack.exists_isolate_regions_without_cycles}
  \leanok
  \uses{thm:lowerbounds-algebraic-cycle-edge-selection}
  Let $P_i$ be connected regions whose vertices have black degree at most
  two. Choose a boundary edge $e_i\in\partial_B P_i$ for each region,
  with all chosen edges distinct. Some set $J$ of at most the original
  cycle rank many indices satisfies both of the following: deleting only
  $\{e_i:i\in J\}$ preserves reachability, and deleting all cuts
  $\bigcup_{i\in J}\partial_B P_i$ leaves no cycle meeting any $P_i$.
  This selects regions for a supplied thin-path system. The construction
  below instantiates it; the later marked-tree reorganization remains open.
\end{theorem}
\begin{proof}
  \leanok
  At a degree-two vertex on a cycle, both incident edges belong to the
  cycle. Propagate this fact through a connected region: a cycle meeting
  it contains every boundary edge. Apply the preceding theorem to the
  chosen edges. A cycle in the resulting graph can contain none of them:
  each is deleted or is a bridge. It therefore avoids every region.
  Deleting the remaining boundary edges of the selected regions cannot
  create a cycle. Injectivity of the chosen edges identifies the number
  of selected regions with the number of selected edges.
\end{proof}

\begin{theorem}[Connectivity outside the isolated regions]
  \label{thm:lowerbounds-algebraic-isolation-core-connectivity}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.reachable_after_isolating}
  \leanok
  \uses{thm:lowerbounds-algebraic-cycle-region-selection}
  Let $P_i$ have at most two black boundary edges, and choose one boundary
  edge $e_i$ for each selected index $i\in J$. Suppose deleting the chosen
  edges preserves reachability. Then deleting the full cuts of the selected
  regions preserves reachability between any two vertices outside
  $\bigcup_{i\in J}P_i$. Disjointness of the regions is not needed.
\end{theorem}
\begin{proof}
  \leanok
  After deleting the chosen edges, each selected region has at most one
  boundary edge left. A simple path between outside vertices cannot cross
  that boundary: entering and leaving would repeat its sole edge. Choose
  a simple path in the graph after the chosen-edge deletions. It avoids
  every remaining selected boundary and therefore survives full isolation.
\end{proof}

\begin{theorem}[Faithful forest quotients across bridges]
  \label{thm:lowerbounds-algebraic-bridge-quotient}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.BridgeQuotient.reachable_project,

  \leanok
  \uses{def:lowerbounds-algebraic-bridge-quotient}

  representatives. If every designated actual edge is a bridge, the
  quotient is a forest and every component is a tree. The designated
  actual edges map bijectively to quotient edges: none becomes a loop
  and distinct edges do not merge. In particular, the quotient has
  $|F\cap E(B)|$ edges. If every cycle avoids a region family, its
  region-boundary quotient is a forest.
\end{theorem}
\begin{proof}
  \leanok

  quotient walks using paths through each connected contracted piece.
  A quotient path avoiding the image of a designated bridge would lift
  to an original path avoiding that bridge, a contradiction. Thus every
  quotient edge is a bridge. The same lifting argument rules out a
  collapsed bridge or two bridges with the same quotient endpoints.
  Finally, every surviving boundary edge of a region avoided by all
  cycles is a bridge, so the general result applies to region boundaries.
\end{proof}

\begin{theorem}[Region vertices and their quotient degrees]
  \label{thm:lowerbounds-algebraic-quotient-region-vertices}

  preserves the induced graph inside each region. If the regions are
  connected, deleting all their boundaries makes each region exactly
  one connected component, hence one distinct vertex of the quotient.
  Every quotient vertex has degree at most the black cut of its support.
  If the designated actual edges are bridges, this bound is an equality.
\end{theorem}
\begin{proof}
  \leanok

  boundaries are deleted, internal connectivity and the empty cut identify
  the region with one component. Different regions give different components.
  For each neighbor of a quotient vertex, choose an original connecting
  edge. Distinct neighbors give distinct edges in the component's cut.
  Conversely, every boundary edge goes to another quotient vertex. When
  these edges are bridges, injectivity of the edge projection ensures
  that distinct boundary edges give distinct neighbors.
\end{proof}

\begin{theorem}[Suppressing path vertices preserves forests and connectivity]
  \label{thm:lowerbounds-algebraic-path-suppression}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.PathSuppression.reachable_iff,
    Algebraic.Cutwidth.Bisection.RedBlack.PathSuppression.connected,
    Algebraic.Cutwidth.Bisection.RedBlack.PathSuppression.isAcyclic,
    Algebraic.Cutwidth.Bisection.RedBlack.PathSuppression.isTree}
  \leanok
  \uses{def:lowerbounds-algebraic-path-suppression,
    thm:lowerbounds-algebraic-bridge-quotient}
  Suppressing an independent set preserves reachability between surviving
  vertices. Suppressing vertices of degree at most two in a forest leaves
  a forest. Consequently, suppressing an independent set of such vertices
  in a tree leaves a tree whenever at least one vertex survives. Isolated
  vertices and leaves are allowed among the suppressed vertices.
\end{theorem}
\begin{proof}
  \leanok
  Assign each removed vertex with a surviving neighbor to one such
  neighbor, and fix surviving vertices. Lift new edges to one- or two-edge
  original paths and project original walks through this assignment.
  Independence ensures reachability survives projection.
  Contract edges whose assigned endpoints agree. Distinct surviving
  vertices remain in distinct contracted components, and the degree-two
  bound makes each new edge an edge of this bridge quotient. Thus the
  suppressed graph embeds into a forest.
\end{proof}

\begin{theorem}[Counting edges after path suppression]
  \label{thm:lowerbounds-algebraic-path-suppression-count}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.PathSuppression.exists_connection_equiv,

  \leanok
  \uses{def:lowerbounds-algebraic-path-suppression}

  and every vertex of $S$ has degree at most two. There is a bijection
  from the degree-two vertices of $S$ to the edges of the suppressed graph.
  The edge corresponding to a removed vertex joins its two original
  neighbors. Isolated vertices and leaves contribute no edge.
\end{theorem}
\begin{proof}
  \leanok
  A degree-two vertex has two distinct neighbors outside $S$.
  Two removed vertices cannot have the same neighbor pair: that would
  give two different simple paths between the same endpoints in a forest.
  Every surviving edge comes from such a pair, since there are no original
  edges inside the complement of $S$.
\end{proof}

\begin{theorem}[The forest on the core pieces]

\end{theorem}
\begin{proof}

\begin{theorem}[Thin regions without supplied path orderings]

  excluding three attachments bounds $|P|$ by $2M$. A closed thin region

\begin{definition}[Attachment marks and the reserved cycle budget]
  \label{def:lowerbounds-algebraic-thin-path-marks}

\end{definition}

\begin{theorem}[Counting the initial attachment marks]
  \label{thm:lowerbounds-algebraic-thin-path-marks}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.PathSystem.attachmentMarks_apply,

    thm:lowerbounds-algebraic-cycle-rank-reserve}
  Suppose eligible vertices have degree at most two and there is no
  positive set of size at most $8M+1$. Shading a subfamily $J$ assigns
  exactly $2|J|$ attachment marks, all to original components of size
  at most $M$. Any collection $\mathcal D$ of components with at least
  two marks satisfies $|\mathcal D|\le |J|$.

  Suppose also that eligible vertices lie in original components larger
  than $M$, and deleting distinct designated boundary edges of $J$
  preserves reachability. Then $|J|$ satisfies the reserved cycle budget.
  Consequently, for any collection $\mathcal C$ of small cyclic components,
  \[
    |\mathcal D|+|\mathcal C|+|V|\le |E(B)|+c(B).
  \]

  swaps and leaf reorganization is a separate obligation.
\end{theorem}
\begin{proof}
  \leanok

  $\alpha(P)(3-\alpha(P))=2$. Sum the finite mark map; counting
  multiplicities makes the identity hold even when targets coincide.
  Each doubly marked component consumes at least two marks. A boundary
  edge of a thin region touches its large original component, so no such
  edge is internal to a small component. The cycle-rank reserve theorem
  applies to the designated deletions and every small cyclic component.
\end{proof}

\begin{theorem}[Constructing and isolating the thin-path family]

  \uses{def:lowerbounds-algebraic-restoration-family,
    thm:lowerbounds-algebraic-thin-region-witness,
    thm:lowerbounds-algebraic-cycle-region-selection,
    thm:lowerbounds-algebraic-thin-path-marks,
    thm:lowerbounds-algebraic-isolation-core-connectivity,
    thm:lowerbounds-algebraic-bridge-quotient}
  Suppose every eligible vertex has black degree at most two and lies in
  an original black component larger than $M$. If no positive set of at
  most $8M+1$ vertices exists, the canonical thin regions form a restoration
  family with one chosen small attachment each, and
  $|P_i\cup A_i|\le3M$. Choose distinct boundary edges and isolate some
  subfamily $J$, with
  \[
    |J|+|V|\le |E(B)|+c(B).
  \]
  The stronger reserved cycle budget holds as well: each original small
  cyclic component reserves another unit in this inequality.
  Deleting its chosen single edges preserves reachability. Deleting its
  full cuts preserves reachability among outside vertices and leaves no
  cycle meeting any thin region. The resulting region-boundary quotient
  is a forest, with every connected component a tree. The simultaneous
  restoration theorem applies with $L=3M$, hence with factor $1+9M$ in
  a subcubic black graph for witnesses avoiding the isolated regions.
  The following theorem completes the suppression of the thin-path
  vertices. Preserving marks through weighted-tree reorganization and the
  final density contradiction in Monien--Preis's core argument remain unproved.
\end{theorem}
\begin{proof}
  \leanok
  Choose one of the one or two attachments of each thin region. Large
  original components containing eligible vertices are disjoint from every
  chosen small component, giving all region--attachment separations.
  The region and attachment sizes are at most $2M$ and $M$ respectively.
  Disjoint cuts give injective boundary choices, so the preceding cycle
  selection theorem applies to this constructed restoration family.
  Apply the core-connectivity theorem to the full-cut deletion, and the
  bridge-quotient theorem to the surviving boundaries of all thin regions.
  The designated deletions leave small original components untouched,
  giving the stronger cycle budget from the preceding mark theorem.
\end{proof}

\begin{theorem}[The core forest of the constructed thin paths]
  \label{thm:lowerbounds-algebraic-constructed-core-forest}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.PathSystem.exists_core_forest}
  \leanok
  \uses{thm:lowerbounds-algebraic-constructed-thin-family,
    thm:lowerbounds-algebraic-isolation-core-connectivity,
    thm:lowerbounds-algebraic-core-forest}
  Under the hypotheses of the constructed thin-family theorem, identify
  all thin regions in the boundary quotient after isolation and suppress
  their vertices. The resulting core graph is a forest, and reachability
  between its vertices agrees with reachability between their representatives
  in the original black graph. The $3M$ restoration bound per region and
  attachment and the original cycle-rank bound on isolated regions are retained.
  This includes the reserve for every original small cyclic component.
\end{theorem}
\begin{proof}
  \leanok
  Isolation deletes no internal edge of any region, so the regions remain
  connected and separated and their boundaries still have size at most two.
  Apply the preceding core-forest construction. A core representative belongs
  to none of the thin regions, so the outside-connectivity theorem upgrades
  reachability in the graph after isolation to original black reachability.
\end{proof}

\begin{theorem}[Counting the surviving thin paths]
  \label{thm:lowerbounds-algebraic-constructed-core-count}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.RestorationFamily.cut_deletedGraph_restrict,
    Algebraic.Cutwidth.Bisection.RedBlack.PathSystem.exists_counted_core_forest}
  \leanok
  \uses{thm:lowerbounds-algebraic-constructed-core-forest,
    thm:lowerbounds-algebraic-path-suppression-count,
    thm:lowerbounds-algebraic-thin-region-witness}
  Under the hypotheses of the constructed thin-family theorem, suppose
  eligible vertices have black degree exactly two. If $\mathcal P$ is the
  original thin-path family, $J$ is the selected shaded subfamily, and
  $T$ is the constructed core forest, then
  \[
    |E(T)|+|J|=|\mathcal P|.
  \]
  The same construction retains the restoration bound, original
  reachability between core representatives, and the cycle-rank bound
  $|J|+|V|\le |E(B)|+c(B)$.
  The stronger reserve for original small cyclic components is retained too.
\end{theorem}
\begin{proof}
  \leanok
  A thin region has a nonempty boundary. Since all its vertices have
  ambient degree two, that boundary has exactly two edges. The boundaries
  of distinct regions are disjoint. Isolating $J$ therefore empties
  precisely its cuts and preserves every other region's cut. The core
  edge count is consequently the number of unshaded regions. Partition
  the original thin-path family into its shaded and unshaded members.
\end{proof}

\begin{theorem}[Endpoint marks of the constructed thin paths]
  \label{thm:lowerbounds-algebraic-constructed-endpoint-marks}
  \lean{Algebraic.Cutwidth.Bisection.RedBlack.PathSystem.degree_endpointMarks,
    Algebraic.Cutwidth.Bisection.RedBlack.PathSystem.endpointMarks_eq_zero_of_mem,
    Algebraic.Cutwidth.Bisection.RedBlack.PathSystem.degree_isolated_add_endpointMarks,
    Algebraic.Cutwidth.Bisection.RedBlack.PathSystem.newly_degree_two_marked}
  \leanok
  \uses{thm:lowerbounds-algebraic-constructed-thin-family,
    thm:lowerbounds-algebraic-endpoint-marks,
    thm:lowerbounds-algebraic-family-restoration}
  For the constructed restoration family and shaded subfamily $J$, endpoint
  marks vanish on every eligible vertex. If eligible vertices have ambient
  degree exactly two, their total is $2|J|$. At every ineligible vertex,
  remaining degree plus endpoint marks equals original degree. Consequently,
  a degree-three vertex made degree two by isolation receives exactly one mark.
  It cannot become a new unmarked degree-two path vertex.

  For a positive witness $X$ avoiding the shaded regions, the original graph
  has a positive witness $Y\supseteq X$ with
  \[
    |Y|\le |X|+3M\sum_{v\in X}m(v).
  \]
  These are the endpoint invariants of the initial shading step. Preservation
  through later color swaps and leaf reorganization remains unproved.
\end{theorem}
\begin{proof}
  \leanok
  Every thin region has two boundary edges, giving two outside marks.
  An eligible vertex adjacent to a vertex of a canonical region belongs
  to the same induced component. Thus no boundary edge can mark an eligible
  vertex. Apply the general degree identity outside the eligible set.
  A degree-three vertex was ineligible, and a drop to degree two gives one mark.
  The constructed family has region-plus-attachment size at most $3M$,
  so the mark-based restoration theorem gives the stated witness bound.
\end{proof}

\begin{theorem}[Accumulating bounded helpful sets]
  \label{thm:lowerbounds-algebraic-helpful-amplification}
  \lean{Algebraic.Cutwidth.Bisection.exists_helpful_set_of_margin}
  \leanok
  \uses{thm:lowerbounds-algebraic-cut-improvement}
  Let $a\ge 0$ and let $M,k$ be natural numbers. Suppose
  $|\partial S|>a|S|+k$, and every $T\subseteq S$ with
  $|\partial T|>a|T|$ contains a set of at most $M$ vertices of
  helpfulness at least one. Then some $X\subseteq S$ has
  $|X|\le kM$ and $h_S(X)\ge k$. This proves the accumulation step in
  Monien--Preis's argument; the general bounded local helpful-set lemma
  is still a hypothesis.
\end{theorem}
\begin{proof}
  \leanok
  Until the total helpfulness reaches $k$, the remaining side has cut
  size greater than $|\partial S|-k>a|S|$, hence greater than $a$
  times its own cardinality. Apply the local lemma, remove its helpful
  set, and add the gain. At most $k$ moves are needed, and their sets
  are disjoint. Each contributes at most $M$ vertices.
\end{proof}

\begin{theorem}[Prescribed-size subsets and logarithmic-cost rebalancing]
  \label{thm:lowerbounds-algebraic-rebalancing}
  \lean{Algebraic.Cutwidth.Bisection.exists_subset_cut_le,
    Algebraic.Cutwidth.Bisection.exists_rebalancing_set,
    Algebraic.Cutwidth.MedianOrdering.card_cutFinset_le_bag}
  \leanok
  \uses{thm:lowerbounds-algebraic-subcubic-endpoint,
    thm:lowerbounds-algebraic-cut-improvement}
  In a cubic graph, every side $S$ and $0\le r\le |S|$ admit a subset
  $T\subseteq S$ of size $r$ with
  \[
    |\partial T|\le
    \max\{ |\partial S|,\lfloor |S|/3\rfloor+1\}
      +\lceil\log_2|S|\rceil+2.
  \]
  Consequently, if $|S|<3|\partial S|$, every prescribed move size
  $k\le |S|$ has a witness $X\subseteq S$ with
  $h_S(X)\ge-\lceil\log_2|S|\rceil-2$.
  No bisection hypothesis or connectivity assumption is used.
  The ceiling logarithm is interpreted as zero at zero.
\end{theorem}
\begin{proof}
  \leanok
  Delete vertices with at least two outside neighbors while the side is
  larger than the requested subset; each deletion decreases the cut.
  Once each remaining vertex has two internal neighbors, take its induced
  graph's prescribed-endpoint decomposition ending at the cut boundary.
  Extend it by one bag containing that boundary and the other side.
  Place edges in their first available bags. Every vertex on the chosen
  side has its median edge before the final bag; every other vertex has
  its median in the final bag. An initial segment of any prescribed
  cardinality is therefore contained in the chosen side. The median
  charging argument bounds its cut by one more than an earlier bag.
  For the move, take $X=S\setminus T$ with $|T|=|S|-k$ and use the exact
  helpfulness identity. The density condition absorbs the maximum.
\end{proof}

\begin{theorem}[Sharp bisection from bounded helpful sets]
  \label{thm:lowerbounds-algebraic-helpful-reduction}
  \lean{Algebraic.Cutwidth.Bisection.exists_bisection_of_helpful,
    Algebraic.Cutwidth.Bisection.exists_bisectionBound_of_helpful}
  \leanok
  \uses{thm:lowerbounds-algebraic-helpful-amplification,
    thm:lowerbounds-algebraic-rebalancing,
    def:lowerbounds-algebraic-bisection}
  Fix $\xi>0$. Suppose every side with cut density greater than
  $1/3+\xi$ contains a set of at most $M$ vertices of helpfulness at
  least one. A cubic graph on $n$ vertices has a bisection of size at most
  $(1/6+\xi)n$ whenever
  \[
    (M+3)(\lceil\log_2 n\rceil+4)<\xi n.
  \]
  A graph-independent $M$ therefore gives
  $\mathrm{BisectionBound}(\xi,N_0)$ for some $N_0$.
  The bounded local helpful-set lemma remains a hypothesis.
\end{theorem}
\begin{proof}
  \leanok
  Start from a minimum cut with one side of size $\lfloor n/2\rfloor$.
  If its size exceeds the bound, accumulate helpfulness
  $k=\lceil\log_2 n\rceil+3$ using at most $kM$ vertices.
  From the enlarged opposite side, select a subset of its original
  cardinality using the preceding theorem. The original-cut term of
  its maximum is paid by the accumulated gain; the side-size term is
  below the original cut by the displayed size condition. This restores
  balance with a strictly smaller cut, contradicting minimality.
  The maximum also handles overshoot of the accumulated gain.
  Finally, a fixed multiple of the logarithm is sublinear.
\end{proof}

\begin{theorem}[A coarse bisection bound]
  \label{thm:lowerbounds-algebraic-bisection-coarse}
  \lean{Algebraic.Cutwidth.bisectionBound_coarse}
  \leanok
  \uses{def:lowerbounds-algebraic-bisection}
  The predicate $\mathrm{BisectionBound}(4/3,0)$ holds. This checks the
  bisection contract with a coarse constant and does not prove the sharp
  theorem for arbitrarily small positive slack.
\end{theorem}
\begin{proof}
  \leanok
  Choose any $\lfloor n/2\rfloor$ vertices. The cut has at most all
  $3n/2$ edges of the cubic graph, by the degree-sum formula.
\end{proof}

\begin{theorem}[Path decomposition from a balanced cut]
  \label{thm:lowerbounds-algebraic-bisection-assembly}
  \lean{Algebraic.Cutwidth.PathDecomposition.exists_of_cut,
    Algebraic.Cutwidth.PathDecomposition.exists_of_balanced_cut,
    Algebraic.Cutwidth.card_cutBoundary_le,
    Algebraic.Cutwidth.card_cutFinset_induce_le}
  \leanok
  \uses{thm:lowerbounds-algebraic-subcubic-endpoint,
    thm:lowerbounds-algebraic-boundary-transition,
    thm:lowerbounds-algebraic-endpoint-glue}
  If a graph of maximum degree three on $n$ vertices has a balanced cut
  with $b$ crossing edges, it has a path decomposition with bags of size
  at most
  \[
    \max\{b,\lfloor(n+1)/6\rfloor+1\}+\lceil\log_2 n\rceil+1.
  \]
  For an arbitrary cut, replace $\lfloor(n+1)/6\rfloor$ by
  $\lfloor\max\{|S|,|S^c|\}/3\rfloor$. The logarithmic term is zero
  for $n=0$.
\end{theorem}
\begin{proof}
  \leanok
  Each vertex boundary has at most $b$ vertices. Apply the prescribed
  endpoint theorem to both induced sides and reverse the second
  decomposition. Between them put the decomposition on the two vertex
  boundaries, with bags of size at most $b+1$. The adjoining bags contain
  all shared vertices, so concatenation preserves the bag-size bound.
\end{proof}

\begin{theorem}[Cubic pathwidth from bisection]
  \label{thm:lowerbounds-algebraic-bisection-pathwidth}
  \lean{Algebraic.Cutwidth.BisectionBound.exists_pathwidthBound,
    Algebraic.Cutwidth.pathwidthBound_of_bisectionBound}
  \leanok
  \uses{def:lowerbounds-algebraic-bisection,
    thm:lowerbounds-algebraic-bisection-assembly}
  If $\xi\ge 0$, $\delta>0$, and $\mathrm{BisectionBound}(\xi,N_0)$
  holds, then $\mathrm{PathwidthBound}(\xi+\delta,N_1)$ holds for some
  $N_1$. Consequently the sharp asymptotic cubic bisection theorem implies
  the sharp asymptotic cubic pathwidth theorem. This proves the reduction
  of Fomin and H{\o}ie; the Monien--Preis bisection theorem remains a
  hypothesis.
\end{theorem}
\begin{proof}
  \leanok
  For all sufficiently large $n$, the ceiling logarithm plus two is less
  than $\delta n$. Absorb the finite assembly's remainder in that slack.
  For the universally quantified conclusion, allocate half the target
  slack to the bisection and half to the remainder.
\end{proof}

\begin{definition}[Flat-source sumset extraction]
  \label{def:lowerbounds-algebraic-sumset-extractor}
  \lean{Algebraic.Cutwidth.FlatSumsetExtractor,
    Algebraic.Cutwidth.flatSumsetExtractor_singleBit}
  \leanok
  A Boolean function $f$ on $n$ bits is a flat-source sumset extractor with
  threshold $K$ and error $\nu$ when, for all sets $P,Q\subseteq\{0,1\}^n$
  of cardinality at least $K$, the fraction of pairs $(x,y)\in P\times Q$
  with $f(x\oplus y)=1$ lies in $[1/2-\nu,1/2+\nu]$. Source pairs are
  counted with multiplicity even when their sums coincide. On one bit, the
  identity function has threshold $2$ and error zero. This finite example
  does not give an asymptotic family.
\end{definition}

\begin{theorem}[Sumset extraction supplies the hard-family properties]
  \label{thm:lowerbounds-algebraic-extractor-hardness}
  \lean{Algebraic.Cutwidth.FlatSumsetExtractor.balanced,
    Algebraic.Cutwidth.eventually_hard_of_flatSumsetExtractor}
  \leanok
  \uses{def:lowerbounds-algebraic-sumset-extractor}
  Suppose $\nu\le 1/4$, $\log_2 K(n)=o(n)$, and eventually $f_n$ is a
  flat-source sumset extractor with threshold $K(n)$ and error $\nu$.
  Then eventually $f_n$ is $K(n)$-rectangle-free and accepts at least
  $2^{n-2}$ inputs. More generally, flat-source extraction implies balance
  on all coordinate rectangles at the same threshold and error.
\end{theorem}
\begin{proof}
  \leanok
  Zero-pad the two rectangle sides on complementary coordinates; XOR
  recovers their glued input, with each source pair retained. An all-one
  rectangle contradicts error below $1/2$. Apply balance to the full cube,
  splitting its coordinates into nearly equal halves, for the density bound.
  Sublinear source entropy ensures that both halves support enough inputs.
\end{proof}

\begin{theorem}[Cutwidth bound, conditional]
  \label{thm:lowerbounds-algebraic-cutwidth}
  \lean{Algebraic.Cutwidth.eventually_lt_size_of_pathwidthBound}
  \leanok
  Assume that for every $\xi > 0$ all sufficiently large simple 3-regular
  graphs on $h$ vertices have pathwidth at most $(1/6 + \xi)h$, that is, a
  path decomposition whose bags have at most $(1/6 + \xi)h + 1$ vertices. Then
  every rectangle-free Boolean function family with polynomial rectangle
  threshold and at least $2^{n-2}$ accepting inputs, as in
  Theorem~\ref{thm:lowerbounds-algebraic-nechiporuk}, needs circuits of more
  than $(4 - \varepsilon)n$ gates over the full binary basis (all sixteen
  binary gates, every gate counted), for every $\varepsilon > 0$ and all large
  $n$. The pathwidth bound is an explicit
  hypothesis of the theorem. It is the published theorem of Fomin and H{\o}ie
  on the pathwidth of cubic graphs. Its reduction to the Monien--Preis
  bisection theorem is formalized in
  Theorem~\ref{thm:lowerbounds-algebraic-bisection-pathwidth}; the bisection
  theorem remains unformalized. As for
  Theorem~\ref{thm:lowerbounds-algebraic-nechiporuk}, no explicit
  rectangle-free family is formalized, so this is a conditional statement
  about any such family. The argument follows Ryan Williams's private working
  note of September 2026, which combines the rectangle bound for read-once
  programs with the circuit-to-read-once compiler of the author's counting
  note.
\end{theorem}
\begin{proof}
  \leanok
  Order the gates by a path decomposition of the circuit's wiring graph and
  count the rectangles cut at each position.
\end{proof}

\begin{theorem}[Subexponential-threshold cutwidth bound, conditional]
  \label{thm:lowerbounds-algebraic-cutwidth-subexponential}
  \lean{Algebraic.Cutwidth.eventually_lt_size_of_pathwidthBound_of_log_sublinear,
    Algebraic.Cutwidth.eventually_lt_size_of_pathwidthBound_of_flatSumsetExtractor,
    Algebraic.Cutwidth.eventually_lt_size_of_bisectionBound_of_log_sublinear,
    Algebraic.Cutwidth.eventually_lt_size_of_bisectionBound_of_flatSumsetExtractor}
  \leanok
  \uses{thm:lowerbounds-algebraic-cutwidth,
    thm:lowerbounds-algebraic-extractor-hardness,
    thm:lowerbounds-algebraic-bisection-pathwidth}
  The polynomial threshold in Theorem~\ref{thm:lowerbounds-algebraic-cutwidth}
  can be replaced by $\log_2 K(n)=o(n)$. In particular, a family of flat-source
  sumset extractors with this threshold and fixed error at most $1/4$
  satisfies the same circuit lower bound, assuming the cubic bisection
  theorem, which now supplies the pathwidth theorem through a checked
  reduction. The explicit extractor construction and its polynomial-time
  evaluation remain unformalized. Li's Theorem 7.13 (2023) provides
  logarithmic source entropy; Chattopadhyay and Liao (2021) already provide
  polylogarithmic source entropy, which is sufficient for this version.
\end{theorem}
\begin{proof}
  \leanok
  Keep the terms involving $\log_2 K$ in the finite counting inequality.
  The loss is $O(\log n+\log K)=o(n)$, so the coefficient remains four.
\end{proof}

\begin{theorem}[Cutwidth bound for nondeterministic circuits, conditional]
  \label{thm:lowerbounds-algebraic-cutwidth-nondet}
  \lean{Algebraic.Cutwidth.nondet_eventually_lt_size_of_pathwidthBound,
    Algebraic.Cutwidth.nondet_eventually_lt_size_of_log_sublinear,
    Algebraic.Cutwidth.nondet_eventually_lt_size_of_flatSumsetExtractor}
  \leanok
  \uses{thm:lowerbounds-algebraic-cutwidth,
    thm:lowerbounds-algebraic-cutwidth-subexponential}
  Under the hypotheses of Theorem~\ref{thm:lowerbounds-algebraic-cutwidth},
  the same bound holds for nondeterministic circuits: for every
  $\varepsilon > 0$ and all large $n$, every circuit on $n + m$ inputs with
  arbitrary $m$, whose accepted inputs $x$ are exactly those with some witness $y$
  making the output $1$ has more than $(4 - \varepsilon)n$ gates.
  Nondeterminism does not reduce the size below this bound. The subexponential
  threshold of Theorem~\ref{thm:lowerbounds-algebraic-cutwidth-subexponential}
  is also sufficient.
\end{theorem}
\begin{proof}
  \leanok
  Witness inputs are read by the wiring graph but carry no port of the
  computed function, so forgetting their ports (\texttt{Network.forget})
  leaves a network with the same multigraph computing the existential
  projection. The excess of the wiring graph counts every reachable input,
  ordinary or witness, so it is at most the gate count minus the ordinary
  inputs read. Every signal except the output feeds a slot; charging signals
  and copy vertices to slots gives at most $2s+1$ wiring vertices, independent
  of the declared witness count. The deterministic assembly then applies.
\end{proof}

\begin{theorem}[Average-case cutwidth bound, conditional]
  \label{thm:lowerbounds-algebraic-cutwidth-average}
  \lean{Algebraic.Cutwidth.eventually_card_agree_le_of_pathwidthBound}
  \leanok
  \uses{thm:lowerbounds-algebraic-cutwidth}
  Call $f$ $(K, \nu)$-balanced when on every rectangle with both sides of
  size at least $K$, under every split of the coordinates, the fraction of
  accepting inputs lies within $\nu$ of one half (a sumset extractor with
  error $\nu$ is balanced with $K$ polynomial). Under the pathwidth
  hypothesis of Theorem~\ref{thm:lowerbounds-algebraic-cutwidth}, for every
  family $f_n$ that is $(K_n, \nu)$-balanced with $K_n \le n^c$, every
  $\varepsilon > 0$, and all large $n$, every circuit with at most
  $(4 - \varepsilon)n$ binary gates agrees with $f_n$ on at most
  $(1/2 + 3\nu)\,2^n + 2^{(1 - \varepsilon/24)n}$ inputs.
\end{theorem}
\begin{proof}
  \leanok
  The worst-case count gives nothing here, since an input whose past set is
  still small at a vertex lies in a thin rectangle on which a balanced
  function is unconstrained. Instead the direction of the wiring graph is
  used: at a cut, the forward-crossing bits are determined by the inputs read
  so far together with the backward-crossing bits, and symmetrically for the
  future. This caps the number of cut assignments a fixed past or future can
  see, so the thin-rectangle mass at a prefix $L$ is at most
  $(K - 1)(2^{n - a(L)} + 2^{n - c(L)})$ with $a(L)$ the inputs read minus
  the forward cut and $c(L)$ the inputs unread minus the backward cut. Along
  the vertex order $a$ grows by at most four per step and $a + c$ is $n$
  minus the cut, so a prefix with both $a$ and $c$ linear exists whenever
  the cutwidth is below $(1 - \Omega(\varepsilon))n$. Circuits reading few
  inputs are constant on subcubes, which balance handles directly.
\end{proof}

\begin{definition}[The KRW conjecture]
  \label{def:lowerbounds-algebraic-krw}
  \lean{Algebraic.KW.NonConstant, Algebraic.KW.KRWDepthWith,
    Algebraic.KW.KRWSizeWith, Algebraic.KW.KRWDepth, Algebraic.KW.KRWSize,
    Algebraic.KW.formulaDepth_compose_le, Algebraic.KW.formulaSize_compose_le,
    Algebraic.KW.not_forall_depth_of_constant_inner,
    Algebraic.KW.not_forall_depth_of_constant_outer,
    Algebraic.KW.not_forall_size_of_constant_inner,
    Algebraic.KW.not_forall_size_of_constant_outer}
  \leanok
  \uses{thm:lowerbounds-algebraic-kw}
  Let $D(f)$ and $L(f)$ be the minimum depth and the minimum number of leaves,
  constant leaves included, of a De Morgan formula of
  Theorem~\ref{thm:lowerbounds-algebraic-kw} computing $f$; they are defined
  for every number of inputs, $0$ included. For $f$ on $m$ bits and $g$ on $n$
  bits, the block composition $f \diamond g$ applies $f$ to the values of $g$
  on $m$ disjoint blocks of $n$ bits. Substituting formulas gives
  $D(f \diamond g) \le D(f) + D(g)$ and $L(f \diamond g) \le L(f)\,L(g)$ for
  all $f$ and $g$; both bounds are proved.

  The Karchmer--Raz--Wigderson conjecture says these bounds are nearly tight.
  It is formalized in its strong form, with a constant slack. The depth form
  says that there is a constant $c$ such that
  $D(f) + D(g) \le D(f \diamond g) + c$ for all $m$, $n$ and all non-constant
  $f$ and $g$. The size form says that there is a constant $c$ such that
  $L(f)\,L(g) \le c \cdot L(f \diamond g)$ for all such $f$ and $g$. The
  constant does not depend on $m$, $n$, $f$ or $g$; forms whose slack grows
  with $m$ or $n$ are not formalized. The conjecture is open; it is formalized
  as a statement only.

  Both non-constancy hypotheses are needed. If $f$ or $g$ is constant, so is
  $f \diamond g$, and it is proved that dropping either hypothesis makes the
  depth form and the size form false for every $c$.
\end{definition}

\section{Essential inputs and gate elimination}
\label{sec:lowerbounds-gate-elimination}

\begin{definition}[Essential input]
  \label{def:lowerbounds-essential}
  \lean{Complexity.IsEssentialInput, Complexity.essentialInputs}
  \leanok
  \uses{def:bitstring}
  Input $i$ is essential for $f : \{0,1\}^N \to \{0,1\}^M$ if flipping bit $i$
  changes $f(x)$ for some $x$.
\end{definition}

\begin{theorem}[Essential-input bound]
  \label{thm:lowerbounds-essential}
  \lean{Complexity.Circuit.card_essentialInputs_le_totalFanIn,
    Complexity.Circuit.card_essentialInputs_le_mul_size}
  \leanok
  \uses{def:lowerbounds-essential, def:circuit, def:circuits-bases}
  Over any basis, the number of essential inputs of the function
  $\{0,1\}^N \to \{0,1\}^M$ computed by a circuit is at most the circuit's
  total fan-in, the number of gate-input occurrences with output gates
  included. For circuits over the AND/OR basis of fan-in at most $k$, it is at
  most $k$ times the size $G + M$.
\end{theorem}
\begin{proof}
  \leanok
  Every essential input is read by some gate input occurrence.
\end{proof}

\begin{definition}[Parity]
  \label{def:lowerbounds-parity}
  \lean{Complexity.Schnorr.xorBool}
  \leanok
  \uses{def:bitstring}
  The parity family $\oplus$ maps $x \in \{0,1\}^N$ to
  $x_0 \oplus \cdots \oplus x_{N-1}$, with value $0$ on the empty string.
\end{definition}

\begin{theorem}[Schnorr]
  \label{thm:lowerbounds-schnorr}
  \lean{Complexity.schnorr_lower_bound_circuit, Complexity.sizeComplexity_xorBool_ge}
  \leanok
  \uses{def:lowerbounds-parity, def:circuits-size-complexity}
  Every fan-in-two AND/OR circuit computing $N$-bit parity or its complement,
  $N \ge 1$, has size at least $2N - 1$, that is, at least $2(N - 1)$ internal
  gates. Hence $\mathrm{size}(\oplus_N) \ge 2N - 1$.
\end{theorem}
\begin{proof}
  \leanok
  Gate elimination: fixing a well-chosen input removes at least two gates and
  leaves parity or its complement on one fewer input.
\end{proof}

\section{Depth, formulas, and communication}
\label{sec:lowerbounds-depth}

\begin{theorem}[Valiant's depth reduction]
  \label{thm:lowerbounds-valiant}
  \lean{Complexity.Valiant.depth_reduction}
  \leanok
  Let $G$ be a finite acyclic digraph with $S$ edges whose longest simple path
  has at most $2^k$ vertices, and let $r \le k$. There is a set $F$ of edges of
  $G$ with $k\,|F| \le r S$ whose removal leaves a digraph whose longest simple
  path has at most $2^k / 2^r$ vertices.
\end{theorem}
\begin{proof}
  \leanok
  Label vertices by longest-path depth, partition the edges by the first bit
  in which the labels of their endpoints differ, and remove the $r$ smallest
  classes.
\end{proof}

\begin{corollary}[Depth reduction for circuits]
  \label{cor:lowerbounds-valiant-circuit}
  \uses{def:circuit}
  Let $C$ be a circuit whose dependency graph has longest paths of at most
  $2^k$ vertices, and let $r \le k$. There is a set $F$ of dependency edges
  with $k\,|F| \le r \cdot \mathrm{totalFanIn}(C)$ whose removal leaves a
  graph whose longest paths have at most $2^{k-r}$ vertices.
\end{corollary}
\begin{proof}
  \uses{thm:lowerbounds-valiant, lem:circuits-dependency-graph}
  The dependency graph is acyclic and has at most $\mathrm{totalFanIn}(C)$
  edges.
\end{proof}

\begin{theorem}[Spira]
  \label{thm:lowerbounds-spira}
  \lean{Complexity.BoolFormula.exists_spira_balanced}
  \leanok
  \uses{def:circuits-formula}
  Every Boolean formula $\varphi$ of size $s$ has an equivalent formula of
  depth at most $15 + 12 \lceil \log_2 s \rceil$ and size at most
  $80 s^8 - 4$.
\end{theorem}
\begin{proof}
  \leanok
  Find a subformula of between one third and two thirds of the size, and
  recombine the two cases for its value; recurse on both parts.
\end{proof}

\begin{definition}[Monotone formula]
  \label{def:lowerbounds-monotone-formula}
  \lean{Complexity.MonotoneFormula, Complexity.MonotoneFormula.eval,
    Complexity.MonotoneFormula.depth, Complexity.MonotoneFormula.leaves,
    Complexity.IsMonotoneBoolFun}
  \leanok
  \uses{def:bitstring}
  A monotone formula over $N$ variables is a tree built from variables
  $x_0, \dots, x_{N-1}$ and binary conjunction and disjunction, without
  constants or negation. Its leaves are its variable occurrences, and its depth
  is the length of a longest root-to-leaf path, variables having depth zero. A
  function $f$ is monotone if $x \le y$ pointwise and $f(x) = 1$ imply
  $f(y) = 1$.
\end{definition}

\begin{lemma}[Monotone formulas]
  \label{lem:lowerbounds-monotone-formula}
  \lean{Complexity.MonotoneFormula.eval_monotone,
    Complexity.MonotoneFormula.card_essentialInputs_le_leaves_of_computes}
  \leanok
  \uses{def:lowerbounds-monotone-formula, def:lowerbounds-essential}
  Every monotone formula computes a monotone function, and a monotone formula
  computing $f$ has at least as many leaves as $f$ has essential inputs.
\end{lemma}
\begin{proof}
  \leanok
  Induction on the formula; an essential input must occur as a leaf.
\end{proof}

\begin{definition}[Karchmer--Wigderson protocol]
  \label{def:lowerbounds-kw}
  \lean{Complexity.KarchmerWigderson.Protocol,
    Complexity.KarchmerWigderson.RootProtocol,
    Complexity.KarchmerWigderson.Protocol.depth}
  \leanok
  \uses{def:bitstring}
  A protocol is a tree indexed by a rectangle $X \times Y$ of sets of inputs in
  $\{0,1\}^N$, $X$ on the one side and $Y$ on the zero side. At an internal
  node, Alice splits $X$ or Bob splits $Y$ by an arbitrary predicate, and the
  two children are indexed by the two parts. A leaf names a coordinate $i$ with
  $x_i = 1$ for all $x \in X$ and $y_i = 0$ for all $y \in Y$. A protocol for
  the monotone Karchmer--Wigderson game of $f : \{0,1\}^N \to \{0,1\}$ is one
  indexed by the root rectangle $f^{-1}(1) \times f^{-1}(0)$, so every node's
  rectangle lies in it. The depth is the height of the tree.
\end{definition}

\begin{theorem}[Monotone Karchmer--Wigderson]
  \label{thm:lowerbounds-kw}
  \lean{Complexity.KarchmerWigderson.Protocol.exists_protocol_of_formula,
    Complexity.KarchmerWigderson.Protocol.exists_formula_of_protocol,
    Complexity.KarchmerWigderson.Protocol.exists_protocol_depth_iff_formula_depth}
  \leanok
  \uses{def:lowerbounds-kw, def:lowerbounds-monotone-formula}
  Every monotone formula computing $f : \{0,1\}^N \to \{0,1\}$ yields a
  protocol for the monotone Karchmer--Wigderson game of $f$ of the same depth,
  and every such protocol yields a monotone formula of the same depth computing
  $f$. Hence, for every $d \in \mathbb{N}$, the game has a protocol of depth at
  most $d$ if and only if some monotone formula of depth at most $d$ computes
  $f$. There is no monotonicity hypothesis on $f$; by
  Lemma~\ref{lem:lowerbounds-monotone-formula}, both sides fail when $f$ is not
  monotone.
\end{theorem}
\begin{proof}
  \leanok
  An OR node is an Alice split and an AND node is a Bob split; both
  translations preserve depth exactly.
\end{proof}

\section{Restrictions, normal forms, and decision trees}
\label{sec:lowerbounds-restrictions}

\begin{definition}[Restriction]
  \label{def:lowerbounds-restriction}
  \lean{Complexity.Restriction, Complexity.Restriction.On,
    Complexity.Restriction.applyTo, Complexity.Restriction.comp,
    Complexity.BoolFormula.restrict}
  \leanok
  \uses{def:circuits-formula}
  A restriction assigns each variable $x_i$, $i \in \mathbb{N}$, either a
  fixed bit or ``free''; the finite-arity version does this for exactly $N$
  variables. The composition $\rho_1 \circ \rho_2$ fixes a variable to
  $\rho_1$'s value when $\rho_1$ fixes it and to $\rho_2$'s value otherwise.
  A restriction acts on a total assignment by overwriting the fixed variables,
  and on a Boolean formula by replacing each fixed variable with the
  corresponding constant.
\end{definition}

\begin{lemma}[Restriction and evaluation]
  \label{lem:lowerbounds-restrict}
  \lean{Complexity.BoolFormula.eval_restrict, Complexity.BoolFormula.restrict_comp,
    Complexity.BoolFormula.restrict_size, Complexity.BoolFormula.restrict_depth}
  \leanok
  \uses{def:lowerbounds-restriction}
  Evaluating $\varphi|_\rho$ at $\alpha$ equals evaluating $\varphi$ at
  $\rho$ applied to $\alpha$. Restricting by $\rho_1$ and then by $\rho_2$
  equals restricting by $\rho_1 \circ \rho_2$, and restriction preserves
  formula size and depth exactly.
\end{lemma}
\begin{proof}
  \leanok
  Induction on the formula.
\end{proof}

\begin{definition}[CNF and DNF]
  \label{def:lowerbounds-normal-forms}
  \lean{Complexity.CNF, Complexity.DNF, Complexity.CNF.width, Complexity.DNF.width,
    Complexity.CNF.complexity, Complexity.DNF.complexity}
  \leanok
  \uses{def:bitstring}
  A CNF (resp.\ DNF) over $N$ variables is a list of clauses (resp.\ terms),
  each a list of literals. Its width is the largest number of literals in a
  clause (term), and its complexity is the number of clauses (terms). These
  finite-arity normal forms are distinct from the CNF formulas used for
  satisfiability and resolution.
\end{definition}

\begin{lemma}[Restricting normal forms]
  \label{lem:lowerbounds-normal-form-restrict}
  \lean{Complexity.CNF.restrict, Complexity.CNF.eval_restrict,
    Complexity.CNF.width_restrict_le, Complexity.DNF.restrict,
    Complexity.DNF.eval_restrict, Complexity.DNF.width_restrict_le}
  \leanok
  \uses{def:lowerbounds-normal-forms, def:lowerbounds-restriction}
  For a finite-arity restriction $\rho$ and a CNF or DNF $\varphi$ over $N$
  variables, the simplified restriction $\varphi|_\rho$ satisfies
  $\varphi|_\rho(x) = \varphi(\rho \text{ applied to } x)$ for every
  $x \in \{0,1\}^N$, and its width is at most the width of $\varphi$.
\end{lemma}
\begin{proof}
  \leanok
  In a CNF delete falsified literals and satisfied clauses; in a DNF delete
  satisfied literals and falsified terms.
\end{proof}

\begin{definition}[Decision tree]
  \label{def:lowerbounds-decision-tree}
  \lean{Complexity.DecisionTree, Complexity.DecisionTree.On,
    Complexity.DecisionTree.On.eval, Complexity.DecisionTree.On.depth}
  \leanok
  \uses{def:bitstring}
  A decision tree queries a variable at each internal node and outputs a bit
  at each leaf. The finite-arity version queries only variables below $N$.
  Its depth is the length of a longest root-to-leaf path.
\end{definition}

\begin{lemma}[Decision trees to normal forms]
  \label{lem:lowerbounds-dt-normal-form}
  \lean{Complexity.DecisionTree.On.eval_toDNF, Complexity.DecisionTree.On.eval_toCNF,
    Complexity.DecisionTree.On.width_toDNF_le_depth,
    Complexity.DecisionTree.On.width_toCNF_le_depth}
  \leanok
  \uses{def:lowerbounds-decision-tree, def:lowerbounds-normal-forms}
  A decision tree of depth $s$ is computed by a DNF of width at most $s$ and
  by a CNF of width at most $s$.
\end{lemma}
\begin{proof}
  \leanok
  Take one term per accepting path, and dually one clause per rejecting path.
\end{proof}

\begin{theorem}[Parity at depth two]
  \label{thm:lowerbounds-parity-depth-two}
  \lean{Complexity.DNF.two_pow_le_complexity_of_xorBool,
    Complexity.CNF.two_pow_le_complexity_of_xorBool}
  \leanok
  \uses{def:lowerbounds-normal-forms, def:lowerbounds-parity}
  For $N \ge 1$, every DNF and every CNF computing $N$-bit parity has at
  least $2^{N-1}$ terms, respectively clauses.
\end{theorem}
\begin{proof}
  \leanok
  A satisfiable term that omits a variable accepts two inputs of different
  parity, so every satisfiable term accepts exactly one input, while parity
  accepts $2^{N-1}$ inputs. CNFs follow by negation.
\end{proof}

\section{The switching lemma and $\cc{AC}^0$}
\label{sec:lowerbounds-ac0}

\begin{definition}[Sparse random restriction]
  \label{def:lowerbounds-random-restriction}
  \lean{Complexity.RandomRestriction.Seed, Complexity.RandomRestriction.decode,
    Complexity.RandomRestriction.eventCount, Complexity.RandomRestriction.card_seed,
    Complexity.RandomRestriction.card_decodeSymbol_none,
    Complexity.RandomRestriction.card_decodeSymbol_some}
  \leanok
  \uses{def:lowerbounds-restriction}
  For $q \in \mathbb{N}$, a seed assigns to each of $N$ coordinates one of
  $2q + 1$ symbols: one free symbol and $q$ labelled copies of each fixed bit,
  so there are exactly $(2q + 1)^N$ seeds. A seed decodes to a finite-arity
  restriction, so a uniform seed leaves each coordinate free with probability
  $1/(2q+1)$ and fixes it to each bit with probability $q/(2q+1)$,
  independently. Events are measured by exact counts of seeds rather than by
  probabilities.
\end{definition}

\begin{theorem}[Switching lemma, counting form]
  \label{thm:lowerbounds-switching}
  \lean{Complexity.DNF.switchingBad,
    Complexity.DNF.switchingBad_width_encoding_bound,
    Complexity.DNF.switchingBad_consistentPart_width_encoding_bound,
    Complexity.CNF.switchingBad_width_encoding_bound,
    Complexity.CNF.switchingBad_consistentPart_width_encoding_bound,
    Complexity.DNF.eval_consistentPart, Complexity.CNF.eval_consistentPart}
  \leanok
  \uses{def:lowerbounds-random-restriction, def:lowerbounds-normal-forms,
    def:lowerbounds-decision-tree}
  Let $\varphi$ be a DNF of width $w$ over $N$ variables, no term of which
  contains a complementary pair of literals, and let $q, s \in \mathbb{N}$.
  Call a restriction bad if the complete-block switching decision tree of the
  restricted formula has depth at least $s$. This tree, rather than the
  canonical tree that queries one variable at a time, is the one used here: at
  the first surviving term it queries all of that term's free variables before
  accepting or continuing.
  Then
  \[ \#\{\text{bad seeds}\} \cdot q^s \le (2q + 1)^N \, (4(w + 1))^s . \]
  Every DNF agrees with its sub-DNF of consistent terms, whose width is no
  larger, and the bound holds for that sub-DNF with $w$ the original width.
  The same bounds hold for CNFs, where the condition is that no clause
  contains a complementary pair, and the switching tree and bad event are
  defined through the De Morgan dual DNF. Reading the counts as
  probabilities, which is not a separate formal statement: a uniform seed is
  bad with probability at most $(4(w+1)/q)^s$ when $q \ge 1$.
\end{theorem}
\begin{proof}
  \leanok
  Encode each bad restriction, together with the labels of its first $s$
  queried coordinates, injectively as a seed plus $s$ advice symbols from an
  alphabet of size $4(w + 1)$ (Razborov's encoding). CNFs follow by De Morgan
  duality.
\end{proof}

\begin{definition}[$\cc{AC}^0$ formula]
  \label{def:lowerbounds-ac0-formula}
  \lean{Complexity.AC0Formula, Complexity.AC0Formula.eval, Complexity.AC0Formula.size,
    Complexity.AC0Formula.depth}
  \leanok
  \uses{def:bitstring}
  A negation-normal unbounded formula over $N$ variables is a tree built from
  constants, literals, and AND and OR gates of any fan-in; an empty AND is true
  and an empty OR is false. Its size counts all nodes (constants, literal
  occurrences, and gates), and its depth counts gates on a longest
  root-to-leaf path.
\end{definition}

\begin{theorem}[Normalization of $\cc{AC}^0$ circuits]
  \label{thm:lowerbounds-ac0-normalization}
  \lean{Complexity.Circuit.outputAC0Formula, Complexity.Circuit.outputAC0Formula_spec}
  \leanok
  \uses{def:lowerbounds-ac0-formula, def:circuits-bases}
  Let $C$ be an unbounded-fan-in AND/OR circuit, with negations only through
  the free per-input flags, with $N$ inputs and $G$ internal gates, and let
  output $j$ have depth $D$, counting the output gate. Some negation-normal
  unbounded formula computes output $j$, has depth at most $D$, and has size at
  most $(2(N + G) + 1)^{D+1}$.
\end{theorem}
\begin{proof}
  \leanok
  Push negations to the literals by De Morgan duality, remove duplicate signed
  inputs of each gate, and unfold the circuit into a tree.
\end{proof}

\begin{theorem}[Iterated switching]
  \label{thm:lowerbounds-staged-switching}
  \lean{Complexity.AC0Formula.stagedDecisionTree,
    Complexity.AC0Formula.eval_stagedDecisionTree,
    Complexity.AC0Formula.stageEventCount_stagedBad_mul_pow_le,
    Complexity.AC0Formula.exists_shallow_stagedDecisionTree_of_counting}
  \leanok
  \uses{def:lowerbounds-ac0-formula, def:lowerbounds-random-restriction,
    def:lowerbounds-decision-tree}
  Let $\varphi$ be a negation-normal unbounded formula over $N$ variables with
  depth at most $d$, and let $s \ge 2$ and $q \in \mathbb{N}$. For any
  sequence of $d$ restrictions, one per level, a staged compiler turns
  $\varphi$ into a decision tree that computes $\varphi$ under their composition,
  earlier restrictions taking precedence. For $d$ independent sparse seeds, the
  number of $d$-tuples of seeds for which this decision tree has depth at least
  $s$, multiplied by $q^s$, is at most
  $\mathrm{size}(\varphi)\,((2q+1)^N)^d\,(4(s + 1))^s$. Consequently, if $q > 0$
  and
  \[ ((2q+1)^N)^d\, s\, q^s + \mathrm{size}(\varphi)\,((2q+1)^N)^d\,(4(s+1))^s N
     < N\, ((2q+1)^{N-1})^d\, q^s , \]
  then some tuple of seeds leaves at least $s$ variables free and reduces
  $\varphi$ to a decision tree of depth less than $s$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:lowerbounds-switching, lem:lowerbounds-dt-normal-form,
    lem:lowerbounds-normal-form-restrict}
  At a gate, the earlier stages have compiled the children to decision trees;
  their conjunction is a CNF and their disjunction a DNF of width at most the
  largest child depth. Unless some child already failed, that width is below
  $s$, so the switching lemma bounds the failures at the gate. A union bound
  over the nodes and the exact first moment of the number of surviving
  variables give the criterion.
\end{proof}

\begin{theorem}[Finite parity obstruction]
  \label{thm:lowerbounds-parity-obstruction}
  \lean{Complexity.AC0Formula.parity_counting_obstruction}
  \leanok
  \uses{def:lowerbounds-ac0-formula, def:lowerbounds-parity}
  Let $\varphi$ be a negation-normal unbounded formula over $N$ variables
  that computes parity, and let $d \ge \mathrm{depth}(\varphi)$, $s \ge 2$ and
  $q \ge 1$. Then
  \[ N\, ((2q+1)^{N-1})^d\, q^s \le ((2q+1)^N)^d\, s\, q^s
     + \mathrm{size}(\varphi)\,((2q+1)^N)^d\,(4(s+1))^s N . \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:lowerbounds-staged-switching}
  Parity restricted to a set of free variables needs a decision tree querying
  all of them, so the conclusion of Theorem~\ref{thm:lowerbounds-staged-switching}
  is impossible and its numeric hypothesis must fail.
\end{proof}

\begin{theorem}[Parity is not in $\cc{AC}^0$]
  \label{thm:lowerbounds-parity-not-ac0}
  \uses{def:NC-AC, def:lowerbounds-parity}
  The parity family is not in $\cc{AC}^0$. The analogous statement for the
  imported library's $\cc{AC}^0$ model is
  Theorem~\ref{thm:lowerbounds-algebraic-parity-ac0}; this one, for
  $\cc{AC}^0$ as defined in this library, is not yet formalized.
\end{theorem}
\begin{proof}
  \uses{thm:lowerbounds-ac0-normalization, thm:lowerbounds-parity-obstruction}
  Normalize a constant-depth polynomial-size family into formulas of constant
  depth $d$ and polynomial size, then choose $q$ and $s$ as powers of $N$ so
  that the inequality of Theorem~\ref{thm:lowerbounds-parity-obstruction}
  fails for large $N$.
\end{proof}

\begin{theorem}[Håstad's size bound]
  \label{thm:lowerbounds-hastad-size}
  \uses{def:lowerbounds-parity, def:circuits-bases}
  For every $d \ge 2$ there is $\varepsilon_d > 0$ such that, for all
  sufficiently large $N$, every unbounded-fan-in AND/OR circuit of depth $d$
  computing $N$-bit parity has size at least $2^{\varepsilon_d N^{1/(d-1)}}$.
  Theorem~\ref{thm:lowerbounds-algebraic-hastad} proves an explicit
  non-asymptotic form of this bound for the imported library's $\cc{AC}^0$
  model; the statement here, for the circuits of this library, is not yet
  formalized.
\end{theorem}
\begin{proof}
  \uses{thm:lowerbounds-ac0-normalization, thm:lowerbounds-staged-switching}
  The same argument with the restriction parameters optimized, keeping the
  bottom fan-in bound separate from the size.
\end{proof}

\begin{lemma}[Parity in $\cc{TC}^0$]
  \label{lem:lowerbounds-parity-tc0}
  \uses{def:circuits-TC, def:lowerbounds-parity}
  The parity family is in $\cc{TC}^0$.
\end{lemma}
\begin{proof}
  For each odd $k \le N$, the conjunction of the threshold gates
  ``at least $k$ ones'' and ``not at least $k + 1$ ones'' detects exactly $k$
  ones; parity is the disjunction of these, a depth-three circuit of linear
  size.
\end{proof}

\begin{corollary}
  \label{cor:lowerbounds-ac0-ne-tc0}
  \uses{def:NC-AC, def:circuits-TC}
  $\cc{AC}^0 \subsetneq \cc{TC}^0$.
\end{corollary}
\begin{proof}
  \uses{prop:circuits-depth-classes, thm:lowerbounds-parity-not-ac0,
    lem:lowerbounds-parity-tc0}
  $\cc{AC}^0 \subseteq \cc{TC}^0$ is formalized, and parity separates them.
\end{proof}

\begin{theorem}[Pseudorandom switching lemma]
  \label{thm:lowerbounds-pseudorandom-restrictions}
  \uses{thm:lowerbounds-switching}
  \notready
  The bound of Theorem~\ref{thm:lowerbounds-switching} holds, up to an additive
  error $\varepsilon$, when the free/fixed pattern and the fixed values are
  drawn from a distribution that $\varepsilon$-fools CNFs of size
  polynomial in $N$ instead of the uniform product distribution
  (Trevisan--Xue). With an explicit generator fooling CNFs, such restrictions
  can be sampled from $\mathrm{polylog}(N / \varepsilon)$ random bits.
\end{theorem}
\begin{proof}
  The bad event of the switching lemma is itself decided by a small CNF-like
  test on the restriction, so fooling that test preserves its probability.
  This requires a formal notion of pseudorandom distributions and of fooling a
  class of tests, which the library does not yet state; pseudorandom
  generators for $\cc{AC}^0$ would build on it.
\end{proof}

\section{Monotone circuits}
\label{sec:lowerbounds-monotone}

\begin{definition}[Monotone circuit]
  \label{def:lowerbounds-monotone-circuit}
  \uses{def:circuit, def:circuits-bases}
  A monotone circuit is an AND/OR circuit all of whose negation flags are
  false. Every monotone circuit computes a monotone function.
\end{definition}

\begin{theorem}[Monotone Boolean matrix product]
  \label{thm:lowerbounds-monotone-matrix}
  \uses{def:lowerbounds-monotone-circuit}
  \notready
  Every monotone fan-in-two circuit computing the product of two
  $n \times n$ Boolean matrices has at least $n^3$ AND gates.
\end{theorem}
\begin{proof}
  Each product $x_{ik} y_{kj}$ is a prime implicant of output $(i, j)$, and a
  monotone circuit must realize each of the $n^3$ of them at a distinct AND
  gate (Pratt; Paterson; Mehlhorn--Galil). The bound is monotone-specific:
  circuits with negation compute the product with $O(n^{\log_2 7})$ gates
  through integer matrix multiplication.
\end{proof}

\begin{lemma}[Sunflower lemma]
  \label{lem:lowerbounds-sunflower}
  \lean{Algebraic.Sunflower.containsSunflower_of_uniform,
    Algebraic.Sunflower.containsSunflower_of_bounded}
  \leanok
  Let $p \ge 2$. Every finite family of more than $\ell!\,(p-1)^\ell$ finite
  sets, each of size exactly $\ell$, contains $p$ distinct sets whose pairwise
  intersections are all equal (to their common intersection). The same holds
  for more than $(\ell + 1)\,\ell!\,(p-1)^\ell$ sets, each of size at most
  $\ell$. This is proved in the imported algebraic-circuits library.
\end{lemma}
\begin{proof}
  \leanok
  Induction on $\ell$: either a maximal pairwise-disjoint subfamily has $p$
  members, or some element lies in many sets and the lemma applies to them
  with that element removed (Erd\H{o}s--Rado). For sets of size at most $\ell$,
  some size class has more than $\ell!\,(p-1)^\ell$ members.
\end{proof}

\begin{theorem}[Monotone clique lower bound]
  \label{thm:lowerbounds-monotone-clique}
  \uses{def:lowerbounds-monotone-circuit}
  \notready
  There is $\varepsilon > 0$ such that for all $k \le n^{1/4}$, every monotone
  circuit deciding whether an $n$-vertex graph, given by its adjacency bits,
  contains a $k$-clique has size at least $2^{\varepsilon \sqrt{k}}$
  (Razborov; Alon--Boppana; Arora--Barak Theorem~14.7). Only a weaker bound at
  special parameters, in the imported library's model, is formalized:
  Theorem~\ref{thm:lowerbounds-algebraic-clique} gives more than
  $2^{t \cdot 2^t}$ gates for $k = 2^{4t}$ and $n = 2^{20t}$, $t \ge 4$, which
  for large $t$ is below $2^{\varepsilon \sqrt{k}} = 2^{\varepsilon 4^t}$.
\end{theorem}
\begin{proof}
  \uses{lem:lowerbounds-sunflower}
  Razborov's method of approximations: replace each gate by an approximator
  that is a small union of clique indicators, use the sunflower lemma to keep
  approximators small, and show that each replacement errs on few positive or
  negative test graphs.
\end{proof}

\section{Proof complexity}
\label{sec:lowerbounds-proof-complexity}

\begin{definition}[Resolution]
  \label{def:lowerbounds-resolution}
  \lean{Complexity.SAT.Clause.resolvent, Complexity.SAT.CNF.Derives}
  \leanok
  Clauses are lists of literals over variables in $\mathbb{N}$, and a CNF is a
  list of clauses; an assignment is a finite list of bits, variables beyond its
  end reading $0$. The resolvent of clauses $C_1$ and $C_2$ on a variable $v$
  deletes every occurrence of $x_v$ from $C_1$ and of $\neg x_v$ from $C_2$
  and concatenates the rest. A clause is derivable from $\varphi$ if it is a
  clause of $\varphi$ or a resolvent of two derivable clauses. There is no
  weakening rule, and the pivot need not occur in the parents.
\end{definition}

\begin{lemma}[Resolution step]
  \label{lem:lowerbounds-resolvent}
  \lean{Complexity.SAT.Clause.resolvent_sound, Complexity.SAT.Clause.resolvent_length_le}
  \leanok
  \uses{def:lowerbounds-resolution}
  Every assignment satisfying $C_1$ and $C_2$ satisfies their resolvent on any
  variable, and the resolvent has at most $|C_1| + |C_2|$ literals.
\end{lemma}
\begin{proof}
  \leanok
  Case on the value of $x_v$: the parent whose pivot literal is false keeps a
  true literal in the resolvent.
\end{proof}

\begin{theorem}[Soundness of resolution]
  \label{thm:lowerbounds-resolution-sound}
  \lean{Complexity.SAT.CNF.entails_of_derives, Complexity.SAT.CNF.refutation_sound}
  \leanok
  \uses{def:lowerbounds-resolution}
  Every clause derivable from $\varphi$ is satisfied by every assignment
  satisfying $\varphi$. In particular, if the empty clause is derivable from
  $\varphi$, then $\varphi$ is unsatisfiable.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:lowerbounds-resolvent}
  Induction on the derivation.
\end{proof}

\begin{theorem}[Completeness of resolution]
  \label{thm:lowerbounds-resolution-complete}
  \uses{def:lowerbounds-resolution}
  If $\varphi$ is unsatisfiable, then the empty clause is derivable from
  $\varphi$.
\end{theorem}
\begin{proof}
  Induction on the number of variables, eliminating one variable by resolving
  all pairs of clauses on it (Davis--Putnam).
\end{proof}

\begin{definition}[Resolution refutations and their measures]
  \label{def:lowerbounds-resolution-proof}
  \uses{def:lowerbounds-resolution}
  A resolution refutation of $\varphi$ is a sequence of clauses ending in the
  empty clause, each a clause of $\varphi$ or a resolvent of two earlier ones.
  Its size is its length and its width is the largest number of literals in
  one of its clauses. A clause is derivable exactly when it ends such a
  sequence.
\end{definition}

\begin{theorem}[Ben-Sasson--Wigderson size--width relation]
  \label{thm:lowerbounds-size-width}
  \uses{def:lowerbounds-resolution-proof}
  \notready
  If an unsatisfiable CNF over $n$ variables whose clauses have at most $k$
  literals has a resolution refutation of size $S$, then it has one of width at
  most $k + O(\sqrt{n \log S})$.
\end{theorem}
\begin{proof}
  Repeatedly restrict a variable occurring in many wide clauses, and combine
  the refutations of the two restrictions.
\end{proof}

\begin{theorem}[Haken]
  \label{thm:lowerbounds-php}
  \uses{def:lowerbounds-resolution-proof}
  \notready
  Let $\mathrm{PHP}^{n+1}_n$ be the CNF over variables $p_{ij}$
  ($i \le n + 1$, $j \le n$) stating that every pigeon $i$ sits in some hole
  and no hole holds two pigeons. Every resolution refutation of
  $\mathrm{PHP}^{n+1}_n$ has size $2^{\Omega(n)}$.
\end{theorem}
\begin{proof}
  \uses{thm:lowerbounds-resolution-sound}
  Restrict the refutation so that every wide clause disappears, then show that
  any refutation of the remaining pigeonhole instance contains a wide clause
  (Beame--Pitassi), or use Haken's bottleneck counting.
\end{proof}

\begin{definition}[Cook--Reckhow proof system]
  \label{def:lowerbounds-proof-system}
  \uses{def:P}
  A propositional proof system for unsatisfiable CNFs is a polynomial-time
  decidable relation $V(\varphi, \pi)$ that is sound ($V(\varphi, \pi)$
  implies that $\varphi$ is unsatisfiable) and complete (every unsatisfiable
  $\varphi$ has some $\pi$ with $V(\varphi, \pi)$). It is polynomially bounded
  if every unsatisfiable $\varphi$ has such a $\pi$ with
  $|\pi| \le p(|\varphi|)$ for a fixed polynomial $p$.
\end{definition}

\begin{proposition}[Resolution is a proof system]
  \label{prop:lowerbounds-resolution-proof-system}
  \uses{def:lowerbounds-proof-system, def:lowerbounds-resolution-proof}
  \notready
  Checking a resolution refutation, encoded as a clause sequence, is a
  Cook--Reckhow proof system.
\end{proposition}
\begin{proof}
  \uses{thm:lowerbounds-resolution-sound, thm:lowerbounds-resolution-complete}
  Checking each step takes polynomial time; soundness and completeness are the
  two theorems above.
\end{proof}

\begin{theorem}[Cook--Reckhow]
  \label{thm:lowerbounds-cook-reckhow}
  \uses{def:lowerbounds-proof-system, def:NP, def:coNP}
  \notready
  A polynomially bounded propositional proof system exists if and only if
  $\cc{NP} = \cc{coNP}$.
\end{theorem}
\begin{proof}
  \uses{thm:cook-levin}
  Unsatisfiability of CNFs is $\cc{coNP}$-complete. A polynomially bounded
  proof system puts it in $\cc{NP}$; conversely an $\cc{NP}$ verifier for it is
  a polynomially bounded proof system.
\end{proof}

\section{Top-down parity and majority lower bounds}
\label{sec:lowerbounds-topdown-parity}

\begin{definition}[General bounded-round KW protocols]
  \label{def:lowerbounds-round-protocol}
  \lean{Complexity.KarchmerWigderson.RoundProtocol,
    Complexity.KarchmerWigderson.RoundProtocol.run,
    Complexity.KarchmerWigderson.RoundProtocol.Solves,
    Complexity.KarchmerWigderson.RoundProtocol.SolvesKW,
    Complexity.KarchmerWigderson.RoundProtocol.swapPlayers}
  \leanok
  A protocol on coordinates $I$ with alphabet $M$ and at most $d$ rounds
  either outputs a fixed coordinate or lets one player send a message in
  $M$, followed by a protocol with one fewer round. The speaker depends
  only on the preceding transcript, and a message depends only on that
  speaker's input. A solving protocol outputs a coordinate where the two
  inputs differ, in either orientation. For a function $f$, Alice's input
  belongs to $f^{-1}(0)$ and Bob's to $f^{-1}(1)$.
  The bound $|M|\le2^m$ expresses per-message cost $m$.
\end{definition}

\begin{lemma}[Player swapping]
  \label{lem:lowerbounds-round-swap}
  \lean{Complexity.KarchmerWigderson.RoundProtocol.run_swapPlayers,
    Complexity.KarchmerWigderson.RoundProtocol.SolvesKW.swapPlayers}
  \leanok
  \uses{def:lowerbounds-round-protocol}
  Swapping every speaker exchanges the inputs in the protocol's run.
  Hence a protocol for $f$ gives a protocol for $\neg f$ with the same
  alphabet and round bound.
\end{lemma}
\begin{proof}
  \leanok
  Induct on the protocol tree.
\end{proof}

\begin{definition}[Density rectangle]
  \label{def:lowerbounds-density-rectangle}
  \lean{Complexity.KarchmerWigderson.DensityRectangle}
  \leanok
  \uses{def:analysis-density-limit}
  A $(p,k)$-density rectangle consists of nonempty sets $X,Y$ of deficit
  at most $k$, and some $0<r\le p$ such that every point on one side is an
  $(r,k)$-limit of the other. The actual rate $r$ is retained because
  density-limit conditions need not be monotone in the sampling rate.
\end{definition}

\begin{theorem}[Density-rectangle adversary]
  \label{thm:lowerbounds-density-adversary}
  \lean{Complexity.KarchmerWigderson.RoundProtocol.not_solves_density,
    Complexity.KarchmerWigderson.RoundProtocol.not_solves_bounded_density}
  \leanok
  \uses{def:lowerbounds-round-protocol, def:lowerbounds-density-rectangle,
    lem:analysis-improved-mirror-set}
  Let $|M|\le2^m$. Suppose, for $0\le i<d$,
  \[
    k_i\ge1,\quad k_{i+1}\ge194k_i,\quad
    k_{i+1}\ge2k_i+2+m,\quad p_{i+1}\ge32768k_ip_i,
  \]
  and $p_i\le1/4$ for $0\le i\le d$. No protocol with at most $d$
  messages solves a $(p_0,k_0)$-density rectangle.
  In particular, for $k\ge1$ and $m+1\le k$, the condition
  $(32768\cdot194^d k)^d p\le1/4$ suffices for a $(p,k)$-rectangle.
\end{theorem}
\begin{proof}
  \leanok
  A largest message cell loses at most $m$ in deficit. Before selecting
  the cell, mirror if needed so the current limit condition faces the
  speaking player. The recurrences preserve the density rectangle.
  A density limit at rate less than $3/4$ has a point of the other set
  agreeing with it at any specified coordinate, so no leaf can solve
  the rectangle. Induct on the remaining messages, allowing early
  termination. For the finite specialization take
  $k_i=194^ik$ and $p_i=(32768\cdot194^d k)^i p$.
\end{proof}

\begin{lemma}[Bilateral initial limits]
  \label{lem:lowerbounds-bilateral-adversary}
  \lean{Complexity.KarchmerWigderson.RoundProtocol.not_solves_bilateral_density,
    Complexity.KarchmerWigderson.RoundProtocol.not_solves_bilateral_density_with_deficit}
  \leanok
  \uses{thm:lowerbounds-density-adversary}
  Suppose $k\ge1$, $X,Y$ are nonempty, each has deficit at most $k$, and
  each consists of $(p,k)$-limits of the other, where $p>0$. If $m\ge0$ and
  $(32768\cdot194^t(m+k))^t p\le1/4$, no protocol with at most $t+1$
  messages from an alphabet of size at most $2^m$ solves $X\times Y$.
\end{lemma}
\begin{proof}
  \leanok
  Select the first message cell without mirroring. Both deficits are
  now at most $m+k$, and the retained limit condition faces the selected
  side. Apply the density adversary for the remaining $t$ rounds.
\end{proof}

\begin{theorem}[Explicit finite parity obstruction]
  \label{thm:lowerbounds-parity-topdown-finite}
  \lean{Complexity.KarchmerWigderson.RoundProtocol.not_solves_parity_finite}
  \leanok
  \uses{lem:lowerbounds-bilateral-adversary, def:lowerbounds-parity}
  If $m\ge0$ and
  $16(32768\cdot194^t(m+1))^t\le n$, no protocol with at most $t+1$
  messages, each from an alphabet of size at most $2^m$, solves the
  general KW game for $n$-bit parity.
\end{theorem}
\begin{proof}
  \leanok
  The two parity classes have deficit one. If a resampled coordinate
  set is nonempty, flipping any selected coordinate pairs the two
  parities, so each parity fiber has density $1/2$. At rate $4/n$ the
  empty-set probability is at most $1/5$, using
  $(1-p)^n(1+np)\le1$. Thus the initial limits are bilateral.
  The size condition implies $n\ge16$ and the required rate budget.
\end{proof}

\begin{theorem}[Circuits give bounded-round protocols]
  \label{thm:lowerbounds-circuit-round-protocol}
  \lean{Complexity.Circuit.exists_roundProtocol}
  \leanok
  \uses{def:circuit, def:circuits-bases, def:lowerbounds-round-protocol,
    lem:lowerbounds-round-swap, lem:circuits-dependency-graph}
  For each output of an unbounded-fan-in AND/OR circuit with free
  input negations, depth at most $d$ gives a general KW protocol with
  at most $d$ messages from an alphabet of size $\mathrm{totalFanIn}$.
  This is the circuit-to-protocol direction of the Karchmer--Wigderson
  correspondence in Korten's Lemma~1. The wire count includes every
  gate-input occurrence, including output gates and repeated wires.
\end{theorem}
\begin{proof}
  \leanok
  Descend through a disagreeing gate input. For OR, Bob selects a true
  child; for AND, Alice selects a false child. Negated inputs swap the
  players. Each gate's fan-in is at most the total wire count. Empty
  gates have an empty KW relation, so an arbitrary coordinate suffices.
\end{proof}

\begin{theorem}[Korten's top-down lower bound, Theorem 3]
  \label{thm:lowerbounds-korten-topdown}
  \lean{Complexity.KarchmerWigderson.parity_communication_lower_bound,
    Complexity.Circuit.parity_wire_lower_bound}
  \leanok
  \uses{thm:lowerbounds-parity-topdown-finite,
    thm:lowerbounds-circuit-round-protocol}
  For every $d\ge2$ there is $\varepsilon_d>0$ such that, for all sufficiently
  large $n$, the general Karchmer--Wigderson game for $n$-bit parity has no
  deterministic protocol of at most $d$ rounds with at most
  $\varepsilon_d n^{1/(d-1)}$ bits in each message. Consequently, a depth-$d$
  unbounded-fan-in De Morgan circuit for parity has more than
  $2^{\varepsilon_d n^{1/(d-1)}}$ wires.
  The analytic prerequisites of this top-down proof are in
  Section~\ref{sec:analysis-harmonic}.
\end{theorem}
\begin{proof}
  \leanok
  Put $t=d-1$, $C=32768\cdot194^t$, $\varepsilon_d=(32C)^{-1}$, and
  $N_d=(32C)^t$. If $n\ge N_d$ and $m\le\varepsilon_d n^{1/t}$,
  then $C(m+1)\le n^{1/t}/16$, so $16(C(m+1))^t\le n$.
  Apply the finite parity obstruction. For the wire conclusion, a
  circuit with at most $2^{\varepsilon_d n^{1/t}}$ wires would give a
  protocol contradicting the communication conclusion.
\end{proof}

\subsection{Majority extension}

The following extends the formalized argument to strict majority. Korten's
Theorem~3 states the parity case; the adjacent-layer initialization below
allows the same adversary to prove the majority bound.

\begin{definition}[Majority boundary layers and deficit bound]
  \label{def:lowerbounds-majority-layers}
  \lean{Complexity.BooleanAnalysis.weightLayer,
    Complexity.BooleanAnalysis.majorityDeficitBound}
  \leanok
  \uses{def:encodings-majority}
  Let $L_{n,r}=\{x\in\{0,1\}^n:|x|=r\}$, where $|x|$ is Hamming weight.
  The boundary layers of strict majority are
  $X=L_{n,\lfloor n/2\rfloor}$ and $Y=L_{n,\lfloor n/2\rfloor+1}$.
  Put $K(n)=128+\log_2(2(n+1))$. A tie belongs to the false side.
\end{definition}

\begin{lemma}[Many neighbors give a sparse density limit]
  \label{lem:lowerbounds-neighbor-density-limit}
  \lean{Complexity.BooleanAnalysis.isDensityLimit_of_many_neighbors}
  \leanok
  \uses{def:analysis-density-limit}
  Suppose $n\ge16$ and at least half the coordinates of $x\in\{0,1\}^n$
  yield a point of $Z$ when flipped. Then $x$ is a $(16/n,128)$-limit of $Z$.
\end{lemma}
\begin{proof}
  \leanok
  Sample each coordinate with probability $16/n$. The probability of
  hitting a valid flip is at least $8/9$, using $(1-p)^s(1+sp)\le1$.
  The expected mask size is $16$, so the chance it exceeds $128$ is at
  most $1/8$. With probability at least $55/72\ge3/4$, a valid flip is
  selected and the mask has at most $128$ coordinates. That single
  completion gives fiber density at least $2^{-128}$.
\end{proof}

\begin{lemma}[Bilateral majority initialization]
  \label{lem:lowerbounds-majority-initialization}
  \lean{Complexity.BooleanAnalysis.majority_layers_deficit,
    Complexity.BooleanAnalysis.majority_layers_limits}
  \leanok
  \uses{def:lowerbounds-majority-layers, lem:lowerbounds-neighbor-density-limit}
  For $n\ge16$, both boundary layers are nonempty and have deficit at most
  $K(n)$. Each consists of $(16/n,K(n))$-limits of the other.
\end{lemma}
\begin{proof}
  \leanok
  The middle binomial coefficient is at least $2^n/(n+1)$, and its upper
  neighbor is at least half as large. Thus both layers have deficit at
  most $\log_2(2(n+1))$. Each point of $X$ has at least $n/2$ zero bits,
  and each point of $Y$ has at least $n/2$ one bits. Flipping any of
  these bits reaches the other layer. Apply the neighbor lemma and
  weaken its deficit parameter from $128$ to $K(n)$.
\end{proof}

\begin{theorem}[Explicit finite majority obstruction]
  \label{thm:lowerbounds-majority-topdown-finite}
  \lean{Complexity.KarchmerWigderson.RoundProtocol.not_solves_majority_finite}
  \leanok
  \uses{lem:lowerbounds-bilateral-adversary, lem:lowerbounds-majority-initialization}
  If $m\ge0$ and
  $64(32768\cdot194^t(m+K(n)))^t\le n$, no protocol with at most $t+1$
  messages, each from an alphabet of size at most $2^m$, solves the
  general KW game for $n$-bit strict majority.
\end{theorem}
\begin{proof}
  \leanok
  The size condition implies $n\ge64$ and gives the rate budget in the
  bilateral adversary at $p=16/n$ and $k=K(n)$. Restrict a putative
  majority protocol to the two boundary layers to obtain a contradiction.
\end{proof}

\begin{theorem}[Top-down majority lower bounds]
  \label{thm:lowerbounds-majority-topdown}
  \lean{Complexity.KarchmerWigderson.majority_communication_lower_bound,
    Complexity.Circuit.majority_wire_lower_bound}
  \leanok
  \uses{thm:lowerbounds-majority-topdown-finite,
    thm:lowerbounds-circuit-round-protocol}
  For every $d\ge2$ there is $\varepsilon_d>0$ such that, for all sufficiently
  large $n$, the general KW game for strict majority has no deterministic
  protocol of at most $d$ rounds with at most $\varepsilon_d n^{1/(d-1)}$
  bits per message. Consequently, every depth-at-most-$d$ unbounded-fan-in
  De Morgan circuit for strict majority has more than
  $2^{\varepsilon_d n^{1/(d-1)}}$ wires. This covers both even and odd $n$.
\end{theorem}
\begin{proof}
  \leanok
  Put $t=d-1$, $C=32768\cdot194^t$, and $\varepsilon_d=(128C)^{-1}$.
  Since $K(n)=o(n^{1/t})$, eventually $K(n)\le\varepsilon_d n^{1/t}$.
  If also $m\le\varepsilon_d n^{1/t}$, then
  $C(m+K(n))\le n^{1/t}/64$, implying the finite obstruction.
  Apply the circuit-to-protocol theorem for the wire conclusion.
\end{proof}
