\chapter{Average-case complexity and the Hirahara program}
\label{chap:averagecase}

This chapter formalizes average-case complexity through explicit finite
ensembles and errorless heuristics, and follows Hirahara's metacomplexity route
toward excluding Impagliazzo's Heuristica, that is, toward proving $\cc{P} =
\cc{NP}$ if and only if $\cc{DistNP} \subseteq \cc{AvgP}$. Formalized so far:
finite and dyadic ensembles, errorless heuristics and $\cc{AvgP}$, Hirahara's
auxiliary-unary distribution with exact $\cc{MINKT}$ probabilities, the
dense-random-string extraction lemma, a finite Nisan--Wigderson reconstruction
carried through list decoding to a Kolmogorov upper bound, conditional
complexity and computational depth, symmetry of information as a named
hypothesis, and a finite conditional theorem deriving $\cc{P} = \cc{NP}$ from
symmetry of information together with $\cc{NP}$-hardness of a conditional gap
problem. The symmetry-of-information hypothesis is machine-relative and holds
trivially for machines that describe nothing, so it has content only together
with the other hypotheses of the theorems that use it
(Definition~\ref{def:avg-soi}). The reconstruction results assume a list-decodable code
family with polynomial parameter bounds, a machine realizing the decoder, and,
in the oracle form, an efficiently universal oracle machine; none of these is
instantiated in the library. Samplable ensembles and $\cc{DistNP}$ are not yet
defined, so none of the headline equivalences is stated yet; they are the main
open direction.

\section{Ensembles and errorless heuristics}

\begin{definition}[Finite and dyadic ensembles]
  \label{def:avg-ensemble}
  \lean{Complexity.FiniteEnsemble, Complexity.DyadicEnsemble,
    Complexity.DyadicEnsemble.toFinite,
    Complexity.DyadicEnsemble.probability_toFinite}
  \leanok
  A finite ensemble assigns to each parameter $n$ a nonempty finite seed space
  $S_n$ (with decidable equality) and a sample map $S_n \to \alpha$; the
  distribution at $n$ is the pushforward of the uniform distribution on $S_n$,
  with multiplicity. A dyadic
  ensemble is the special case $S_n = \{0,1\}^{\ell(n)}$, and its embedding into
  finite ensembles preserves every event probability. No sample map is called
  efficient until a machine realizes it (Definition~\ref{def:avg-distnp}).
  Samples need not have length $n$.
\end{definition}

\begin{lemma}[Independent repetition]
  \label{lem:avg-repetition}
  \lean{Complexity.uniformAtLeastOneProbability_eq_one_sub_pow,
    Complexity.trials_mul_div_one_add_le_uniformAtLeastOneProbability,
    Complexity.half_le_uniformAtLeastOneProbability_of_singleDrawLower}
  \leanok
  If an event has probability $p$ under the uniform distribution on a nonempty
  finite sample space, the probability that at least one of $k$ independent
  uniform draws hits it is exactly $1 - (1-p)^k$, and it is at least
  $kp/(1+kp)$; in particular it is at least $1/2$ whenever $kp' \ge 1$ for
  some lower bound $p' \le p$.
\end{lemma}
\begin{proof}
  \leanok
  Count the complementary product event; then use $1 - (1-p)^k \ge kp/(1+kp)$,
  which is at least $1/2$ once $kp \ge 1$.
\end{proof}

\begin{definition}[Errorless heuristics and $\cc{AvgP}$]
  \label{def:avg-avgp}
  \lean{Complexity.HeuristicAnswer, Complexity.HeuristicAlgorithm.IsErrorlessFor,
    Complexity.HeuristicAlgorithm.IsPolynomialTime,
    Complexity.HeuristicAlgorithm.FailsWithProbabilityAtMost,
    Complexity.DistributionalProblem, Complexity.AvgPAt,
    Complexity.inversePolynomialFailure, Complexity.AvgP}
  \leanok
  \uses{def:avg-ensemble, def:language, def:FP}
  A heuristic is a deterministic map from strings to the three answers accept,
  reject and fail; it is errorless for $L$ if it never accepts outside $L$ and
  never rejects inside $L$. It runs in polynomial time when its answer, encoded
  as $[\mathrm{true}]$, $[\mathrm{false}]$ or the empty string respectively, is
  a function in $\cc{FP}$. A distributional problem is a language with a finite
  ensemble of input strings. For $\delta \colon \mathbb{N} \to \mathbb{Q}$,
  $\cc{AvgP}_\delta$ consists of the distributional problems having a
  polynomial-time errorless heuristic whose failure probability on slice $n$ is
  at most $\delta(n)$ for every $n$. $\cc{AvgP} = \bigcap_{c \in \mathbb{N}}
  \cc{AvgP}_{\delta_c}$ with $\delta_c(n) = 1/\max(1,n)^c$, which is $n^{-c}$
  for $n \ge 1$ and $1$ at $n = 0$: for each $c$ there is a possibly different
  polynomial-time errorless heuristic meeting $\delta_c$. This is not the
  single-scheme formulation in which the heuristic receives $1/\delta$ as an
  input; relating the two is part of Definition~\ref{def:avg-heurp}.
\end{definition}

\begin{proposition}[Basic facts about errorless heuristics]
  \label{prop:avg-avgp-basic}
  \lean{Complexity.DistributionalProblem.mem_AvgPAt_of_decision,
    Complexity.DistributionalProblem.mem_AvgP_of_decision,
    Complexity.HeuristicAlgorithm.IsErrorlessFor.one_sub_languageProbability_sub_failure_le_reject}
  \leanok
  \uses{def:avg-avgp}
  (a) If a Boolean function $d$ satisfies $d(x) = \mathrm{true}$ exactly when
  $x$ lies in the language of a distributional problem, and the one-bit
  function $x \mapsto [d(x)]$ is in $\cc{FP}$, then the problem lies in
  $\cc{AvgP}_\delta$ for every nonnegative $\delta$ and in $\cc{AvgP}$, whatever
  its ensemble. (b) An errorless heuristic for $L$ (efficient or not) rejects
  with probability at least $1 - \Pr[L] - \Pr[\mathrm{fail}]$ on every slice of
  every finite ensemble.
\end{proposition}
\begin{proof}
  \leanok
  (a) Use the never-failing heuristic. (b) Errorlessness makes rejection
  exactly the complement of $L$ minus failures.
\end{proof}

\begin{definition}[Samplable ensembles and $\cc{DistNP}$]
  \label{def:avg-distnp}
  \uses{def:avg-ensemble, def:avg-avgp, def:NP, def:FP}
  An ensemble is polynomial-time samplable when a single machine computes its
  sample map from a polynomially long seed in time polynomial in $n$, with
  closure under polynomial-time maps, pairing, padding, products and
  efficiently chosen mixtures. $\cc{DistNP}$ consists of pairs $(L, D)$ with
  $L \in \cc{NP}$ and $D$ polynomial-time samplable. Levin-style domination is
  kept as a separate notion if needed.
\end{definition}

\begin{definition}[Error-prone heuristics and $\cc{HeurP}$]
  \label{def:avg-heurp}
  \uses{def:avg-avgp}
  Heuristics that may answer incorrectly on a set of bounded probability,
  $\cc{Heur}_\delta\cc{P}$ and $\cc{HeurP}$, the single-scheme versions of
  $\cc{AvgP}$ and $\cc{HeurP}$, randomized variants, and the standard
  amplification and inclusion lemmas.
\end{definition}

\begin{theorem}[The easy direction]
  \label{thm:avg-easy-direction}
  \uses{def:avg-distnp, def:avg-avgp, def:P, def:NP}
  \notready
  If $\cc{P} = \cc{NP}$, then $\cc{DistNP} \subseteq \cc{AvgP}$.
\end{theorem}
\begin{proof}
  \uses{prop:avg-avgp-basic}
  Promote a $\cc{P}$ decider to a one-bit decision function in $\cc{FP}$
  (Lemma~\ref{lem:polytime-decision-fn}) and apply Proposition~\ref{prop:avg-avgp-basic}(a). What is missing is the
  definition of $\cc{DistNP}$ needed to state the theorem.
\end{proof}

\begin{theorem}[Excluding Heuristica]
  \label{thm:avg-heuristica}
  \uses{def:avg-distnp, def:avg-avgp, def:P, def:NP}
  \notready
  $\cc{P} = \cc{NP}$ if and only if $\cc{DistNP} \subseteq \cc{AvgP}$. This is
  an open problem; the nontrivial direction is $\cc{P} \neq \cc{NP} \Rightarrow
  \cc{DistNP} \not\subseteq \cc{AvgP}$. The conditional results of this chapter
  isolate hypotheses that would imply it.
\end{theorem}
\begin{proof}
  \uses{thm:avg-easy-direction}
  One direction is Theorem~\ref{thm:avg-easy-direction}.
\end{proof}

\section{MINKT under the auxiliary-unary distribution}

\begin{definition}[Auxiliary-unary ensemble]
  \label{def:avg-auxiliary-unary}
  \lean{Complexity.FiniteEnsemble.auxiliaryUnary, Complexity.AuxiliaryUnarySeed.card}
  \leanok
  \uses{def:avg-ensemble, def:pair}
  Hirahara's 2018 distribution $D^u$: at parameter $m > 0$, choose
  $n \in \{0, \dots, m-1\}$ uniformly and $x \in \{0,1\}^n$ uniformly, and output
  $\mathrm{pair}(x, 1^{m-n})$. The library realizes this with the seed space
  $\{0, \dots, m-1\} \times \{0,1\}^m$, keeping the first $n$ bits of the
  $m$-bit string as $x$, so the slice has $m \cdot 2^m$ equiprobable seeds. The
  parameter $m$ is the sum $|x| + (m-n)$ of the component lengths, not the
  length of the encoded pair. The slice $m = 0$ is totalized as the point
  $\mathrm{pair}(\epsilon, \epsilon)$ (a single seed), a library convention.
\end{definition}

\begin{lemma}[$\cc{MINKT}$ mass under $D^u$]
  \label{lem:avg-auxiliary-unary-minkt}
  \lean{Complexity.FiniteEnsemble.mass_auxiliaryUnary_pair,
    Complexity.FiniteEnsemble.probability_auxiliaryUnary_minkt_eq_average,
    Complexity.FiniteEnsemble.probability_auxiliaryUnary_minkt_le_average_incompressibility}
  \leanok
  \uses{def:avg-auxiliary-unary, def:meta-minkt}
  For $0 < m$ and $|x| = n < m$, $D^u_m$ gives $\mathrm{pair}(x, 1^{m-n})$ mass
  $1/(m \cdot 2^n)$. For every machine $U$ and threshold $r$,
  \[
    \Pr_{D^u_m}[\cc{MINKT}(U,r)] = \frac{1}{m}\sum_{n<m}
      \Pr_{x \in \{0,1\}^n}\bigl[C^{m-n}_U(x) < r(n)\bigr]
    \le \frac{1}{m}\sum_{n<m} \frac{2^{r(n)}-1}{2^n}.
  \]
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:meta-incompressibility}
  Condition on the split length; retaining the $n$-bit prefix of a uniform
  $m$-bit seed preserves prefix-event probabilities. Then apply finite
  incompressibility at each length.
\end{proof}

\begin{theorem}[Dense-random-string extraction]
  \label{thm:avg-dense-extraction}
  \lean{Complexity.MINKT.auxiliaryUnary_rejectProbability_ge_average}
  \leanok
  \uses{def:avg-auxiliary-unary, def:avg-avgp, def:meta-minkt}
  For every $m > 0$, every machine $U$, every threshold $r$, and every errorless
  heuristic $A$ for $\cc{MINKT}(U, r)$ (no efficiency is assumed),
  \[
    \Pr_{D^u_m}[A \text{ rejects}] \ \ge\ 1 - \frac{1}{m}\sum_{n<m}
      \frac{2^{r(n)}-1}{2^n} - \Pr_{D^u_m}[A \text{ fails}].
  \]
  Thus, when the average and the failure probability are small, an errorless
  heuristic must certify, by rejecting, a dense set of strings of high
  time-bounded complexity.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:avg-auxiliary-unary-minkt, prop:avg-avgp-basic}
  Combine Proposition~\ref{prop:avg-avgp-basic}(b) with
  Lemma~\ref{lem:avg-auxiliary-unary-minkt}.
\end{proof}

\section{Nisan--Wigderson reconstruction}

\begin{definition}[Generators and time-bounded random tests]
  \label{def:avg-bitgenerator}
  \lean{Complexity.BitGenerator, Complexity.BitGenerator.distinguishingAdvantage,
    Complexity.BitGenerator.IsTimeBoundedRandomTest,
    Complexity.BitGenerator.IsDenseTest,
    Complexity.BitGenerator.HasLowTimeBoundedComplexity}
  \leanok
  \uses{def:kolmogorov}
  A generator is any map $G \colon \{0,1\}^d \to \{0,1\}^m$ (no efficiency is
  built in). A test $T \subseteq \{0,1\}^m$ has distinguishing advantage
  $|\Pr_s[G(s) \in T] - \Pr_y[y \in T]|$. Relative to a machine $M$, clock $t$
  and threshold $r$: $T$ is a random test if every $y \in T$ has $C^t_M(y) \ge
  r$; $T$ is $\delta$-dense if $\Pr_y[y \in T] \ge \delta$; and $G$ has low
  complexity if every output satisfies $C^t_M(G(s)) < r$.
\end{definition}

\begin{lemma}[Statistical test, hybrid, and prediction steps]
  \label{lem:avg-test-hybrid}
  \lean{Complexity.BitGenerator.density_le_distinguishingAdvantage_of_randomTest,
    Complexity.BitGenerator.exists_oriented_hybridGap_ge_average,
    Complexity.NextBitPrediction.successProbability_eq_half_add_gap}
  \leanok
  \uses{def:avg-bitgenerator}
  (a) A $\delta$-dense random test has advantage at least $\delta$ against every
  low-complexity generator. (b) If $m > 0$ and the advantage is at least
  $\delta$, then for $T$ or its complement there is $i < m$ such that the Yao
  hybrid whose first $i+1$ coordinates come from $G$ (the rest uniform) is
  accepted with probability at least $\delta/m$ more than the hybrid whose
  first $i$ coordinates do. (c) Over any
  nonempty finite background space, the predictor that outputs a uniform
  candidate bit when the test accepts and its complement otherwise succeeds with
  probability exactly $1/2$ plus the acceptance gain from substituting the true
  bit for the candidate.
\end{lemma}
\begin{proof}
  \leanok
  (a) No generator output lies in $T$. (b) Telescope the hybrid gaps and
  average. (c) Expand the two candidate fibers.
\end{proof}

\begin{definition}[Nisan--Wigderson designs and generator]
  \label{def:avg-nw-design}
  \lean{Complexity.NWDesign, Complexity.NWDesign.HasOverlapBudget,
    Complexity.NWDesign.generator}
  \leanok
  \uses{def:avg-bitgenerator}
  An NW design is an ordered family $S_1, \dots, S_m$ of $\ell$-element subsets
  of $[d]$, each enumerated injectively. It has overlap budget $b$ if for every
  $i$, $\sum_{j<i} 2^{|S_i \cap S_j|} + (m - i) \le b$ (Hirahara's weak-design
  cost). For $f \colon \{0,1\}^\ell \to \{0,1\}$ the generator is
  $\mathrm{NW}^f(s) = (f(s|_{S_1}), \dots, f(s|_{S_m}))$.
\end{definition}

\begin{theorem}[Finite NW reconstruction]
  \label{thm:avg-nw-reconstruction}
  \lean{Complexity.NWDesign.exists_fixed_reconstruction_of_randomTest,
    Complexity.NWDesign.half_le_checkedReconstructionBatch_of_randomTest}
  \leanok
  \uses{def:avg-nw-design, def:avg-bitgenerator}
  Let $m > 0$, let $\mathrm{NW}^f$ have low complexity, and let $T$ be a
  $\delta$-dense random test for the same machine, clock and threshold, and let
  the design have overlap budget $b$. (a) There are a polarity, a coordinate $i$
  and fixed advice whose predictor agrees with $f$ on at least a $1/2 +
  \delta/m$ fraction of inputs, and whose stored data (predecessor tables,
  later output bits, outside seed bits and the candidate bit, excluding codec
  overhead) has at most $b + (d - \ell) + 1$ bits.
  (b) If $\delta > 0$, exact checking of $\lceil 2m/\delta \rceil$ uniformly
  sampled trials finds a certificate with agreement at least $1/2 +
  \delta/(2m)$ with probability at least $1/2$, and every returned certificate
  satisfies the same payload bound.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:avg-test-hybrid, lem:avg-repetition}
  Hybrid and prediction steps give the predictor; each predecessor block depends
  only on its intersection with $S_i$ and is hardwired by a table of
  $2^{|S_i \cap S_j|}$ entries; fiber averaging fixes the outside seed and later
  tail. A reverse-Markov bound gives single-trial success at least
  $\delta/(2m)$, and repetition gives (b).
\end{proof}

\begin{theorem}[NW reconstruction yields short descriptions]
  \label{thm:avg-nw-kolmogorov}
  \lean{Complexity.NWDesign.half_le_timeBoundedKolmogorovComplexity_of_inverseDensity,
    Complexity.NWDesign.half_le_efficientlyUniversalKolmogorovComplexity_of_inverseDensity,
    Complexity.NWDesign.half_le_oracleTimeBoundedKolmogorovComplexity_of_inverseDensity,
    Complexity.NWDesign.half_le_efficientlyUniversalOracleKolmogorovComplexity_of_inverseDensity,
    Complexity.NWDesign.UniformEncodedMessageDecoderRealization.half_le_efficientlyUniversalKolmogorovComplexity_of_inverseDensity,
    Complexity.NWDesign.UniformOracleEncodedMessageDecoderRealization.half_le_efficientlyUniversalKolmogorovComplexity_of_inverseDensity}
  \leanok
  \uses{thm:avg-nw-reconstruction, def:meta-list-code-family, def:universal,
    def:avg-conditional}
  Assume: a list-code family that is list-decodable at inverse accuracy (up to
  relative distance $1/2 - 1/q$ at every inverse accuracy $q \ge 2$) and has
  polynomial parameter bounds; $m > 0$ and $D > 0$; an NW design over the
  coordinates of the code at inverse accuracy $q = 2mD$ with overlap budget $b$
  and seed length $d$; a message $\mu$ such that $\mathrm{NW}^{\mathrm{Enc}(\mu)}$
  has low complexity for $(M, t, r)$; and a $(1/D)$-dense random test $T$ for
  $(M, t, r)$. Let $B = 1 + \lceil\log_2 m\rceil + (b + (d - \ell) + 1) +
  \lceil \log_2 (C_L (q+1)^{e_L}) \rceil$, where $\ell$ is the coordinate length
  and $C_L (q+1)^{e_L}$ the list-size bound. In each case below, checked
  sampling with $\lceil 2mD \rceil$ trials at agreement threshold
  $1/2 + 1/(2mD)$ succeeds with probability at least $1/2$, and the stated
  bound on $\mu$ holds for every returned certificate.
  (a) If a machine $\mathrm{Dec}$ with monotone clock $t_{\mathrm{dec}}$
  realizes the indexed-message decoder for this design, code and test, then
  $C^{t_{\mathrm{dec}}(B)}_{\mathrm{Dec}}(\mu) \le B$; if moreover $U$ is an
  efficiently universal machine, there are constants $c, a, k$ with
  $C^{a(B + t_{\mathrm{dec}}(B) + 1)^k}_U(\mu) \le B + c$.
  (b) If instead one oracle machine $\mathrm{Dec}$ with monotone clock
  $t_{\mathrm{dec}}$ realizes the decoder for this design and code and every
  finite test given as a membership oracle, the same two bounds hold for
  oracle complexity relative to the membership oracle of $T$, now with $U$ an
  efficiently universal oracle machine. Moreover the returned $c, a, k$ satisfy
  a transfer law for every finite test $T'$ and message $\mu'$: if some
  description of length at most $b'$ decodes to $\mu'$ under $T'$, then
  $C^{a(b' + t_{\mathrm{dec}}(b') + 1)^k,\, T'}_U(\mu') \le b' + c$.
  (c) If a single decoder machine serves every parameter choice of the family,
  reading the design and test from an explicit encoding $e$ framed with the
  description (or, in the oracle form, the numeric parameters and design from
  a canonical encoding $e$, with the test as an oracle), then for efficiently
  universal $U$ one triple $c, a, k$, chosen before all parameters, designs,
  tests and messages, gives the bound of (a) (respectively of (b)) on every
  instance satisfying the assumptions above, with $B$ replaced by
  $2|e| + 2 + B$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:meta-list-decoding, thm:meta-kolmogorov-invariance}
  List-decode the predictor stored in the certificate, append a
  $\lceil \log_2 L \rceil$-bit index of the correct candidate, the polarity bit
  and the coordinate, and transfer the resulting description through the
  universal compiler. In (a) and the ordinary form of (c) the universality
  hypothesis can be discharged by $\mathtt{utmTM}$; the code family, the
  decoder realization and (in (b) and the oracle form of (c)) the universal
  oracle machine remain hypotheses.
\end{proof}

\begin{lemma}[Explicit weak designs]
  \label{lem:avg-rrv-design}
  \uses{def:avg-nw-design}
  For every $m$, $\ell$ and $\rho > 1$ there is a polynomial-time computable NW
  design with $\sum_{j<i} 2^{|S_i \cap S_j|} \le \rho(m-1)$ for every $i$ and
  seed length $d = \lceil \ell/\ln \rho \rceil \cdot \ell$
  (Raz--Reingold--Vadhan weak designs); its overlap budget is at most
  $\rho(m-1) + m$.
\end{lemma}

\begin{theorem}[An efficiently universal oracle machine]
  \label{thm:avg-oracle-utm}
  \uses{def:universal, thm:utm}
  There is an oracle machine that is efficiently universal for oracle machines,
  analogous to $\mathtt{utmTM}$.
\end{theorem}

\begin{theorem}[Concrete reconstruction with square-root loss]
  \label{thm:avg-nw-concrete}
  \uses{thm:avg-nw-kolmogorov, lem:avg-rrv-design,
    thm:meta-explicit-code-family, thm:avg-oracle-utm, thm:utm}
  Instantiating Theorem~\ref{thm:avg-nw-kolmogorov} with an explicit code
  family, explicit weak designs and concrete universal machines, and realizing
  the design parser, sampler, checker and decoder by machines, yields
  Hirahara's 2018 search step: a dense random test gives a randomized algorithm
  that on input $(x, 1^t)$ finds a description of $x$ whose length exceeds
  $C^t(x)$ by $O(\sqrt{C^t(x)})$ up to logarithmic factors, running within a
  polynomially larger clock. The exact finite parameter theorem is proved first
  and the asymptotic loss derived from it.
\end{theorem}

\section{Conditional complexity, depth, and symmetry of information}

\begin{definition}[Oracle and random-access conditional complexity]
  \label{def:avg-conditional}
  \lean{Complexity.OracleTM.timeBoundedKolmogorovComplexity,
    Complexity.RandomAccessCondition.oracle,
    Complexity.OracleTM.randomAccessConditionalTimeBoundedKolmogorovComplexity}
  \leanok
  \uses{def:kolmogorov}
  For an oracle machine $N$ and Boolean oracle $A$, $C^{t,A}_N(x)$ is the
  minimum length of a program on which $N^A$ produces $x$ within $t$ steps
  ($\infty$ if there is none). The random-access oracle of a string $y$ answers
  canonical tagged queries ``bit $i$ of $y$'' and ``is $i < |y|$'' (malformed
  and out-of-range queries answer false), so it determines both the contents
  and the length of $y$. Conditional complexity is
  $C^t_N(x \mid y) = C^{t, \mathrm{oracle}(y)}_N(x)$. Other conditioning
  conventions (sequential input, paper-specific evaluators) are not identified
  with this one.
\end{definition}

\begin{lemma}[Basic laws of oracle complexity]
  \label{lem:avg-oracle-kolmogorov}
  \lean{Complexity.OracleTM.timeBoundedKolmogorovComplexity_mono,
    Complexity.TM.toOracleTM_timeBoundedKolmogorovComplexity_eq,
    Complexity.OracleTM.SimulatesInTime.kolmogorov_transfer}
  \leanok
  \uses{def:avg-conditional, def:barriers-to-oracle,
    def:barriers-oracle-universal}
  Fix an oracle machine $N$ and an oracle $A$. (a) If $t \le t'$ then
  $C^{t',A}_N(x) \le C^{t,A}_N(x)$. (b) For an ordinary machine $M$ viewed as
  an oracle machine that never queries, $C^{t,A}_M(x) = C^t_M(x)$ for every
  oracle $A$. (c) Suppose $N$ simulates an oracle machine $N'$ in time $\tau$
  under a compiler $c$ (bounded productions transfer for every oracle; bounded
  halting is not required), with $|c(p)| \le |p| + C$ and $\tau(p, t) \le a
  (|p| + t + 1)^e$ for all $p, t$. Then there are $a', e'$, chosen once for
  all oracles, such that for every oracle $A$, string $x$, clock $t$ and bound
  $b$, $C^{t,A}_{N'}(x) \le b$ implies
  $C^{a'(b + t + 1)^{e'},A}_N(x) \le b + C$.
\end{lemma}
\begin{proof}
  \leanok
  (a) A program producing $x$ within $t$ steps also does within $t'$. (b) The
  embedded machine's runs do not depend on the oracle. (c) Compile a
  witnessing program of length at most $b$ and bound the simulation clock using
  that length.
\end{proof}

\begin{definition}[Computational depth]
  \label{def:avg-depth}
  \lean{Complexity.descriptionDifference, Complexity.TM.computationalDepthBetween,
    Complexity.TM.computationalDepth}
  \leanok
  \uses{def:kolmogorov}
  $\mathrm{cd}^{s,t}_M(x) = C^s_M(x) - C^t_M(x)$ and
  $\mathrm{cd}^{s}_M(x) = C^s_M(x) - C_M(x)$, where the difference is $\infty$
  if either term is $\infty$ and truncated natural subtraction otherwise. Both
  are defined for all clocks; the second subtraction is always exact, and the
  first is exact when $s \le t$ (then $C^t_M(x) \le C^s_M(x)$), the case in
  which its laws are stated.
\end{definition}

\begin{theorem}[Upper chain rule from a composition contract]
  \label{thm:avg-upper-chain}
  \lean{Complexity.timeBoundedKolmogorovComplexity_pair_le_add_of_conditional_composition}
  \leanok
  \uses{def:avg-conditional, def:pair}
  Suppose a compiler $\kappa$ turns every program for $y$ (on machine $M_1$,
  clock $t_1$, length at most $b_1$) and every conditional program for $x$
  given $y$ (on $N$, clock $t_2$, length at most $b_2$) into a program on which
  $M$ produces $\mathrm{pair}(x, y)$ within $t$ steps, with $|\kappa(p, q)| \le
  |p| + |q| + c$ for all such $p, q$. If $C^{t_1}_{M_1}(y) \le b_1$ and
  $C^{t_2}_N(x \mid y) \le b_2$ (both finite), then
  $C^t_M(\mathrm{pair}(x,y)) \le C^{t_2}_N(x \mid y) + C^{t_1}_{M_1}(y) + c$.
\end{theorem}
\begin{proof}
  \leanok
  Compile attaining programs.
\end{proof}

\begin{definition}[Time-bounded symmetry of information]
  \label{def:avg-soi}
  \lean{Complexity.TimeBoundedSymmetryOfInformation,
    Complexity.IsAdmissibleKolmogorovClock,
    Complexity.PolynomialTimeBoundedSymmetryOfInformation}
  \leanok
  \uses{def:avg-conditional, def:pair}
  For an ordinary machine $M$, a conditional oracle machine $N$, a clock
  transformation $\kappa$ and a loss $\lambda$, $\mathrm{SoI}(M, N, \kappa,
  \lambda)$ asserts, for all $x, y, t$ with $|x| + |y| \le t$,
  \[
    C^{\kappa(t)}_N(x \mid y) + C^{\kappa(t)}_M(y) \ \le\
    C^t_M(\mathrm{pair}(x,y)) + \lambda(t).
  \]
  The polynomial package asks for some $\kappa$ with $\kappa(t) \ge t$ and
  $\kappa(t) \le a(t+1)^k$ for all $t$, and $\lambda(t) = \lfloor \log_2
  \kappa(t) \rfloor + c$ for a constant $c$. The inequality holds trivially
  wherever $C^t_M(\mathrm{pair}(x,y)) = \infty$, in particular whenever $t$ is
  too small for $M$ to print $\mathrm{pair}(x,y)$, and everywhere for a machine
  $M$ that describes nothing; so SoI has content only for machines constrained
  by other hypotheses. (An earlier version also required
  $C^t_M(\mathrm{pair}(x,y)) < \infty$ whenever $|x| + |y| \le t$. That is
  impossible, since a run of $t$ steps writes at most $t$ output bits while
  $|\mathrm{pair}(x,y)| = 2|x| + 2 + |y| > t$ at $t = |x| + |y|$, and the
  requirement was removed.) SoI is a hypothesis; nothing in the library
  asserts it.
\end{definition}

\begin{theorem}[SoI cancels the condition's complexity]
  \label{thm:avg-soi-cancellation}
  \lean{Complexity.TimeBoundedSymmetryOfInformation.conditional_le_of_pair_upper,
    Complexity.TimeBoundedSymmetryOfInformation.conditional_le_of_composition}
  \leanok
  \uses{def:avg-soi, def:avg-depth, thm:avg-upper-chain}
  (a) Assume $\mathrm{SoI}(M, N, \kappa, \lambda)$, $|x| + |y| \le t$, $t' \le
  \kappa(t)$, and a paired upper bound
  $C^t_M(\mathrm{pair}(x,y)) \le C^{t''}_{N'}(x \mid y) + C^{t'}_M(y) + u$
  for some oracle machine $N'$ and clock $t''$. Then
  \[
    C^{\kappa(t)}_N(x \mid y) \ \le\ C^{t''}_{N'}(x \mid y) +
      \mathrm{cd}^{t', \kappa(t)}_M(y) + u + \lambda(t).
  \]
  (b) In particular the same conclusion holds with $u = c$ when the paired
  upper bound is replaced by the hypotheses of
  Theorem~\ref{thm:avg-upper-chain} for joint machine $M$ at clock $t$,
  condition machine $M$ at clock $t'$ and conditional machine $N'$ at clock
  $t''$. Neither part assumes $C^t_M(\mathrm{pair}(x,y)) < \infty$: when
  $C^{\kappa(t)}_M(y) = \infty$, the depth term, and with it the right-hand
  side, is $\infty$.
\end{theorem}
\begin{proof}
  \leanok
  If $C^{\kappa(t)}_M(y) = \infty$ the conclusion is trivial. Otherwise
  substitute the upper bound into the SoI inequality and cancel
  $C^{\kappa(t)}_M(y)$ using the exact additive decomposition of depth; for
  (b), obtain the upper bound from Theorem~\ref{thm:avg-upper-chain}.
\end{proof}

\begin{theorem}[Concrete universal evaluators]

\begin{definition}[Logarithmic $\cc{GapMINKT}$]
  \label{def:avg-gapminkt-log}
  \lean{Complexity.GapMINKT.Logarithmic.Parameters,
    Complexity.GapMINKT.Logarithmic.problem,
    Complexity.GapMINKT.Logarithmic.Estimator.SatisfiesBounds,
    Complexity.GapMINKT.Logarithmic.Estimator.SatisfiesBoundsOn}
  \leanok
  \uses{def:meta-gapminkt}
  Hirahara 2022, Definition~3.3: for a widening clock $\tau(n, t) \ge t$, the
  promise on canonical $(x, 1^t, 1^s)$ with yes side $C^t_U(x) \le s$ and no
  side $C^{\tau}_U(x) > s + \lfloor \log_2 \tau \rfloor$, where $\tau =
  \tau(|x|, t)$. An estimator $B$ satisfies Fact~3.4's sandwich if $B(x, 1^t)
  \le C^t_U(x)$ and $C^{\tau}_U(x) \le B(x, 1^t) + \lfloor \log_2 \tau
  \rfloor$ on every instance $(x, 1^t)$; it satisfies it on a set $E$ of
  instances if these hold for every instance in $E$.
\end{definition}

\begin{theorem}[Fact 3.4: estimators versus gap solvers]
  \label{thm:avg-fact34}
  \lean{Complexity.GapMINKT.Logarithmic.problem_mem_PromiseP_of_estimatorLanguage_mem_P,
    Complexity.GapMINKT.Logarithmic.Efficient.executableEstimator_satisfiesBoundsOn,
    Complexity.GapMINKT.Logarithmic.Efficient.fp_and_satisfiesBoundsOn_lengthWithinTime,
    Complexity.GapMINKT.Logarithmic.Efficient.fp_and_satisfiesBoundsOn_printerClock}
  \leanok
  \uses{def:avg-gapminkt-log, def:promise, def:universal, def:FP}
  Fix a machine $U$ and a widening clock $\tau$. (a) If an estimator satisfies
  the sandwich on every instance and its threshold language
  $\{(x,1^t,1^s) : B(x,1^t) \le s\}$ is in $\cc{P}$, then logarithmic
  $\cc{GapMINKT}$ is in $\cc{PromiseP}$. (b) Conversely, let a Boolean
  function $d$ solve logarithmic $\cc{GapMINKT}$ (accepting yes instances and
  rejecting no instances) with $x \mapsto [d(x)]$ in $\cc{FP}$. The unary
  threshold sweep built from $d$ is in $\cc{FP}$, and its output length
  satisfies the sandwich on every set of instances on which $C^t_U(x) <
  \infty$; in particular on the paper's domain $|x| \le t$ provided
  $C^t_U(x) < \infty$ for every such instance. If $U$ is efficiently universal,
  no finiteness proviso is needed on a printer-clock domain instead: there are
  $a, k$ depending only on $U$ such that, for every widening clock $\tau$ and
  every such $d$, the sweep satisfies the sandwich on every instance with
  $a(2|x|+3)^k \le t$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:meta-universal-kolmogorov}
  (a) Threshold the estimator. (b) Scan $s = 0, \dots, t$ and return the least
  accepted threshold, whose sandwich needs only $C^t_U(x) < \infty$ (then
  $C^t_U(x) \le t$); efficient universality supplies that finiteness above the
  printer clock.
\end{proof}

\begin{definition}[Conditional gap problems]
  \label{def:avg-gapminckt}
  \lean{Complexity.MINCKT.Instance, Complexity.GapMINCKT,
    Complexity.GapMINCKT.Multiplicative.problem,
    Complexity.GapMINCKT.Estimator.SatisfiesBounds}
  \leanok
  \uses{def:avg-conditional, def:avg-depth, def:promise}
  Hirahara 2022, Definitions~6.1 and~6.5. Fix an ordinary machine $M$ and a
  conditional oracle machine $N$. For a widening trivariate clock
  $\tau(|x|, |y|, t) \ge t$, the promise $\cc{GapMINcKT}$ on canonical
  $(x, y, 1^t, 1^s)$ has yes side $C^t_N(x \mid y) + \mathrm{cd}^{t,\tau}_M(y)
  \le s$ and no side $C^{\tau}_N(x \mid y) > s + \lfloor \log_2 \tau \rfloor$.
  The multiplicative version with factor $\phi \colon \mathbb{N} \to
  \mathbb{N}$, $\phi \ge 1$, has the same yes side and no side
  $C^{\tau}_N(x \mid y) > \phi(|x|) \cdot s + \lfloor \log_2 \tau \rfloor$. A
  conditional estimator $B$ satisfies Proposition~6.2's sandwich if $B(x, y,
  1^t) \le C^t_N(x \mid y) + \mathrm{cd}^{t,\tau}_M(y)$ and $C^{\tau}_N(x \mid
  y) \le B(x, y, 1^t) + \lfloor \log_2 \tau \rfloor$ on every instance.
\end{definition}

\begin{theorem}[Proposition 6.2 core and hardness transfer]
  \label{thm:avg-gapminckt-estimator}
  \lean{Complexity.GapMINCKT_mem_PromiseP_of_estimatorLanguage_mem_P,
    Complexity.GapMINCKT.Multiplicative.promiseNPHard_additive,
    Complexity.GapMINCKT.Multiplicative.P_eq_NP_of_hard_of_estimatorLanguage_mem_P}
  \leanok
  \uses{def:avg-gapminckt, def:meta-promise-reduction}
  Fix $M$, $N$ and a widening clock $\tau$. (a) If an estimator $B$ satisfies
  $B \le C^t_N(x \mid y) + \mathrm{cd}^{t,\tau}_M(y)$ and $C^{\tau}_N(x \mid y)
  \le B + \lfloor \log_2 \tau \rfloor$ on every instance and its threshold
  language $\{(x, y, 1^t, 1^s) : B(x, y, 1^t) \le s\}$ is in $\cc{P}$, then
  $\cc{GapMINcKT} \in \cc{PromiseP}$. (b) Promise-$\cc{NP}$-hardness of the
  multiplicative problem with factor $\phi \ge 1$ implies promise-$\cc{NP}$-hardness
  of the additive one with the same clock. (c) Consequently, multiplicative
  hardness together with such an estimator implies $\cc{P} = \cc{NP}$. The
  second half of the sandwich forces $C^{\tau}_N(x \mid y) < \infty$ on every
  instance, a genuine restriction on $\tau$ and $N$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:meta-promise-collapse}
  (a) Threshold the estimator. (b) The identity is a promise reduction, since the
  multiplicative no side is contained in the additive one. (c) Combine with
  Theorem~\ref{thm:meta-promise-collapse}.
\end{proof}

\begin{lemma}[No estimator is correct on every ordinary instance]
  \label{lem:avg-no-full-estimator}
  \lean{Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.not_satisfiesBounds_ordinaryParameters}
  \leanok
  \uses{def:avg-gapminkt-log}
  For every machine $U$ and every function $p \colon \mathbb{N} \to
  \mathbb{N}$, take the ordinary parameters of the iterated plan, whose
  transformed clock is $\tau(n, t) = p(t)$ regardless of output length. No
  estimator satisfies Fact~3.4's sandwich for $U$ and $\tau$ on every instance:
  at source time $0$, no machine prints a string of length $p(0) + 1$ within
  $p(0)$ steps. The estimator hypotheses below are therefore restricted to the
  plan's own queries.
\end{lemma}
\begin{proof}
  \leanok
  The output cell at position $p(0) + 1$ is untouched by any run of at most
  $p(0)$ steps, so it stays blank.
\end{proof}

\begin{theorem}[Conditional collapse from symmetry of information]
  \label{thm:avg-soi-collapse}
  \lean{Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.Slack.P_eq_NP_of_multiplicative_hard_of_SoI,
    Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.Slack.P_eq_NP_of_multiplicative_hard_of_SoI_of_implementations,
    Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.Slack.P_eq_NP_of_multiplicative_hard_of_SoI_of_logarithmic_solver,
    Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.Slack.not_timeBoundedSymmetryOfInformation_of_P_ne_NP,
    Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.Slack.not_timeBoundedSymmetryOfInformation_of_P_ne_NP_of_implementations,
    Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.Slack.not_timeBoundedSymmetryOfInformation_of_P_ne_NP_of_logarithmic_solver,
    Complexity.GapMINCKT.DifferenceEstimator.Unconditional.Iterated.Slack.IsAdmissibleClock.parameters_admissible}
  \leanok
  \uses{def:avg-soi, def:avg-gapminckt, def:avg-gapminkt-log,
    def:meta-promise-reduction}
  Fix an ordinary machine $M$, a conditional oracle machine $N$, a monotone
  primitive clock $p$ with $p(t) \ge t$ and $p(t) \le a(t+1)^k$ for all $t$,
  and constants $c_{\mathrm{SoI}}, c_{\mathrm{pair}}$. The plan queries the
  ordinary estimator on $(\mathrm{pair}(x,y), 1^{p(t')})$ and on
  $(y, 1^{p^3(t')})$, where $t' = \max(t, |x|+|y|)$. From $p$,
  $c_{\mathrm{SoI}}$ and $c_{\mathrm{pair}}$ the library builds an explicit
  slack-amplified clock $\tau(|x|, |y|, t) = 2^{e}\,(p^4(t+|x|+|y|)+1)
  (t+|x|+|y|+1)$, whose exponent $e$ collects the logarithmic losses and the
  two constants, and proves it widening and polynomially bounded. Assume:
  (i) a pair-composition hypothesis: a compiler producing $\mathrm{pair}(x,y)$ on
  $M$ within $p(t')$ steps from a program for $y$ on $M$ and a conditional
  program for $x$ given $y$ on $N$ (each running within $t$ steps), with length
  loss $c_{\mathrm{pair}}$ and program budgets that contain $C^t_M(y)$ and
  $C^t_N(x \mid y)$ whenever these minima are finite;
  (ii) $\mathrm{SoI}(M, N, p, \lambda)$ with $\lambda(t) = \lfloor \log_2 p(t)
  \rfloor + c_{\mathrm{SoI}}$;
  (iii) promise-$\cc{NP}$-hardness of the multiplicative $\cc{GapMINcKT}$ for
  $M$, $N$ (factor $\phi \ge 1$) at the slack-amplified clock $\tau$; and
  (iv) either an ordinary estimator satisfying Fact~3.4's sandwich for $M$ at
  the ordinary clock $(n, t) \mapsto p(t)$ on the plan's queries, whose induced two-query
  threshold language is in $\cc{P}$ (or which has an $\cc{FP}$ implementation
  whose output length is the estimate, together with $\cc{FP}$ query and ruler
  builders for the plan), or a Boolean function $d$ with $x \mapsto [d(x)]$ in
  $\cc{FP}$ solving logarithmic $\cc{GapMINKT}$ for $M$ at that clock,
  together with those $\cc{FP}$ builders and the premise that every query of
  the plan has finite complexity on $M$.
  Then $\cc{P} = \cc{NP}$. Equivalently, if $\cc{P} \neq \cc{NP}$, then (i),
  (iii) and each form of (iv) rule out (ii).
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:avg-upper-chain, thm:avg-soi-cancellation, thm:avg-fact34,
    thm:avg-gapminckt-estimator}
  Query the ordinary estimator on $\mathrm{pair}(x,y)$ at clock $p$ and on $y$
  at clock $p^{3}$, subtract with an explicit rounding correction, and use SoI
  and the upper chain to verify both halves of the conditional sandwich (the
  cancellation of Theorem~\ref{thm:avg-soi-cancellation} is redone inline from
  inequality (ii) and the depth decomposition), then apply
  Theorem~\ref{thm:avg-gapminckt-estimator}(c); in the solver form, the
  threshold sweep of Theorem~\ref{thm:avg-fact34}(b) supplies the estimator. The
  final clock is $p^{4}$ amplified by a power-of-two slack that pays all
  logarithmic and additive losses and is proved polynomially bounded. When
  $C^t_M(y)$ or $C^t_N(x \mid y)$ is infinite, the upper chain holds trivially,
  so hypothesis (i) needs no finiteness.

  An earlier formalization demanded finite minima at every instance, an
  estimator correct on every instance, and (inside SoI) a finite joint
  complexity at every clock $t \ge |x| + |y|$. All three are unsatisfiable (no
  machine prints a nonempty string in zero steps,
  Lemma~\ref{lem:avg-no-full-estimator}, and the output-length bound of
  Definition~\ref{def:avg-soi}), and the current statements avoid all three.
  No instance of hypothesis (i) for concrete universal machines is constructed
  yet.
\end{proof}

\begin{theorem}[Hardness of conditional gap MinKT excludes Heuristica]
  \label{thm:avg-hardness-heuristica}
  \uses{def:avg-gapminckt, def:avg-distnp, def:avg-avgp}
  \notready
  If, for the evaluators of Theorem~\ref{thm:avg-evaluator} and every admissible
  polynomial clock, the multiplicative conditional gap problem is
  $\cc{NP}$-hard under the exact reductions of the paper, then
  $\cc{DistNP} \subseteq \cc{AvgP}$ implies $\cc{P} = \cc{NP}$. Published

\end{theorem}
\begin{proof}
  \uses{thm:avg-soi-collapse, thm:avg-distnp-soi, thm:avg-fact34,
    thm:avg-minkt-2018}
  $\cc{DistNP} \subseteq \cc{AvgP}$ gives SoI and, through the paper's
  worst-case to average-case step, an efficient logarithmic $\cc{GapMINKT}$
  solver; apply Theorem~\ref{thm:avg-soi-collapse}.
\end{proof}

\begin{theorem}[Hirahara 2018: worst-case to average-case for $\cc{MINKT}$]

\section{Open directions}

Relativization, oracle-independence, natural-reduction and black-box-reduction
barriers for this program should be recorded as formal oracle implications or
countermodels once the oracle layer supports them. They constrain which proof
interfaces can work and do not license assuming a nonrelativizing result.
